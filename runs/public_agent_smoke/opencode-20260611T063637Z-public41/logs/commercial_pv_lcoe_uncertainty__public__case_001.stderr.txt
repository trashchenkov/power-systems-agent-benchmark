import json, itertools

case = {
    'system_size_kw_dc': 2000,
    'hard_cost_usd_per_kw': 760,
    'soft_cost_components_usd_per_kw': {'permitting': 80, 'interconnection': 95, 'customer_acquisition': 45, 'engineering': 60},
    'capital_recovery_factor': 0.082,
    'fixed_om_usd_per_kw_year': 18,
    'annual_energy_mwh_year1': 3200,
    'soft_cost_scenarios': {
        's_low': {'permitting': 0.75, 'interconnection': 0.8, 'customer_acquisition': 0.85, 'engineering': 0.9},
        's_base': {'permitting': 1.0, 'interconnection': 1.0, 'customer_acquisition': 1.0, 'engineering': 1.0},
        's_perm': {'permitting': 1.35, 'interconnection': 1.05, 'customer_acquisition': 1.1, 'engineering': 1.0},
        's_inter': {'permitting': 1.05, 'interconnection': 1.45, 'customer_acquisition': 1.05, 'engineering': 1.1},
        's_high': {'permitting': 1.4, 'interconnection': 1.5, 'customer_acquisition': 1.25, 'engineering': 1.2}
    }
}

scenario_lcoe = {}
for sid, mult in case['soft_cost_scenarios'].items():
    sum_soft = sum(case['soft_cost_components_usd_per_kw'][k] * v for k, v in mult.items())
    capex = case['system_size_kw_dc'] * (case['hard_cost_usd_per_kw'] + sum_soft)
    num = capex * case['capital_recovery_factor'] + case['fixed_om_usd_per_kw_year'] * case['system_size_kw_dc']
    lcoe = num / case['annual_energy_mwh_year1']
    scenario_lcoe[sid] = round(lcoe, 2)

print('Scenario LCOEs:', scenario_lcoe)

vals = sorted(scenario_lcoe.values())
n = len(vals)

def quantile(vals, p):
    pos = p * (n - 1)
    lo = int(pos)
    hi = min(lo + 1, n - 1)
    frac = pos - lo
    return round(vals[lo] + frac * (vals[hi] - vals[lo]), 2)

p10 = quantile(vals, 0.1)
p50 = quantile(vals, 0.5)
p90 = quantile(vals, 0.9)
mean = round(sum(scenario_lcoe.values()) / n, 2)

print('p10:', p10)
print('p50:', p50)
print('p90:', p90)
print('mean:', mean)

submission = {
    'scenario_lcoe_usd_per_mwh': scenario_lcoe,
    'lcoe_quantiles_usd_per_mwh': {'p10': p10, 'p50': p50, 'p90': p90},
    'expected_lcoe_usd_per_mwh': mean
}
print(json.dumps(submission, indent=2))
