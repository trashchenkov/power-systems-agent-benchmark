sqrt3 = math.sqrt(3)

distances = [0.2, 4.0, 8.0]
labels = ['near_bus', 'mid_feeder', 'remote_end']

I_3ph_vals = []

for d, lab in zip(distances, labels):
    Z_f = complex(R_per_km * d, X_per_km * d)
    Z_total = Z_s + Z_t + Z_f
    Z_mag = abs(Z_total)
