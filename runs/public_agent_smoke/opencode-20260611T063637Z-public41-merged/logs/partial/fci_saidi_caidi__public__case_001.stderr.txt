total_customers = 320
min_restore = 25
deployed = {'n2', 'n4'}

fault_sections = {
    's12': {'rate': 0.18, 'cust': 120, 'base': 95},
    's23': {'rate': 0.14, 'cust': 70, 'base': 80},

}
