import math, json

current_limit_pu = 1.2
prefault_id_pu = 0.92
deadband = 0.9
k = 2.0
floor = 0.45
max_dur = 0.2

samples = [
    ('t000', 0.0, 0.88),
    ('t080', 0.08, 0.52),
    ('t160', 0.16, 0.48),
    ('t240', 0.24, 0.74),
]

profile = {}
for sid, t, v in samples:
    if v < deadband:
        iq_req = k * (deadband - v)
        iq = min(iq_req, current_limit_pu)
    else:
        iq = 0.0
    id_max = math.sqrt(current_limit_pu**2 - iq**2)
    id_val = min(prefault_id_pu, id_max)
    total = math.sqrt(id_val**2 + iq**2)
    profile[sid] = {
        'id_pu': round(id_val, 6),
        'iq_pu': round(iq, 6),
        'total_current_pu': round(total, 6)
    }
    print(f'{sid}: iq={iq:.6f}, id_max={id_max:.6f}, id={id_val:.6f}, total={total:.6f}')

iq_vals = [profile[s]['iq_pu'] for s in profile]
tot_vals = [profile[s]['total_current_pu'] for s in profile]
min_iq = min(iq_vals)
max_tot = max(tot_vals)
print(f'min_iq_pu = {min_iq}')
print(f'max_total_pu = {max_tot}')

# FRT compliance: all voltages >= floor? reactive current met?
v_allowed = all(v >= floor for _, _, v in samples)
print(f'All above floor: {v_allowed}')

# duration: V<0.9 starts at t=0, at t=0.24 still <0.9 => duration >= 0.24 > 0.2
event_duration = 0.24  # at least
