import json, math

with open("task.json") as f:
    task = json.load(f)

net = task["network"]
buses = net["buses"]
n = len(buses)
bus_idx = {b: i for i, b in enumerate(buses)}

# Ybus as complex matrix
Y = [[0j] * n for _ in range(n)]

# Branches
for br in net["branches"]:
    i = bus_idx[br["from_bus"]]
    j = bus_idx[br["to_bus"]]
    r = br["r_pu"]
    x = br["x_pu"]
    b = br["b_pu"]
    z = complex(r, x)
    y = 1.0 / z
    y_sh = 1j * b / 2.0
    Y[i][i] += y + y_sh
    Y[j][j] += y + y_sh
    Y[i][j] += -y
    Y[j][i] += -y

# Shunts
for sh in net["shunts"]:
    i = bus_idx[sh["bus"]]
    Y[i][i] += complex(sh["g_pu"], sh["b_pu"])

# Build submission
def to_entry(z):
    return {"re": round(z.real, 12), "im": round(z.imag, 12)}

ybus = [[to_entry(Y[i][j]) for j in range(n)] for i in range(n)]

sub = {
    "buses": buses,
    "ybus": ybus,
    "rationale": "Constructed Ybus per instructions: y=1/(r+jx), y+jb/2 on diagonals, -y off-diagonals, plus explicit shunts."
}
with open("submission.json", "w") as f:
    json.dump(sub, f, indent=2)

# Print for verification
for i in range(n):
    for j in range(n):
        print(f"Y[{i}][{j}] = {Y[i][j]}")
