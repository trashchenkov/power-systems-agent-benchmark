import json, math
from pathlib import Path
case=json.loads(Path('task.json').read_text())['wind_interval_case']
ref=case['reference_air_density_kg_m3']
count=case['turbine']['count']; rated=case['turbine']['rated_mw_each']; cap=count*rated
pc=case['power_curve']

def power(s):
    vc=s['wind_speed_m_s']*((s['air_density_kg_m3']/ref)**(1/3))
    if vc < pc['cut_in_m_s'] or vc >= pc['cut_out_m_s']:
        p=0.0
    elif vc >= pc['rated_m_s']:
        p=cap
    else:
        p=cap*((vc-pc['cut_in_m_s'])/(pc['rated_m_s']-pc['cut_in_m_s']))**3
    return p, vc

def q(xs, qq):
    xs=sorted(xs); n=len(xs); pos=qq*(n-1); lo=math.floor(pos); hi=math.ceil(pos)
    if lo==hi: return xs[lo]
    return xs[lo] + (xs[hi]-xs[lo])*(pos-lo)

out={}
widths=[]
for h in case['horizons']:
    vals=[power(s)[0] for s in h['scenarios']]
    qs={f'p{int(qq*100)}': q(vals, qq) for qq in [0.1,0.5,0.9]}
    out[h['timestamp']]=qs
    widths.append(qs['p90']-qs['p10'])
    print(h['timestamp'])
    print('powers', vals)
    print(qs)
print('avg', sum(widths)/len(widths))
print(json.dumps({'prediction_interval_mw_by_timestamp': out, 'average_interval_width_mw': sum(widths)/len(widths)}, indent=2))
