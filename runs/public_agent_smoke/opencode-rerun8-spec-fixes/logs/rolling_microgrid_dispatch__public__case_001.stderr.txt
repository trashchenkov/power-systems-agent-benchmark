dt = 0.0833333333  # 5 minutes in hours

# Try schedule: battery discharge (8,10,8,0) -> sum=26, diesel (12,17,29,35)

    dsl = net - bat_mw[t]
    soc_change = -bat_mw[t] * dt / bat["disch_eff"] if bat_mw[t] > 0 else -bat_mw[t] * dt * bat["ch_eff"]
