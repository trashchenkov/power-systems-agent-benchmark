# Microgrid Economic Dispatch
load = {"t1": 45.0, "t2": 62.0, "t3": 50.0}
pv = {"t1": 10.0, "t2": 8.0, "t3": 18.0}
gen = {"pmin": 15.0, "pmax": 70.0, "cost": 85.0}
bat = {"init_soc": 18.0, "soc_min": 6.0, "soc_max": 30.0,
       "ch_pmax": 8.0, "dis_pmax": 12.0, "ch_eff": 0.95, "dis_eff": 0.9, "step": 2.0}

# Brute-force enumerate all feasible battery schedules
import itertools

valid_battery_values = [round(v, 1) for v in 
    [i * bat["step"] for i in range(-int(bat["ch_pmax"]/bat["step"]), int(bat["dis_pmax"]/bat["step"])+1)]]

print("Valid battery values:", valid_battery_values)

best_cost = float('inf')
best_schedules = []

for b1, b2, b3 in itertools.product(valid_battery_values, repeat=3):
    battery = {"t1": b1, "t2": b2, "t3": b3}
    
    # Simulate SOC
    soc = bat["init_soc"]
    feasible = True
    schedule = {}
    
    for t in ["t1", "t2", "t3"]:
        b = battery[t]
        # SOC change
        if b > 0:  # discharge
            soc_change = -b / bat["dis_eff"]
        elif b < 0:  # charge
            soc_change = -b * bat["ch_eff"]  # -(-|b|) = positive
        else:
            soc_change = 0
        
        soc_new = soc + soc_change
        
        if soc_new < bat["soc_min"] or soc_new > bat["soc_max"]:
            feasible = False
            break
        
        # Power balance: diesel + pv - pv_curtailment + battery = load
        # pv_curtailment = 0 (no need to curtail), shed = 0 (not allowed)
        diesel = load[t] - pv[t] - b
        
        if diesel < gen["pmin"] or diesel > gen["pmax"]:
            feasible = False
            break
        
        schedule[t] = {"diesel": diesel, "battery": b, "soc": soc_new}
        soc = soc_new
    
    if feasible:
        total_diesel = sum(schedule[t]["diesel"] for t in ["t1", "t2", "t3"])
        total_cost = total_diesel * gen["cost"]
        
        if total_cost < best_cost - 1e-6:
            best_cost = total_cost
            best_schedules = [(battery, schedule, total_cost)]
        elif abs(total_cost - best_cost) < 1e-6:
            best_schedules.append((battery, schedule, total_cost))

print(f"Optimal cost: ${best_cost:.2f}")
print(f"Number of optimal schedules: {len(best_schedules)}")
for bat_sched, sched, cost in best_schedules[:5]:
    print(f"  Battery: {bat_sched}")
    for t in ["t1", "t2", "t3"]:
        print(f"    {t}: diesel={sched[t]['diesel']}, battery={sched[t]['battery']}, soc={sched[t]['soc']:.6f}")
    print(f"    total_cost={cost}")

# Chosen schedule: discharge 10 MW in t1 only

for t in ["t1", "t2", "t3"]:
    b = battery_mw[t]
    if b > 0:
        soc_delta = -b / 0.9

        soc_delta = -b * 0.95
    else:
        soc_delta = 0.0

        "battery_mw": b,
