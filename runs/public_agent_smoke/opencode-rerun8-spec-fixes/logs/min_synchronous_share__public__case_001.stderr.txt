nom_freq = 50.0
largest_loss = 180.0
max_rocof = 2.0

units = {
    "g1": {"p": 120.0, "h": 4.5, "sc": 280.0},
    "g2": {"p": 180.0, "h": 5.0, "sc": 360.0},
    "g3": {"p": 240.0, "h": 3.8, "sc": 440.0},
    "g4": {"p": 90.0, "h": 6.0, "sc": 190.0},
}

import itertools

feasible = []
for r in range(1, len(units)+1):
    for combo in itertools.combinations(units.keys(), r):
        total_p = sum(units[u]["p"] for u in combo)
        total_hp = sum(units[u]["h"] * units[u]["p"] for u in combo)
        inertia_gws = total_hp / 1000.0
        rocof = largest_loss * nom_freq / (2 * inertia_gws * 1000)
        total_sc = sum(units[u]["sc"] for u in combo)
        scr = (grid_sc_base + total_sc) / ibr_capacity
        share_pct = total_p / demand * 100
