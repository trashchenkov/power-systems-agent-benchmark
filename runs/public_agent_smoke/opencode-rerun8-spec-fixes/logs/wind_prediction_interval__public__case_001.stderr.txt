import json, math

with open('task.json') as f:
    task = json.load(f)

case = task['wind_interval_case']
ref_dens = case['reference_air_density_kg_m3']
count = case['turbine']['count']
rated = case['turbine']['rated_mw_each']
cut_in = case['power_curve']['cut_in_m_s']
rated_ms = case['power_curve']['rated_m_s']
cut_out = case['power_curve']['cut_out_m_s']

def power(wind_speed, air_density):
    v_c = wind_speed * (air_density / ref_dens) ** (1/3)
    if v_c < cut_in or v_c >= cut_out:
        return 0.0
    if v_c >= rated_ms:
        return count * rated
    return count * rated * ((v_c - cut_in) / (rated_ms - cut_in)) ** 3

def quantile(sorted_vals, q):
    n = len(sorted_vals)
    pos = q * (n - 1)
    lo = int(math.floor(pos))
    hi = int(math.ceil(pos))
    if lo == hi:
        return sorted_vals[lo]
    frac = pos - lo
    return sorted_vals[lo] + frac * (sorted_vals[hi] - sorted_vals[lo])

results = {}
widths = []
for h in case['horizons']:
    ts = h['timestamp']
    powers = sorted([power(s['wind_speed_m_s'], s['air_density_kg_m3']) for s in h['scenarios']])
    p10 = quantile(powers, 0.1)
    p50 = quantile(powers, 0.5)
    p90 = quantile(powers, 0.9)
    results[ts] = {'p10': round(p10, 4), 'p50': round(p50, 4), 'p90': round(p90, 4)}
    widths.append(p90 - p10)
    print(f'{ts}: powers={powers} -> p10={p10:.4f} p50={p50:.4f} p90={p90:.4f}')

avg_width = sum(widths) / len(widths)
print(f'average_interval_width_mw = {avg_width:.4f}')
print(json.dumps({'prediction_interval_mw_by_timestamp': results, 'average_interval_width_mw': round(avg_width, 4)}, indent=2))
