import json

tc = 320
min_res = 25

fault_sections = {
    "s12": {"annual_fault_rate": 0.18, "customers_interrupted": 120, "base_restoration_minutes": 95},
    "s23": {"annual_fault_rate": 0.14, "customers_interrupted": 70, "base_restoration_minutes": 80},
    "s24": {"annual_fault_rate": 0.12, "customers_interrupted": 90, "base_restoration_minutes": 85},
    "s45": {"annual_fault_rate": 0.09, "customers_interrupted": 40, "base_restoration_minutes": 75},
}

deployed = ["n2", "n4"]

fci_locations = {
    "n2": {"restoration_time_reduction_min": {"s12": 6, "s23": 28, "s24": 26, "s45": 23}},
    "n4": {"restoration_time_reduction_min": {"s12": 0, "s23": 4, "s24": 18, "s45": 34}},
}

# Baseline
baseline_cust_min = sum(
    s["annual_fault_rate"] * s["customers_interrupted"] * s["base_restoration_minutes"]
    for s in fault_sections.values()
)
baseline_cust_int = sum(
    s["annual_fault_rate"] * s["customers_interrupted"]
    for s in fault_sections.values()
)

baseline_saidi = baseline_cust_min / tc
baseline_caidi = baseline_cust_min / baseline_cust_int
baseline_saifi = baseline_cust_int / tc

print(f"Baseline customer-minutes: {baseline_cust_min}")
print(f"Baseline customer-interruptions: {baseline_cust_int}")
print(f"Baseline SAIDI: {baseline_saidi}")
print(f"Baseline CAIDI: {baseline_caidi}")
print(f"Baseline SAIFI: {baseline_saifi}")

# With FCI
fci_cust_min = 0
for sec_id, sec in fault_sections.items():
    best_reduction = max(
        fci_locations[d][ "restoration_time_reduction_min"][sec_id]
        for d in deployed
    )
    new_time = max(sec["base_restoration_minutes"] - best_reduction, min_res)
    contrib = sec["annual_fault_rate"] * sec["customers_interrupted"] * new_time
    fci_cust_min += contrib
    print(f"{sec_id}: best_reduction={best_reduction}, new_time={new_time}, contrib={contrib}")

fci_saidi = fci_cust_min / tc
fci_caidi = fci_cust_min / baseline_cust_int
fci_saifi = baseline_cust_int / tc

print(f"\nFCI customer-minutes: {fci_cust_min}")
print(f"FCI SAIDI: {fci_saidi}")
print(f"FCI CAIDI: {fci_caidi}")
print(f"FCI SAIFI: {fci_saifi}")

saidi_improvement = baseline_saidi - fci_saidi
caidi_improvement = baseline_caidi - fci_caidi

print(f"\nSAIDI improvement: {saidi_improvement}")
print(f"CAIDI improvement: {caidi_improvement}")

# Build submission
submission = {
    "baseline": {
        "saidi_min_per_customer_year": round(baseline_saidi, 6),
        "caidi_min_per_interruption": round(baseline_caidi, 6),
        "saifi_interruptions_per_customer_year": round(baseline_saifi, 6)
    },
    "with_fci": {
        "saidi_min_per_customer_year": round(fci_saidi, 6),
        "caidi_min_per_interruption": round(fci_caidi, 6),
        "saifi_interruptions_per_customer_year": round(fci_saifi, 6)
    },
    "saidi_improvement_min_per_customer_year": round(saidi_improvement, 6),
    "caidi_improvement_min_per_interruption": round(caidi_improvement, 6),
    "rationale": "Baseline and FCI reliability indices computed per spec."
}

print("\n--- submission.json ---")
print(json.dumps(submission, indent=2))
