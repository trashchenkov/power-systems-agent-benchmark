import math

# Inputs
I = 180.0            # design current A
pf = 0.9             # power factor
V = 400.0            # system voltage V
L = 250.0            # length m
max_drop = 5.0       # max voltage drop %
ambient = 0.96       # ambient factor
grouping = 0.8       # grouping factor
I_fault = 12000.0    # fault current A
t = 0.4              # clearing time s
k = 143.0            # adiabatic constant

# Combined derating factor
derating = ambient * grouping
print(f"Derating factor: {derating}")

# Minimum thermal section
S_min = I_fault * math.sqrt(t) / k
print(f"S_min (thermal): {S_min:.2f} mm²")

sin_phi = math.sqrt(1 - pf**2)

catalog = [
    (50,  192.0, 0.387, 0.085),
    (70,  246.0, 0.268, 0.082),
    (95,  298.0, 0.193, 0.082),
    (120, 346.0, 0.153, 0.080),
    (150, 399.0, 0.124, 0.080),
    (185, 456.0, 0.0991, 0.080),
]

for cs, base_r, r, x in catalog:
    derated = base_r * derating
    amp_ok = derated >= I
    
    r_comp = r * pf + x * sin_phi
    dU = math.sqrt(3) * I * (L / 1000) * r_comp
    dU_pct = dU / V * 100
    vd_ok = dU_pct <= max_drop
    
    therm_ok = cs >= S_min
    
    status = "PASS" if (amp_ok and vd_ok and therm_ok) else "FAIL"
    print(f"\n{cs} mm²: derated={derated:.3f}A {'OK' if amp_ok else 'FAIL'}, "
          f"dU={dU_pct:.3f}% {'OK' if vd_ok else 'FAIL'}, "
          f"therm={cs}≥{S_min:.2f}mm² {'OK' if therm_ok else 'FAIL'} → {status}")

# The first passing
for cs, base_r, r, x in catalog:
    derated = base_r * derating
    r_comp = r * pf + x * sin_phi
    dU = math.sqrt(3) * I * (L / 1000) * r_comp
    dU_pct = dU / V * 100
    if derated >= I and dU_pct <= max_drop and cs >= S_min:
        print(f"\n\nSelected: {cs} mm²")
        print(f"Derated ampacity: {derated:.3f} A")
        print(f"Voltage drop: {dU_pct:.6f}%")
        print(f"Min thermal section: {S_min:.6f} mm²")
        break
