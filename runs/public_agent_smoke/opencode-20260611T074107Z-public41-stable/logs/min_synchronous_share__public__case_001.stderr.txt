import itertools

demand = 1000.0
ibr_mva = 720.0
fnom = 50.0
loss = 180.0
max_rocof = 2.0
min_scr = 2.25
grid_sc = 1180.0

units = [
    ('g1', 120.0, 4.5, 280.0),
    ('g2', 180.0, 5.0, 360.0),
    ('g3', 240.0, 3.8, 440.0),
    ('g4', 90.0, 6.0, 190.0),
]

feasible = []
for r in range(1, 5):
    for combo in itertools.combinations(units, r):
        ids = [u[0] for u in combo]
        p_total = sum(u[1] for u in combo)
        inertia = sum(u[1] * u[2] for u in combo) / 1000.0
        rocof = loss * fnom / (2 * inertia * 1000)
        sc_total = sum(u[3] for u in combo)
        scr = (grid_sc + sc_total) / ibr_mva
        share = p_total / demand * 100
        
        rocof_ok = rocof <= max_rocof
        scr_ok = scr >= min_scr
        feasible.append((p_total, share, ids, inertia, rocof, scr, rocof_ok and scr_ok))
        
        print(f'{ids}: P={p_total:.0f} share={share:.2f}% inertia={inertia:.4f} rocof={rocof:.4f} scr={scr:.4f} {\"FEASIBLE\" if rocof_ok and scr_ok else \"\"}')

print()
feasible_pass = [f for f in feasible if f[6]]
feasible_pass.sort(key=lambda x: x[0])
print('Minimum feasible combination:')
for f in feasible_pass[:1]:
    print(f)
