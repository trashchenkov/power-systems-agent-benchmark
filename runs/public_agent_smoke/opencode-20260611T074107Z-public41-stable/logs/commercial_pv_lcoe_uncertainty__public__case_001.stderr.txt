# quantiles using numpy-style linear interpolation
import math
values = sorted(scenario_lcoe.values())
n = len(values)

def quantile(p):
    idx = p * (n - 1)
    lo = int(math.floor(idx))
    hi = int(math.ceil(idx))
    if lo == hi:
        return values[lo]
    frac = idx - lo
    return values[lo] * (1 - frac) + values[hi] * frac

p10 = round(quantile(0.1), 6)
p50 = round(quantile(0.5), 6)
p90 = round(quantile(0.9), 6)
