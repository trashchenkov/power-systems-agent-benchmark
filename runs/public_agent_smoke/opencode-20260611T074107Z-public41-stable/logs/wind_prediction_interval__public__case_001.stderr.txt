def single_turbine_power(v):
    """Power from a single turbine given corrected wind speed."""
    if v < cut_in or v > cut_out:
        return 0.0
    if v >= rated_v:
        return P_rated
    # Cubic region between cut_in and rated
    p = P_rated * (v**3 - cut_in**3) / (rated_v**3 - cut_in**3)
    return p

def farm_power(v, rho):
    """Farm total power with air density correction."""
    v_corr = v * (rho / ref_rho) ** (1/3)
    return single_turbine_power(v_corr) * n_turbines

        p = farm_power(v, rho)
        powers.append(p)
    powers.sort()
    n = len(powers)
    # p10: 10th percentile, p50: 50th, p90: 90th
    # For 5 samples, linear interpolation between order statistics
    def percentile(sorted_data, p):
        # Using linear interpolation method
        n = len(sorted_data)
        if n == 0:
            return 0.0
        index = p / 100.0 * (n - 1)
        lo = int(math.floor(index))
        hi = int(math.ceil(index))
        if lo == hi:
            return sorted_data[lo]
        frac = index - lo
        return sorted_data[lo] * (1 - frac) + sorted_data[hi] * frac
    
    p10 = percentile(powers, 10)
    p50 = percentile(powers, 50)

import json, math
with open("task.json") as f:
    task = json.load(f)
case = task["wind_interval_case"]
ref_rho = case["reference_air_density_kg_m3"]
n_turb = case["turbine"]["count"]
P_rated = case["turbine"]["rated_mw_each"]
cut_in = case["power_curve"]["cut_in_m_s"]
rated_v = case["power_curve"]["rated_m_s"]
cut_out = case["power_curve"]["cut_out_m_s"]

def tp(v):
    if v < cut_in or v > cut_out: return 0.0
    if v >= rated_v: return P_rated
    return P_rated * (v**3 - cut_in**3) / (rated_v**3 - cut_in**3)

def fp(v, rho):
    return tp(v * (rho / ref_rho) ** (1/3)) * n_turb

def q(d, p):
    n, i = len(d), p / 100.0 * (len(d) - 1)
    l, h = int(math.floor(i)), int(math.ceil(i))
    if l == h: return d[l]
    return d[l] * (1 - i + l) + d[h] * (i - l)

r = {}
for h in case["horizons"]:
    ts = h["timestamp"]
    pws = sorted(fp(s["wind_speed_m_s"], s["air_density_kg_m3"]) for s in h["scenarios"])
    r[ts] = {k: round(q(pws, {"p10": 10, "p50": 50, "p90": 90}[k]), 6) for k in ("p10", "p50", "p90")}

aw = sum(r[ts]["p90"] - r[ts]["p10"] for ts in r) / len(r)
sub = {"prediction_interval_mw_by_timestamp": r, "average_interval_width_mw": round(aw, 6)}
with open("submission.json", "w") as f:
