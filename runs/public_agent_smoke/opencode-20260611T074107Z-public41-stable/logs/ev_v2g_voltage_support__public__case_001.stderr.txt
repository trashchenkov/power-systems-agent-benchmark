import math, json

# data
target = 0.98
base = {"bus_2": 0.944, "bus_3": 0.952}
sens = {"bus_2": 0.0016, "bus_3": 0.0012}

evs = {
    "evse_1": {"bus": "bus_2", "s": 18.0, "p": 10.0},
    "evse_2": {"bus": "bus_2", "s": 12.0, "p": 6.0},
    "evse_3": {"bus": "bus_3", "s": 16.0, "p": 8.0},
}

# aggregate Q capability per bus
q_cap = {}
for name, ev in evs.items():
    b = ev["bus"]
    q = math.sqrt(ev["s"]**2 - ev["p"]**2)
    q_cap[b] = q_cap.get(b, 0.0) + q

print("Q capability per bus:", q_cap)

# compute dispatch
q_dispatch = {}
corrected_v = {}
for bus in ["bus_2", "bus_3"]:
    v_def = target - base[bus]
    q_needed = v_def / sens[bus]
    q_actual = min(q_needed, q_cap[bus])
    q_dispatch[bus] = round(q_actual, 6)
    corrected_v[bus] = round(base[bus] + sens[bus] * q_actual, 6)

total_abs_q = round(abs(q_dispatch["bus_2"]) + abs(q_dispatch["bus_3"]), 6)

print("\nq_dispatch:", q_dispatch)
print("corrected_voltage_pu:", corrected_v)
print("total_abs_q_kvar:", total_abs_q)

submission = {
    "q_dispatch_kvar": q_dispatch,
    "corrected_voltage_pu": corrected_v,
    "total_abs_q_kvar": total_abs_q,
    "rationale": "Computed reactive dispatch from linear voltage sensitivity clipped by aggregate EV inverter apparent-power capability."
}
print("\n--- submission.json ---")
print(json.dumps(submission, indent=2))
