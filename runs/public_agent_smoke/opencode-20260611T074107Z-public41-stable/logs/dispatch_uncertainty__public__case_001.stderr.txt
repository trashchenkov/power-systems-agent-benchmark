import json

# Generators
g1 = {'pmin': 10, 'pmax': 70, 'cost': 18}
g2 = {'pmin': 5, 'pmax': 45, 'cost': 36}

# Scenarios
scenarios = {
    'normal':       {'load': {'t1':78,'t2':85,'t3':75}, 'ren': {'t1':18,'t2':24,'t3':20}},
    'high_load':    {'load': {'t1':84,'t2':92,'t3':82}, 'ren': {'t1':12,'t2':16,'t3':13}},
    'low_renewable':{'load': {'t1':80,'t2':87,'t3':77}, 'ren': {'t1':8,'t2':10,'t3':9}},
}

reserve = 8.0

# Max net load per period
times = ['t1','t2','t3']
max_net = {}
for t in times:
    vals = [scenarios[s]['load'][t] - scenarios[s]['ren'][t] for s in scenarios]
    max_net[t] = max(vals)

print('Max net load:', max_net)

# Required generation
req = {t: max_net[t] + reserve for t in times}
print('Required gen:', req)

# Least-cost dispatch: maximize cheap g1 (18), use g2 (36) for remainder
dispatch = {'g1': {}, 'g2': {}}
total_cost = 0
margin = {}
for t in times:
    need = req[t]
    g1_disp = min(g1['pmax'], need)
    g2_disp = need - g1_disp
    # Check pmin
    assert g2_disp >= g2['pmin'] or g2_disp == 0, f'g2 pmin violated at {t}'
    assert g1_disp >= g1['pmin'] or g1_disp == 0, f'g1 pmin violated at {t}'
    dispatch['g1'][t] = g1_disp
    dispatch['g2'][t] = g2_disp
    total_cost += g1_disp * g1['cost'] + g2_disp * g2['cost']
    margin[t] = g1_disp + g2_disp - max_net[t]

print('Dispatch:', dispatch)
print('Margin:', margin)
print('Total cost:', total_cost)

submission = {
    'dispatch_mw': dispatch,
    'worst_case_margin_mw': margin,
    'total_cost': total_cost,
    'rationale': 'g1 (18/MWh) dispatched at pmax 70 each period; g2 (36/MWh) covers the remainder. All periods satisfy max net load + 8 MW reserve at minimum variable cost.'
}

with open('submission.json', 'w') as f:
    json.dump(submission, f, indent=2)
