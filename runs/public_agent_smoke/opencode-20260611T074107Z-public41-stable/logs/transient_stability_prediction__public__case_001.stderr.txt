cases = {
    'fault_a': {'fct': 0.1, 'cct': 0.17, 'angle': 142.0, 'vmin': 0.76, 'damp': 5.0},
    'fault_b': {'fct': 0.16, 'cct': 0.14, 'angle': 188.0, 'vmin': 0.68, 'damp': 2.2},
    'fault_c': {'fct': 0.08, 'cct': 0.11, 'angle': 171.0, 'vmin': 0.72, 'damp': 3.4},
}
angle_lim = 180.0
vfloor = 0.7
damp_min = 3.0

results = {}
for cid, v in cases.items():
    margin = round(v['cct'] - v['fct'], 4)
    r_angle = v['angle'] / angle_lim
    r_volt = vfloor / v['vmin']
    r_damp = damp_min / v['damp']
    r_cct = v['fct'] / v['cct']
    risk = round(max(r_angle, r_volt, r_damp, r_cct), 4)
    
    stable = (v['angle'] <= angle_lim and v['vmin'] >= vfloor and v['damp'] >= damp_min and margin >= 0)
    label = 'stable' if stable else 'unstable'
    
    results[cid] = {'label': label, 'risk_score': risk, 'clearing_margin_s': margin}
    print(f'{cid}: label={label}, risk={risk}, margin={margin}')
    print(f'  ratios: angle={r_angle:.4f} volt={r_volt:.4f} damp={r_damp:.4f} cct={r_cct:.4f}')

stable_count = sum(1 for r in results.values() if r['label'] == 'stable')
highest_risk = max(results.items(), key=lambda x: x[1]['risk_score'])[0]
print(f'stable_count={stable_count}, highest_risk={highest_risk}')

import json
sub = {
    'case_results': results,
    'stable_case_count': stable_count,
    'highest_risk_case': highest_risk,
}
print(json.dumps(sub, indent=2))
