import json, math

inputs = {
    'current_limit_pu': 1.2,
    'prefault_active_current_pu': 0.92,
    'reactive_deadband_pu': 0.9,
    'k_reactive_current_pu_per_pu': 2.0,
    'frt_voltage_floor_pu': 0.45,
    'max_low_voltage_duration_s': 0.2,
    'samples': [
        {'sample_id': 't000', 'time_s': 0.0, 'voltage_pu': 0.88},
        {'sample_id': 't080', 'time_s': 0.08, 'voltage_pu': 0.52},
        {'sample_id': 't160', 'time_s': 0.16, 'voltage_pu': 0.48},
        {'sample_id': 't240', 'time_s': 0.24, 'voltage_pu': 0.74},
    ]
}

I_max = inputs['current_limit_pu']
id_pre = inputs['prefault_active_current_pu']
V_db = inputs['reactive_deadband_pu']
k = inputs['k_reactive_current_pu_per_pu']
V_floor = inputs['frt_voltage_floor_pu']

profile = {}
for s in inputs['samples']:
    sid = s['sample_id']
    V = s['voltage_pu']
    
    if V >= V_db:
        iq = 0.0
    else:
        iq = k * (V_db - V)
        iq = min(iq, I_max)
    
    id_avail = math.sqrt(max(0, I_max**2 - iq**2))
    id = min(id_pre, id_avail)
    
    itot = math.sqrt(id**2 + iq**2)
    
    profile[sid] = {'id_pu': round(id, 6), 'iq_pu': round(iq, 6), 'total_current_pu': round(itot, 6)}

min_iq = min(v['iq_pu'] for v in profile.values())
max_itot = max(v['total_current_pu'] for v in profile.values())

# FRT compliance check
# All voltages above floor?
voltages = [s['voltage_pu'] for s in inputs['samples']]
times = [s['time_s'] for s in inputs['samples']]
above_floor = all(v >= V_floor for v in voltages)

# Low voltage duration: time when V < deadband
# Since V < deadband from t=0 to at least t=0.24 (last sample still below 0.9)
# The low-voltage period spans at least 0.24s
low_voltage_duration = max(times) - min(times)
low_voltage_duration_exceeds = low_voltage_duration > inputs['max_low_voltage_duration_s']

frt_compliant = above_floor and not low_voltage_duration_exceeds

print('Profile:', json.dumps(profile, indent=2))
print('min_iq:', min_iq)
print('max_itot:', max_itot)
print('above_floor:', above_floor)
print('low_voltage_duration:', low_voltage_duration)
print('frt_compliant:', frt_compliant)
