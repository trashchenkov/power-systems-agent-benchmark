import json, itertools

load = {'t1': 45.0, 't2': 62.0, 't3': 50.0}
pv = {'t1': 10.0, 't2': 8.0, 't3': 18.0}
periods = ['t1', 't2', 't3']
pmin, pmax, cost_per = 15.0, 70.0, 85.0
init_soc, soc_min, soc_max = 18.0, 6.0, 30.0
ch_pmax, disch_pmax = 8.0, 12.0
ch_eff, disch_eff = 0.95, 0.9
step = 2.0

b_vals = [round(i*step,1) for i in range(int(-ch_pmax/step), int(disch_pmax/step)+1)]

best_cost = float('inf')
best = None

for b1,b2,b3 in itertools.product(b_vals, repeat=3):
    batt = {'t1':b1,'t2':b2,'t3':b3}
    prev_soc = init_soc
    feas = True
    dies = {}
    curt = {}
    socs = {}
    for t in periods:
        d = load[t] - pv[t] - batt[t]
        if d < pmin:
            nc = pmin - d
            if nc <= pv[t]:
                curt[t] = nc
                d = pmin
            else:
                feas = False; break
        elif d > pmax:
            feas = False; break
        else:
            curt[t] = 0.0
        dies[t] = d
        if batt[t] > 0:
            socs[t] = prev_soc - batt[t] / disch_eff
        elif batt[t] < 0:
            socs[t] = prev_soc + (-batt[t]) * ch_eff
        else:
            socs[t] = prev_soc
        if socs[t] < soc_min - 1e-9 or socs[t] > soc_max + 1e-9:
            feas = False; break
        prev_soc = socs[t]
    if not feas:

    total_d = sum(dies[t] for t in periods)
    cost = total_d * cost_per

                    'shed_mw': 0.0
