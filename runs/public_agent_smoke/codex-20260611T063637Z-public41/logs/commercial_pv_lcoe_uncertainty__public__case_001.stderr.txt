}
def lcoe(mult):
    soft=sum(components[k]*mult[k] for k in components)
    capex=system_size*(hard+soft)
    return (capex*crf+fom*system_size)/energy
vals={sid:lcoe(m) for sid,m in scenarios.items()}

def q(p):
    pos=(len(sorted_vals)-1)*p
    lo=int(pos)
    hi=min(lo+1,len(sorted_vals)-1)
    frac=pos-lo
    return sorted_vals[lo]*(1-frac)+sorted_vals[hi]*frac
