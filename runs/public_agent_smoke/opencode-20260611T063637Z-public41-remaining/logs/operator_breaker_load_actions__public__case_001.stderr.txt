# Base flows
flows = {"l12": 118, "l23": 72, "l13": 65}
limits = {"l12": 100, "l23": 90, "l13": 80}

# Candidate actions
actions = {
    "open_tie_13": {"type": "open_breaker", "mw": 0, "delta": {"l12": -8, "l23": 12}},
    "shed_bus3_15": {"type": "shed_load", "mw": 15, "delta": {"l12": -14, "l23": -6}},
    "shed_bus2_10": {"type": "shed_load", "mw": 10, "delta": {"l12": -9, "l23": -2}},
}

from itertools import combinations

best = None

for r in range(1, 3):
    for combo in combinations(actions.keys(), r):
        f = dict(flows)
        shed = 0
        for a in combo:
            info = actions[a]
            if info["type"] == "open_breaker":
                f["l13"] = 0
            else:
                shed += info["mw"]
            for branch, delta in info["delta"].items():
                f[branch] += delta

        ok = all(f[b] <= limits[b] for b in limits)
        if ok:
            score = (shed, len(combo))
            print(f"FEASIBLE: {combo} -> flows={f}, shed={shed}MW, score={score}")
            if best is None or score < best["score"]:
                best = {"actions": combo, "flows": f, "shed": shed, "score": score}
        else:
            print(f"INFEASIBLE: {combo} -> flows={f}, shed={shed}MW")

if best:
    print(f"\nOPTIMAL: {best['actions']}, flows={best['flows']}, shed={best['shed']}MW")
