import itertools

demand_mw = 1000.0
ibr_capacity_mva = 720.0
nominal_freq = 50.0
largest_loss_mw = 180.0
max_rocof = 2.0
min_scr = 2.25
grid_sc_base = 1180.0

units = [
    {"id": "g1", "p": 120.0, "h": 4.5, "sc": 280.0},
    {"id": "g2", "p": 180.0, "h": 5.0, "sc": 360.0},
    {"id": "g3", "p": 240.0, "h": 3.8, "sc": 440.0},
    {"id": "g4", "p": 90.0, "h": 6.0, "sc": 190.0},
]

best = None
best_share = float('inf')

for r in range(1, len(units) + 1):
    for combo in itertools.combinations(units, r):
        ids = [u["id"] for u in combo]
        total_p = sum(u["p"] for u in combo)
        inertia_gws = sum(u["h"] * u["p"] for u in combo) / 1000.0
        rocof = largest_loss_mw * nominal_freq / (2 * inertia_gws * 1000)
        total_sc = grid_sc_base + sum(u["sc"] for u in combo)
        scr = total_sc / ibr_capacity_mva
        share_pct = total_p / demand_mw * 100

        feasible = rocof <= max_rocof and scr >= min_scr
        status = "FEASIBLE" if feasible else "INFEASIBLE"

        print(f"{ids}: P={total_p:.0f}MW, H={inertia_gws:.4f}GWs, RoCoF={rocof:.4f}Hz/s, SCR={scr:.4f}, share={share_pct:.2f}% [{status}]")

        if feasible and share_pct < best_share:
            best_share = share_pct
            best = (ids, total_p, inertia_gws, rocof, scr, share_pct)

print(f"\nBEST: {best}")
