# Build graph and analyze contingencies
branches = {
    "l12": ("b1", "b2", 100.0),
    "l23": ("b2", "b3", 100.0),
    "l13": ("b1", "b3", 70.0),
    "l34": ("b3", "b4", 60.0),
}

contingencies = {
    "out_l12": {"l23": 95.0, "l13": 88.0, "l34": 30.0},
    "out_l23": {"l12": 91.0, "l13": 55.0, "l34": 30.0},
    "out_l13": {"l12": 99.0, "l23": 80.0, "l34": 30.0},
    "out_l34": {"l12": 80.0, "l23": 60.0, "l13": 40.0},
}

outage_map = {
    "out_l12": "l12",
    "out_l23": "l23",
    "out_l13": "l13",
    "out_l34": "l34",
}

threshold = 100.0

def check_islanding(outaged_branch):
    # Adjacency
    adj = {b: set() for b in ["b1", "b2", "b3", "b4"]}
    for bid, (f, t, r) in branches.items():
        if bid == outaged_branch:
            continue
        adj[f].add(t)
        adj[t].add(f)
    # BFS from b1
    visited = set()
    stack = ["b1"]
    while stack:
        n = stack.pop()
        if n in visited:
            continue
        visited.add(n)
        for nb in adj[n]:
            if nb not in visited:
                stack.append(nb)
    return len(visited) < len(adj)

results = []
for cid, flows in contingencies.items():
    out_branch = outage_map[cid]
    islanding = check_islanding(out_branch)
    max_loading = 0.0
    overloads = []
    for bid, flow in flows.items():
        f, t, rating = branches[bid]
        loading = flow / rating * 100
        max_loading = max(max_loading, loading)
        if loading > threshold:
            overloads.append((bid, loading))
    results.append((cid, max_loading, islanding, overloads))

# Sort by: critical overload first, then islanding, then by max loading descending
def sort_key(item):
    cid, max_loading, islanding, overloads = item
    is_critical_overload = 1 if overloads else 0
    is_critical_islanding = 1 if islanding else 0
    # Priority: critical overload (2) > islanding (1) > other (0)
    # Within same priority, higher max loading first
    if overloads:
        cat = 2
    elif islanding:
        cat = 1
    else:
        cat = 0
