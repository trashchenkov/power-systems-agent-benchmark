# Design curve with margin:
# [0.90, 0.20], [0.94, 0.15], [1.00, 0.00], [1.07, -0.30], [1.10, -0.30]

def q_from_curve(V, curve):
    # curve is list of (v, q)
    if V <= curve[0][0]:
        return curve[0][1]
    if V >= curve[-1][0]:
        return curve[-1][1]
    for i in range(len(curve)-1):
        v1, q1 = curve[i]
        v2, q2 = curve[i+1]
        if v1 <= V <= v2:
            t = (V - v1) / (v2 - v1)
            return q1 + t * (q2 - q1)

curve = [(0.90, 0.20), (0.94, 0.15), (1.00, 0.00), (1.07, -0.30), (1.10, -0.30)]

timesteps = [
    ('t_low', 0.940, 60.0, 100.0, 0.00070),
    ('t_nom', 1.000, 80.0, 100.0, 0.00055),
    ('t_high', 1.070, 60.0, 100.0, 0.00070),
]

print('Curve check:')
for v, q in curve:
    print(f'  V={v}, q={q}')

print()
for name, Vbase, kW, kVA, sens in timesteps:
    q = q_from_curve(Vbase, curve)
    Q = q * kVA
    maxQ = (kVA**2 - kW**2)**0.5
    Vcorr = Vbase + Q * sens
    print(f'{name}: Vbase={Vbase}, q={q:.4f}, Q={Q:.2f} kVAR, |Q|={abs(Q):.2f}, maxQ={maxQ:.2f}, Vcorr={Vcorr:.4f}, feasible={0.95<=Vcorr<=1.05 and abs(Q)<=maxQ+1e-9}')

print(f'Monotonic non-increasing: {monotonic}')
