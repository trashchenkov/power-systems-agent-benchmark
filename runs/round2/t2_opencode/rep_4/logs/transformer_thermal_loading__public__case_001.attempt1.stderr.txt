top_oil={};hot={};rate={};total=0.0;mx=-1e9

    ss=r["top_oil_rise_rated_k"]*((1+r["loss_ratio_r"]*K**2)/(1+r["loss_ratio_r"]))**r["oil_exponent_x"]
