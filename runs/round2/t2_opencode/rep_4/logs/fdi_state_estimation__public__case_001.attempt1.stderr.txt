import numpy as np
# buses order: b2, b3 state vars
# branch flows
# l12 x=0.1: (t1-t2)/0.1 = -10 t2
# l13 x=0.125: (t1-t3)/0.125 = -8 t3
# l23 x=0.2: (t2-t3)/0.2 = 5 t2 -5 t3
# inj b2 = (t2-t1)/0.1 + (t2-t3)/0.2 = 15 t2 -5 t3
# inj b3 = (t3-t1)/0.125 + (t3-t2)/0.2 = -5 t2 +13 t3
H = np.array([
 [-10, 0],
 [0, -8],
 [5, -5],
 [15, -5],
 [-5, 13],
], dtype=float)
ids = ["flow_l12","flow_l13","flow_l23","inj_b2","inj_b3"]
z = np.array([0.402,0.557,0.151,-0.1,-0.713])
sigma = np.array([0.01,0.01,0.01,0.01,0.01])
W = np.diag(1/sigma**2)

def wls(H,z,W):
    G = H.T@W@H
    return np.linalg.solve(G, H.T@W@z), G

active = list(range(len(ids)))
flagged=[]
while True:
    Ha=H[active]; za=z[active]; Wa=W[np.ix_(active,active)]
    x,G = wls(Ha,za,Wa)
    Hfull=H[active]
    r = z[active] - Hfull@x
    # omega = sigma^2 - h G^-1 h^T
    Ginv = np.linalg.inv(G)
    omega = np.diag(Wa)[::-1] # no
    sig_a = sigma[active]
    om = sig_a**2 - np.einsum('ij,jk,ik->i', Hfull, Ginv, Hfull)
    rn = np.abs(r)/np.sqrt(om)
    idx = int(np.argmax(rn))
    print("active", [ids[i] for i in active], "x", x, "rn", dict(zip([ids[i] for i in active], rn)))
    if rn[idx] > 3.0:
        flagged.append(ids[active[idx]])
        active.pop(idx)
    else:
        break
print("flagged", flagged)
print("final x theta2,theta3", x)
