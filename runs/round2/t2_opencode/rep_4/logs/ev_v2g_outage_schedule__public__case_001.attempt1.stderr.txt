import itertools, json
dt=0.5
loads={'t1':10.0,'t2':12.0,'t3':9.0}
evs={
 'ev_a':dict(cap=40,init=24,res=16,cmax=4,dmax=8,ce=0.95,de=0.9,step=4.0),
 'ev_b':dict(cap=32,init=20,res=12,cmax=3,dmax=6,ce=0.95,de=0.9,step=3.0),
}
def opts(e):
    s=e['step']; n=int(e['dmax']/s); m=int(e['cmax']/s)
    return [round(i*s,6) for i in range(-m,n+1)]
best=None
A=opts(evs['ev_a']); B=opts(evs['ev_b'])
periods=['t1','t2','t3']
for combo in itertools.product(itertools.product(A,repeat=3),itertools.product(B,repeat=3)):
    ca,cb=combo
    soc={'ev_a':evs['ev_a']['init'],'ev_b':evs['ev_b']['init']}
    served=0; feas=True
    for i,t in enumerate(periods):
        ds={'ev_a':ca[i],'ev_b':cb[i]}
        for k,d in ds.items():
            e=evs[k]
            if d>=0: soc[k]-=d*dt/e['de']
            else: soc[k]+=(-d)*dt*e['ce']
            if soc[k]<e['res']-1e-9: feas=False
        if not feas: break
        served+=min(loads[t],sum(max(d,0) for d in ds.values()))*dt
    if not feas: continue
    key=(served,)
    if best is None or served>best[0]:
        best=(served,ca,cb,dict(soc))
print(best)
