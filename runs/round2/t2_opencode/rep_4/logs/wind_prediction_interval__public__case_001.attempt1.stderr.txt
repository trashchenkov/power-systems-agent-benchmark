import json, math

case = json.load(open('<sandbox_a1>/task.json'))['wind_interval_case']
ref = case['reference_air_density_kg_m3']
count = case['turbine']['count']
rated_each = case['turbine']['rated_mw_each']
pc = case['power_curve']
ci, rated, co = pc['cut_in_m_s'], pc['rated_m_s'], pc['cut_out_m_s']
total = count*rated_each

def power(v, rho):
    vc = v*(rho/ref)**(1/3)
    if vc < ci or vc >= co:
        return 0.0
    if vc >= rated:
        return total
    return total*((vc-ci)/(rated-ci))**3

def quantile(xs, q):
    xs = sorted(xs)
    n = len(xs)
    pos = q*(n-1)
    lo = math.floor(pos); hi = math.ceil(pos)
    if lo == hi:
        return xs[lo]
    frac = pos-lo
    return xs[lo] + frac*(xs[hi]-xs[lo])

res = {}
widths = []
for h in case['horizons']:
    ps = [power(s['wind_speed_m_s'], s['air_density_kg_m3']) for s in h['scenarios']]
    p10 = quantile(ps,0.1); p50 = quantile(ps,0.5); p90 = quantile(ps,0.9)
    res[h['timestamp']] = {'p10': round(p10,6), 'p50': round(p50,6), 'p90': round(p90,6)}
    widths.append(p90-p10)
avg = sum(widths)/len(widths)
sub = {'prediction_interval_mw_by_timestamp': res, 'average_interval_width_mw': round(avg,6),
       'rationale': 'Applied cubic power curve with air-density cube-root correction; p10/p50/p90 via linear-interpolation quantiles across scenarios.'}
print(json.dumps(sub, indent=2))
open('<sandbox_a1>/submission.json','w').write(json.dumps(sub, indent=2))
