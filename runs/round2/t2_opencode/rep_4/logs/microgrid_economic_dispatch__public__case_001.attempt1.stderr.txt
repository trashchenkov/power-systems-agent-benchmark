print("battery choices:",choices)

def sim(bs):
    soc=soc0
    total=0.0
    rows={}
    for t,b in zip(periods,bs):
        if b>=0:
            soc_new=soc - b/ed
        else:
            soc_new=soc + (-b)*ec
        if soc_new < socmin-1e-9 or soc_new > socmax+1e-9:
            return None
        diesel = loads[t]-pvs[t]-b
        if diesel < pmin-1e-9 or diesel > pmax+1e-9:
            return None
        soc=soc_new
        total+=diesel*cost
        rows[t]=(diesel,b,soc)
    return total,rows

best=None
for bs in itertools.product(choices,repeat=3):
    r=sim(bs)
    if r and (best is None or r[0]<best[0]-1e-9):
        best=r
