import itertools, json

periods = ["t00","t05","t10","t15"]
dt = 0.0833333333
load = {"t00":38.0,"t05":42.0,"t10":48.0,"t15":44.0}
pv   = {"t00":18.0,"t05":15.0,"t10":11.0,"t15":9.0}
net = {t: load[t]-pv[t] for t in periods}
pmin,pmax=12.0,55.0
cost=92.0
soc0=5.0; socmin=2.5; socmax=8.0
chgmax=6.0; dismax=10.0
eta_c=0.94; eta_d=0.9
step=2.0

def sim(bs):
    soc=soc0
    ds={}
    for t,b in zip(periods,bs):
        d = net[t]-b
        if d < pmin-1e-9 or d > pmax+1e-9:
            return None
        if b>0:
            soc = soc - b*dt/eta_d
        else:
            soc = soc + (-b)*dt*eta_c
        if soc < socmin-1e-9 or soc > socmax+1e-9:
            return None
        ds[t]=(d,b,soc)
    return ds

best=None
# candidate battery values constrained by power limits and diesel limits
vals=[]
n=int(round(step*1000))
for i in range(-6,11):
    if i % 2==0:
        vals.append(float(i))
print("vals",vals)
best=None; bestbs=None
for bs in itertools.product(vals,repeat=4):
    r=sim(bs)
    if r is None: continue
    # check power limits
    ok=True
    for t,b in zip(periods,bs):
        if b>dismax+1e-9 or b<-chgmax-1e-9: ok=False
    if not ok: continue
    tot=sum(r[t][0] for t in periods)*dt*cost
    if best is None or tot<best-1e-12:
        best=tot; bestbs=bs; bestr=r
print("best cost",best,"bs",bestbs)
print(json.dumps({t:bestr[t] for t in periods},indent=1))
