import json, math

rho_ref=1.225
count=8; rated_each=2.5
cut_in=3.0; rated=12.0; cut_out=25.0

def power(v,rho):
    vc=v*(rho/rho_ref)**(1/3)
    if vc<cut_in or vc>=cut_out: return 0.0
    if vc>=rated: return count*rated_each
    return count*rated_each*((vc-cut_in)/(rated-cut_in))**3

def quant(x,q):
    x=sorted(x); n=len(x); pos=q*(n-1)
    lo=math.floor(pos); hi=math.ceil(pos)
    if lo==hi: return x[lo]
    return x[lo]+(x[hi]-x[lo])*(pos-lo)

data=json.load(open('<sandbox_a1>/task.json'))
case=data['wind_interval_case']
res={}
widths=[]
for h in case['horizons']:
    ps=[power(s['wind_speed_m_s'],s['air_density_kg_m3']) for s in h['scenarios']]
    p10=quant(ps,0.10); p50=quant(ps,0.50); p90=quant(ps,0.90)
    res[h['timestamp']]={'p10':round(p10,6),'p50':round(p50,6),'p90':round(p90,6)}
    widths.append(p90-p10)
    print(h['timestamp'],ps,p10,p50,p90)
avg=sum(widths)/len(widths)
print('avg',avg)
out={'prediction_interval_mw_by_timestamp':res,'average_interval_width_mw':round(avg,6),'rationale':'Computed per instructions power curve and linear-interpolation quantiles.'}
print(json.dumps(out,indent=2))
json.dump(out,open('<sandbox_a1>/submission.json','w'),indent=2)
