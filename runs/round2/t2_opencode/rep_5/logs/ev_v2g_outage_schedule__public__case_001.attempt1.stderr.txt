import itertools, json
periods=["t1","t2","t3"]
load={"t1":10.0,"t2":12.0,"t3":9.0}
h=0.5
evs={
 "ev_a":dict(cap=40.0,soc0=24.0,res=16.0,cmax=4.0,dmax=8.0,ceff=0.95,deff=0.9,step=4.0),
 "ev_b":dict(cap=32.0,soc0=20.0,res=12.0,cmax=3.0,dmax=6.0,ceff=0.95,deff=0.9,step=3.0),
}
def opts(e):
    o=[]
    k=-e["cmax"]
    while k<=e["dmax"]+1e-9:
        o.append(round(k,6)); k+=e["step"]
    return o
print({k:opts(v) for k,v in evs.items()})
best=None
for combo in itertools.product(itertools.product(opts(evs["ev_a"]),repeat=3),
                               itertools.product(opts(evs["ev_b"]),repeat=3)):
    a,b=combo
    soc={k:evs[k]["soc0"] for k in evs}
    feasible=True
    served=0.0
    for i,t in enumerate(periods):
        da,db=a[i],b[i]
        # battery updates
        for k,d,eff in (("ev_a",da,evs["ev_a"]),("ev_b",db,evs["ev_b"])):
            if d>=0:
                soc[k]-=d*h/evs[k]["deff"]
            else:
                soc[k]+=(-d)*h*evs[k]["ceff"]
        for k in evs:
            if soc[k]<evs[k]["res"]-1e-9 or soc[k]>evs[k]["cap"]+1e-9:
                feasible=False
        if not feasible: break
        net=max(da+db,0.0)
        served+=min(net,load[t])*h
    if feasible:
        if best is None or served>best[0]+1e-12:
            best=(served,a,b,dict(soc))
        elif abs(served-best[0])<1e-12:
            # track if final soc differs
            pass
print("BEST served",best[0])
print("dispatch",best[1],best[2])
print("final soc",best[3])
