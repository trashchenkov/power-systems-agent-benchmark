import numpy as np, itertools

buses=['b1','b2','b3','b4']
idx={b:i for i,b in enumerate(buses)}
P=np.array([100.,-20.,-50.,-30.])
branches={
'l12':('b1','b2',0.1,75.,True),
'l13':('b1','b3',0.1,40.,False),
'l23':('b2','b3',0.1,50.,True),
'l24':('b2','b4',0.1,50.,True),
'l34':('b3','b4',0.1,50.,True),
'l14':('b1','b4',0.3,50.,False),
}

def solve(closed):
    n=4
    B=np.zeros((n,n))
    for bid,br in branches.items():
        if bid not in closed: continue
        f,t,x,r,_=br
        i,j=idx[f],idx[t]
        b=1/x
        B[i,i]+=b; B[j,j]+=b; B[i,j]-=b; B[j,i]-=b
    # slack bus 0
    Bred=B[1:,1:]
    Pred=P[1:]
    th=np.zeros(n)
    th[1:]=np.linalg.solve(Bred,Pred)
    # connectivity via closed branches
    parent=list(range(n))
    def find(a):
        while parent[a]!=a:
            parent[a]=parent[parent[a]]; a=parent[a]
        return a
    for bid in closed:
        f,t,_,_,_=branches[bid]
        a,bb=find(idx[f]),find(idx[t])
        if a!=bb: parent[a]=bb
    roots={find(i) for i in range(n)}
    connected=len(roots)==1
    lf={}
    for bid in closed:
        f,t,x,r,_=branches[bid]
        flow=(th[idx[f]]-th[idx[t]])/x
        lf[bid]=flow
    maxload=max(abs(v)/branches[b][3]*100 for b,v in lf.items())
    return connected,maxload,lf,th

base={'l12','l23','l24','l34'}
print("base",solve(base))

for c in cands:
    closed=set(base)
    if c=='l14': closed.add('l14')
    else: closed.discard(c)
    print(c,cands[c],solve(closed)[:2])

def gauss(A,b):
    n=len(A)
    M=[row[:]+[b[i]] for i,row in enumerate(A)]
    for c in range(n):
        p=max(range(c,n),key=lambda r:abs(M[r][c]))
        M[c],M[p]=M[p],M[c]
        pv=M[c][c]
        for r in range(n):
            if r==c: continue
            f=M[r][c]/pv
            for k in range(c,n+1):
                M[r][k]-=f*M[c][k]
    return [M[i][n]/M[i][i] for i in range(n)]
