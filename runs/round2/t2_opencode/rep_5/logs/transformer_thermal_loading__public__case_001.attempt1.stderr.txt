import math, json
c = json.load(open('<sandbox_a1>/task.json'))['transformer_thermal_case']
r = c['rated']
R = r['loss_ratio_r']; x = r['oil_exponent_x']; y = r['winding_exponent_y']
grad = r['hot_spot_gradient_rated_k']; tau = r['oil_time_constant_h']
tr = r['top_oil_rise_rated_k']
ref = c['ageing']['reference_hot_spot_c']; dbl = c['ageing']['doubling_interval_k']
maxhs = c['limits']['max_hot_spot_c']

top_oil = {}; hot_spot = {}; ageing = {}
rise = c['initial_top_oil_rise_k']
total = 0.0
mx = -1e9
for s in c['load_steps']:
    K = s['load_factor_pu']; dt = s['duration_h']; amb = s['ambient_c']
    ss = tr * ((1 + R*K**2)/(1+R))**x
    rise = ss + (rise - ss)*math.exp(-dt/tau)
    to = amb + rise
    hs = to + grad * K**y
    ar = 2**((hs - ref)/dbl)
    top_oil[s['id']] = to
    hot_spot[s['id']] = hs
    ageing[s['id']] = ar
    total += ar*dt
    mx = max(mx, hs)
out = {
  "top_oil_c": top_oil,
  "hot_spot_c": hot_spot,
  "ageing_rate": ageing,
  "total_ageing_hours": total,
  "max_hot_spot_c": mx,
  "overload_admissible": mx <= maxhs,
  "rationale": "IEC 60076-7 exponential top-oil/hot-spot model walked step-by-step from initial top-oil rise 41.3 K."
}
print(json.dumps(out, indent=2))
