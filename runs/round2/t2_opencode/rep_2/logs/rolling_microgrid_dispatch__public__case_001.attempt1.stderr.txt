import itertools, json
dt=0.0833333333
load={'t00':38.0,'t05':42.0,'t10':48.0,'t15':44.0}
pv={'t00':18.0,'t05':15.0,'t10':11.0,'t15':9.0}
periods=['t00','t05','t10','t15']
pmin,pmax=12.0,55.0
soc0,socmin,socmax=5.0,2.5,8.0
chpmax,dispmax=6.0,10.0
ech,edis=0.94,0.9
cost_mwh=92.0
step=2.0

def run(strict_step=True):
    vals=[i for i in range(-6,11)] # MW units for b
    if strict_step:
        bvals=[-6,-4,-2,0,2,4,6,8,10]
    else:
        bvals=[i*0.5 for i in range(-12,21)]
    best=None
    for bs in itertools.product(bvals,repeat=4):
        soc=soc0
        ok=True
        dies=[]
        for i,b in enumerate(bs):
            if b>=0:
                if b>dispmax: ok=False;break
                soc2=soc-b*dt/edis
            else:
                if -b>chpmax: ok=False;break
                soc2=soc+(-b)*dt*ech
            if soc2<socmin-1e-9 or soc2>socmax+1e-9: ok=False;break
            d=load[periods[i]]-pv[periods[i]]-b
            if d<pmin-1e-9 or d>pmax+1e-9: ok=False;break
            dies.append(d)
            soc=soc2
        if ok:
            c=sum(dies)*dt*cost_mwh
            if best is None or c<best[0]-1e-9:
                best=(c,bs,list(dies),soc)
    return best

print("strict step",run(True))
print("fine",run(False))
