import math, json
c=json.load(open('task.json'))
d=c['cable_sizing_case']
I=d['design_current_a']; pf=d['power_factor']; V=d['system_voltage_v']; L=d['length_m']

thermal=If*math.sqrt(t)/k
print("derate",kder,"thermal",thermal,"sin",sin)
sel=None
for e in d['catalog']:
    da=e['base_ampacity_a']*kder
    dU=math.sqrt(3)*I*(L/1000)*(e['r_ohm_per_km']*pf+e['x_ohm_per_km']*sin)
    dUp=dU/V*100
    ok = da>=I and dUp<=dmax and e['cross_section_mm2']>=thermal
    print(e['cross_section_mm2'], round(da,4), round(dUp,4), ok)
    if ok and sel is None:
        sel=(e['cross_section_mm2'],da,dUp)
print("SEL",sel)

print(json.dumps(out,indent=2))
open('submission.json','w').write(json.dumps(out,indent=2))
