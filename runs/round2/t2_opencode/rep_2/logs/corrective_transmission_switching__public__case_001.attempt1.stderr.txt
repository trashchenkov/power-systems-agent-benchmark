import numpy as np
buses={'b1':100.0,'b2':-20.0,'b3':-50.0,'b4':-30.0}
br={'l12':('b1','b2',0.1,75.0,True),'l13':('b1','b3',0.1,40.0,False),
'l23':('b2','b3',0.1,50.0,True),'l24':('b2','b4',0.1,50.0,True),
'l34':('b3','b4',0.1,50.0,True),'l14':('b1','b4',0.3,50.0,False)}
idx={'b1':0,'b2':1,'b3':2,'b4':3}
def solve(closed):
    ns=[b for b in idx if b!='b1']
    n=len(ns); B=np.zeros((n,n)); P=np.array([buses[b] for b in ns])
    for bid in closed:
        f,t,x,r,c=br[bid]; b=1/x
        if f!='b1':
            i=ns.index(f); B[i,i]+=b
            if t!='b1': j=ns.index(t); B[i,j]-=b; B[j,i]-=b; B[j,j]+=b
        if t!='b1':
            j=ns.index(t); B[j,j]+=b
            if f!='b1':
                i=ns.index(f); B[i,j]-=b; B[j,i]-=b; B[i,i]+=b
    th=np.linalg.solve(B,P)
    theta={'b1':0.0}
    for i,b in enumerate(ns): theta[b]=th[i]
    flows={}
    for bid in closed:
        f,t,x,r,c=br[bid]; flows[bid]=(theta[f]-theta[t])/x
    # connectivity
    adj={b:set() for b in idx}
    for bid in closed:
        f,t,x,r,c=br[bid]; adj[f].add(t); adj[t].add(f)
    seen=set(); st=['b1']
    while st:
        u=st.pop()
        if u in seen: continue
        seen.add(u); st+=list(adj[u]-seen)
    conn=len(seen)==4
    mx=max(abs(v)/br[k][3]*100 for k,v in flows.items())
    return flows,conn,mx
