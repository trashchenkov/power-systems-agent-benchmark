    if v < db:
        iq=min(k*(db-v), lim)

    rem=math.sqrt(max(lim*lim-iq*iq,0.0))
    idc=min(id0, rem)
    tot=math.hypot(idc,iq)
    prof[sid]={"id_pu":idc,"iq_pu":iq,"total_current_pu":tot}
    print(sid,v,"iq=%.6f id=%.6f tot=%.6f"%(iq,idc,tot))
print("min_iq", min(p["iq_pu"] for p in prof.values()))
print("max_tot", max(p["total_current_pu"] for p in prof.values()))
