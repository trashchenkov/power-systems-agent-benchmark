flagged = []
while True:
    idx = [i for i in range(5) if mask[i]]
    H = H_full[idx]; z = z_full[idx]; sig = sig_full[idx]
    W = np.diag(1/sig**2)
    G = H.T @ W @ H
    x = np.linalg.solve(G, H.T @ W @ z)
    r = z - H @ x
    Ginv = np.linalg.inv(G)
    norm = np.abs(r)/np.sqrt(sig**2 - np.einsum('ij,jk,ik->i', H, Ginv, H))
    print("x=",x)
    print("norm residuals:", {ids[idx[i]]: norm[i] for i in range(len(idx))})
    j = int(np.argmax(norm))
    if norm[j] > 3.0:
        flagged.append(ids[idx[j]])

        print("flag", ids[idx[j]])
    else:
        break

print("final x:", x)
