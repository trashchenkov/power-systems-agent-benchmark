import json, math
task=json.load(open('<sandbox_a1>/task.json'))
c=task['wind_interval_case']
ref=c['reference_air_density_kg_m3']
count=c['turbine']['count']
rated=c['turbine']['rated_mw_each']
pc=c['power_curve']
ci,rr,co=pc['cut_in_m_s'],pc['rated_m_s'],pc['cut_out_m_s']
total=count*rated
def power(v,rho):
    vc=v*(rho/ref)**(1/3)
    if vc<ci or vc>=co: return 0.0
    if vc>=rr: return total
    return total*((vc-ci)/(rr-ci))**3
def quant(x,q):
    x=sorted(x); n=len(x); pos=q*(n-1)
    lo=math.floor(pos); hi=math.ceil(pos)
    if lo==hi: return x[lo]
    return x[lo]+(x[hi]-x[lo])*(pos-lo)
out={}
widths=[]
for h in c['horizons']:
    ps=[power(s['wind_speed_m_s'],s['air_density_kg_m3']) for s in h['scenarios']]
    p10=quant(ps,0.10); p50=quant(ps,0.50); p90=quant(ps,0.90)
    out[h['timestamp']]={'p10':round(p10,4),'p50':round(p50,4),'p90':round(p90,4)}
    widths.append(p90-p10)
res={'prediction_interval_mw_by_timestamp':out,'average_interval_width_mw':round(sum(widths)/len(widths),4),'rationale':'Applied cubic power curve with density-corrected wind speed; empirical linear-interpolated quantiles per horizon.'}
json.dump(res,open('<sandbox_a1>/submission.json','w'),indent=2)
print(json.dumps(res,indent=2))
for h in c['horizons']:
    ps=[power(s['wind_speed_m_s'],s['air_density_kg_m3']) for s in h['scenarios']]
    print(h['timestamp'], [round(p,4) for p in ps])
