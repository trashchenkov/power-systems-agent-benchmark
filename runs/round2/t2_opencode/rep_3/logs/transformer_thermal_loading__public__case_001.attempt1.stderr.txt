import math
c={
 "initial_top_oil_rise_k":41.3,
 "rated":{"top_oil_rise_rated_k":55.0,"loss_ratio_r":5.0,"oil_exponent_x":0.8,"winding_exponent_y":1.3,"hot_spot_gradient_rated_k":23.0,"oil_time_constant_h":3.0},

 "max":120.0,
 "steps":[("s1",2.0,25.0,0.8),("s2",1.0,28.0,1.15),("s3",2.0,30.0,1.0),("s4",1.0,27.0,0.7)]
}
r=c["rated"]; a=c["ageing"]
rise=c["initial_top_oil_rise_k"]
top={};hot={};rate={};tot=0.0;mx=-1e9
for sid,dur,amb,K in c["steps"]:
    ss=r["top_oil_rise_rated_k"]*((1+r["loss_ratio_r"]*K**2)/(1+r["loss_ratio_r"]))**r["oil_exponent_x"]
    rise=ss+(rise-ss)*math.exp(-dur/r["oil_time_constant_h"])

    hs=to+r["hot_spot_gradient_rated_k"]*K**r["winding_exponent_y"]

    top[sid]=to;hot[sid]=hs;rate[sid]=ar
    tot+=ar*dur
    mx=max(mx,hs)
    print(sid,"ss=%.6f rise=%.6f top=%.6f hot=%.6f rate=%.6f"%(ss,rise,to,hs,ar))
print("total=%.6f max=%.6f admissible=%s"%(tot,mx,mx<=c["max"]))
