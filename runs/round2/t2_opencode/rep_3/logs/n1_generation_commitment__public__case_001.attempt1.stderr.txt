from itertools import combinations
units={'g1':(120,400,80),'g2':(160,550,100),'g3':(180,700,120),'g4':(90,220,60)}
D=260; R=40; req=D+R
def solve(n1_extra_reserve):
    best=None
    ids=list(units)
    for r in range(1,5):
        for c in combinations(ids,r):
            cap=sum(units[u][0] for u in c)
            cost=sum(units[u][1]+units[u][2] for u in c)
            if cap<req: continue
            mx=max(units[u][0] for u in c)
            n1=cap-mx
            if n1_extra_reserve:
                if n1<req: continue
            else:
                if n1<D: continue
            key=(cost,cap)
            if best is None or key<best[0]:
                best=(key,c,cap,n1,cost)
    return best
print("N-1>=D      :",solve(False))
print("N-1>=D+R    :",solve(True))
