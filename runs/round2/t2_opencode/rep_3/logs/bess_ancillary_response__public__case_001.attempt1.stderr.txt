import json, math

c = json.load(open('<sandbox_a1>/task.json'))['bess_response_case']
f0=c['nominal_frequency_hz']; v0=c['target_voltage_pu']
kf=c['frequency_droop_mw_per_hz']; kv=c['voltage_droop_mvar_per_pu']
S=c['s_rating_mva']
pd=c['p_discharge_limit_mw']; pc=c['p_charge_limit_mw']
dt=c['interval_seconds']/3600.0
eta_c=c['charge_efficiency']; eta_d=c['discharge_efficiency']
soc=c['initial_soc_mwh']

P={}; Q={}
for t,m in c['measurements'].items():
    p=(f0-m['frequency_hz'])*kf
    p=max(-pc,min(pd,p))
    qlim=math.sqrt(S*S-p*p)
    q=(v0-m['voltage_pu'])*kv
    q=max(-qlim,min(qlim,q))
    P[t]=p; Q[t]=q
    if p>0:
        soc-=p*dt/eta_d
    elif p<0:
        soc+=(-p)*dt*eta_c

print(json.dumps({'p_response_mw':P,'q_response_mvar':Q,'final_soc_mwh':soc},indent=2))
