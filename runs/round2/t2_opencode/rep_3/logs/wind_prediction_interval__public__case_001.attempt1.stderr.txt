import json, math

data = json.load(open('<sandbox_a1>/task.json'))
case = data['wind_interval_case']
ref = case['reference_air_density_kg_m3']
count = case['turbine']['count']
rated_each = case['turbine']['rated_mw_each']
pc = case['power_curve']
ci, rm, co = pc['cut_in_m_s'], pc['rated_m_s'], pc['cut_out_m_s']
P_rated = count*rated_each

def power(v, rho):
    vc = v*(rho/ref)**(1/3)
    if vc < ci or vc >= co:
        return 0.0
    if vc >= rm:
        return P_rated
    return P_rated*((vc-ci)/(rm-ci))**3

def quantile(xs, q):
    xs = sorted(xs)
    n = len(xs)
    pos = q*(n-1)
    lo = math.floor(pos); hi = math.ceil(pos)
    if lo == hi:
        return xs[int(lo)]
    return xs[int(lo)] + (xs[int(hi)]-xs[int(lo)])*(pos-lo)

out = {}
widths = []
for h in case['horizons']:
    ps = [power(s['wind_speed_m_s'], s['air_density_kg_m3']) for s in h['scenarios']]
    p10 = quantile(ps,0.10); p50 = quantile(ps,0.50); p90 = quantile(ps,0.90)
    out[h['timestamp']] = {'p10':round(p10,6),'p50':round(p50,6),'p90':round(p90,6)}
    widths.append(p90-p10)
    print(h['timestamp'], [round(p,5) for p in sorted(ps)], p10,p50,p90)

avg = sum(widths)/len(widths)
print('avg', avg)
res = {'prediction_interval_mw_by_timestamp':out,'average_interval_width_mw':round(avg,6)}
json.dump(res, open('/tmp/res.json','w'), indent=2)
print(json.dumps(res,indent=2))
