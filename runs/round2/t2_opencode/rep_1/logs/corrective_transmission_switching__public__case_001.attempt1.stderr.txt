import numpy as np, itertools

buses = {"b1":100.0,"b2":-20.0,"b3":-50.0,"b4":-30.0}
bl = {
 "l12":("b1","b2",0.1,75.0),"l13":("b1","b3",0.1,40.0),
 "l23":("b2","b3",0.1,50.0),"l24":("b2","b4",0.1,50.0),
 "l34":("b3","b4",0.1,50.0),"l14":("b1","b4",0.3,50.0),
}
idx={"b1":0,"b2":1,"b3":2,"b4":3}
P=np.array([buses[b] for b in ["b1","b2","b3","b4"]])

def solve(closed):
    n=4
    B=np.zeros((n,n))
    for e in closed:
        f,t,x,r=bl[e]
        i,j=idx[f],idx[t]
        b=1/x
        B[i,i]+=b;B[j,j]+=b;B[i,j]-=b;B[j,i]-=b
    # slack 0
    Bred=B[1:,1:]; Pred=P[1:]
    th=np.zeros(n)
    th[1:]=np.linalg.solve(Bred,Pred)
    flows={}
    for e in closed:
        f,t,x,r=bl[e]
        flows[e]=(th[idx[f]]-th[idx[t]])/x
    return th,flows

def connected(closed):
    adj={b:set() for b in buses}
    for e in closed:
        f,t,_,_=bl[e]
        adj[f].add(t);adj[t].add(f)
    seen=set();st=["b1"]
    while st:
        u=st.pop()
        if u in seen:continue
        seen.add(u);st+=list(adj[u]-seen)
    return len(seen)==4

base=[e for e in bl if e!="l13" and e!="l14"]
print("base:", {k:round(v,3) for k,v in solve(base)[1].items()}, "conn",connected(base))

cands=[("close","l14"),("open","l23"),("open","l24"),("open","l34")]
for act,br in cands:
    c=set(base)
    if act=="close":c.add(br)
    else:c.discard(br)
    th,fl=solve(list(c))
    load={k:round(abs(v)/bl[k][3]*100,3) for k,v in fl.items()}
    print(f"{act} {br}: conn={connected(list(c))} maxload={max(load.values()):.3f} loads={load}")
