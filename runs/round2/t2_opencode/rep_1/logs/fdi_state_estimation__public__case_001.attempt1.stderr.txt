import numpy as np

# branch: (from,to,x)
branches = {
 'l12': ('b1','b2',0.1),
 'l13': ('b1','b3',0.125),
 'l23': ('b2','b3',0.2),
}
# state vars theta_2,theta_3 ; theta_1=0
def h_flow(bid):
    f,t,x = branches[bid]
    h = np.zeros(2)
    # P_ft = (theta_f-theta_t)/x
    for node,sign in [(f,-1),(t,+1)]:
        if node=='b2': h[0]+= sign/x
        if node=='b3': h[1]+= sign/x
    return h

def h_inj(bus):
    # injection = sum of flows out of bus = sum over branches where bus is 'from' (theta_bus-theta_other)/x
    h=np.zeros(2)
    for bid,(f,t,x) in branches.items():
        if f==bus:
            for node,sign in [(f,1),(t,-1)]:
                if node=='b2': h[0]+=sign/x
                if node=='b3': h[1]+=sign/x
        if t==bus:
            for node,sign in [(t,1),(f,-1)]:
                if node=='b2': h[0]+=sign/x
                if node=='b3': h[1]+=sign/x
    return h

meas = [
 ('flow_l12','flow','l12',0.402,0.01),
 ('flow_l13','flow','l13',0.557,0.01),
 ('flow_l23','flow','l23',0.151,0.01),
 ('inj_b2','inj','b2',-0.1,0.01),
 ('inj_b3','inj','b3',-0.713,0.01),
]

def run(meas, H_sign=1):
    active=list(meas)
    flagged=[]
    while True:
        n=len(active)
        H=np.zeros((n,2)); z=np.zeros(n); R=np.zeros(n)
        for i,(mid,typ,ref,val,sig) in enumerate(active):
            if typ=='flow': h=h_flow(ref)
            else: h=H_sign*h_inj(ref)
            H[i]=h; z[i]=val; R[i]=sig**2
        W=np.diag(1/R)
        G=H.T@W@H
        Ginv=np.linalg.inv(G)
        x=Ginv@H.T@W@z
        r=z-H@x
        denom=np.sqrt(R - np.einsum('ij,jk,ik->i',H,Ginv,H))
        rn=r/denom
        idx=np.argmax(np.abs(rn))
        print('H_sign',H_sign,'active',[m[0] for m in active],'x',x,'rn',dict(zip([m[0] for m in active],np.round(rn,3))))
        if abs(rn[idx])>3.0:
            flagged.append(active[idx][0])
            active.pop(idx)
        else:
            break
    return flagged,x

for s in [1,-1]:
    print('==== sign',s)
    f,x=run(meas,s)
    print('flagged',f,'state',x)

import numpy as np
# STANDARD convention: flow_ij=(theta_i-theta_j)/x ; inj_i = sum_j flow_ij
branches={'l12':('b1','b2',0.1),'l13':('b1','b3',0.125),'l23':('b2','b3',0.2)}
def hv(node,sign,h):
    if node=='b2': h[0]+=sign
    if node=='b3': h[1]+=sign
def h_flow(bid):
    f,t,x=branches[bid]; h=np.zeros(2)
    hv(f,1/x,h); hv(t,-1/x,h); return h
def h_inj(bus):
    h=np.zeros(2)
    for f,t,x in branches.values():
        if f==bus: hv(f,1/x,h); hv(t,-1/x,h)
        if t==bus: hv(t,1/x,h); hv(f,-1/x,h)
    return h
