  "ageing": {"reference_hot_spot_c":98.0,"doubling_interval_k":6.0},
  "load_steps":[{"id":"s1","duration_h":2.0,"ambient_c":25.0,"load_factor_pu":0.8},
                {"id":"s2","duration_h":1.0,"ambient_c":28.0,"load_factor_pu":1.15},
                {"id":"s3","duration_h":2.0,"ambient_c":30.0,"load_factor_pu":1.0},

}
r=c["rated"]; a=c["ageing"]
R=r["loss_ratio_r"]; x=r["oil_exponent_x"]; y=r["winding_exponent_y"]
tau=r["oil_time_constant_h"]; g=r["hot_spot_gradient_rated_k"]
prev=c["initial_top_oil_rise_k"]
top={}; hs={}; rate={}; total=0.0; mx=-1e9
for s in c["load_steps"]:
    K=s["load_factor_pu"]; t=s["duration_h"]; amb=s["ambient_c"]
    steady=r["top_oil_rise_rated_k"]*((1+R*K**2)/(1+R))**x
    end=steady+(prev-steady)*math.exp(-t/tau)
    topo=amb+end
    hotspot=topo+g*(K**y)
    ar=2**((hotspot-a["reference_hot_spot_c"])/a["doubling_interval_k"])
    top[s["id"]]=topo; hs[s["id"]]=hotspot; rate[s["id"]]=ar
    total+=ar*t
    mx=max(mx,hotspot)
    prev=end
    print(s["id"],"steady",steady,"end",end,"top",topo,"hs",hotspot,"rate",ar)
print("total_ageing_hours",total,"max",mx,"admissible",mx<=c["limits"]["max_hot_spot_c"])
