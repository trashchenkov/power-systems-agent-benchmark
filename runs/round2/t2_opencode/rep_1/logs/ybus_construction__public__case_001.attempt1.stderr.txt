import json
r=json.load(open('task.json'))
net=r['network']
buses=net['buses']; n=len(buses); idx={b:i for i,b in enumerate(buses)}
Y=[[0j]*n for _ in range(n)]
for br in net['branches']:
    i=idx[br['from_bus']]; j=idx[br['to_bus']]
    y=1/(complex(br['r_pu'],br['x_pu']))
    diag=y+1j*br['b_pu']/2
    Y[i][i]+=diag; Y[j][j]+=diag
    Y[i][j]-=y; Y[j][i]-=y
for sh in net['shunts']:
    i=idx[sh['bus']]; Y[i][i]+=complex(sh['g_pu'],sh['b_pu'])
out={'buses':buses,'ybus':[[{'re':Y[i][j].real,'im':Y[i][j].imag} for j in range(n)] for i in range(n)],'rationale':'Ybus from branch series admittances, line charging b/2, and bus shunt.'}
json.dump(out,open('submission.json','w'),indent=2)
