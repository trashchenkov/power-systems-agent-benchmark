import itertools

# Fault sections data
sections = {
    "s12": {"annual_fault_rate": 0.18, "customers_interrupted": 120},
    "s23": {"annual_fault_rate": 0.14, "customers_interrupted": 70},
    "s24": {"annual_fault_rate": 0.12, "customers_interrupted": 90},
    "s45": {"annual_fault_rate": 0.09, "customers_interrupted": 40}
}

# Candidate locations data
candidates = {
    "n1": {"s12": 18, "s23": 16, "s24": 15, "s45": 14},
    "n2": {"s12": 6, "s23": 28, "s24": 26, "s45": 23},
    "n3": {"s12": 0, "s23": 24, "s24": 2, "s45": 2},
    "n4": {"s12": 0, "s23": 4, "s24": 18, "s45": 34}
}

def calculate_objective(selected):
    """Calculate the objective value for a set of selected locations."""
    total = 0
    section_reductions = {}
    
    for section_id, section_data in sections.items():
        # Find the maximum gain among selected locations for this section
        max_gain = 0
        for loc in selected:
            gain = candidates[loc][section_id]
            if gain > max_gain:
                max_gain = gain
        
        section_reductions[section_id] = max_gain
        
        # Add to objective: rate * customers * gain
        contribution = section_data["annual_fault_rate"] * section_data["customers_interrupted"] * max_gain
        total += contribution
    
    return total, section_reductions

# Enumerate all combinations of 2 locations
best_value = -1
best_selection = None
best_reductions = None

for combo in itertools.combinations(candidates.keys(), 2):
    value, reductions = calculate_objective(list(combo))
    print(f"{combo}: objective = {value:.4f}")
    if value > best_value:
        best_value = value
        best_selection = combo
        best_reductions = reductions

print(f"\nBest selection: {best_selection}")
print(f"Best objective: {best_value:.4f}")
print(f"Section reductions: {best_reductions}")

# Calculate expected customer-minutes saved
print(f"\nDetailed calculation for {best_selection}:")
total_saved = 0
for section_id, section_data in sections.items():
    gain = best_reductions[section_id]
    contribution = section_data["annual_fault_rate"] * section_data["customers_interrupted"] * gain
    total_saved += contribution
    print(f"  {section_id}: rate={section_data['annual_fault_rate']}, customers={section_data['customers_interrupted']}, gain={gain} min, contribution={contribution:.4f}")

print(f"\nTotal expected customer-minutes saved: {total_saved:.4f}")
