import math

# Input data
target_voltage_pu = 0.98
base_voltage_pu = {"bus_2": 0.944, "bus_3": 0.952}
q_sensitivity_pu_per_kvar = {"bus_2": 0.0016, "bus_3": 0.0012}

evs = {
    "evse_1": {"bus": "bus_2", "s_rating_kva": 18.0, "active_power_kw": 10.0},
    "evse_2": {"bus": "bus_2", "s_rating_kva": 12.0, "active_power_kw": 6.0},
    "evse_3": {"bus": "bus_3", "s_rating_kva": 16.0, "active_power_kw": 8.0}
}

# Calculate max Q capability for each EV
def calc_q_max(s_rating, p_active):
    return math.sqrt(s_rating**2 - p_active**2)

# Aggregate Q capability by bus
bus_q_max = {"bus_2": 0.0, "bus_3": 0.0}
for ev_id, ev_data in evs.items():
    bus = ev_data["bus"]
    q_max = calc_q_max(ev_data["s_rating_kva"], ev_data["active_power_kw"])
    print(f"{ev_id} at {bus}: Q_max = sqrt({ev_data['s_rating_kva']}^2 - {ev_data['active_power_kw']}^2) = {q_max:.6f} kvar")
    bus_q_max[bus] += q_max

print(f"\nAggregate Q_max: {bus_q_max}")

# Calculate required Q for each bus to reach target
q_dispatch_kvar = {}
corrected_voltage_pu = {}

for bus in ["bus_2", "bus_3"]:
    v_deviation = target_voltage_pu - base_voltage_pu[bus]
    q_required = v_deviation / q_sensitivity_pu_per_kvar[bus]
    q_max = bus_q_max[bus]
    
    # Clip by aggregate capability
    q_dispatch = min(q_required, q_max)
    
    # Calculate corrected voltage
    v_corrected = base_voltage_pu[bus] + q_dispatch * q_sensitivity_pu_per_kvar[bus]
    
    q_dispatch_kvar[bus] = round(q_dispatch, 6)
    corrected_voltage_pu[bus] = round(v_corrected, 6)
    
    print(f"\n{bus}:")
    print(f"  Voltage deviation: {v_deviation:.6f} pu")
    print(f"  Q required: {q_required:.6f} kvar")
    print(f"  Q max available: {q_max:.6f} kvar")
    print(f"  Q dispatched: {q_dispatch:.6f} kvar")
    print(f"  Corrected voltage: {v_corrected:.6f} pu")

total_abs_q_kvar = sum(abs(q) for q in q_dispatch_kvar.values())

print(f"\nTotal absolute Q: {total_abs_q_kvar:.6f} kvar")

# Prepare submission
submission = {
    "q_dispatch_kvar": q_dispatch_kvar,
    "corrected_voltage_pu": corrected_voltage_pu,
    "total_abs_q_kvar": round(total_abs_q_kvar, 6)
}

import json
print("\n" + "="*50)
print("SUBMISSION:")
print(json.dumps(submission, indent=2))
