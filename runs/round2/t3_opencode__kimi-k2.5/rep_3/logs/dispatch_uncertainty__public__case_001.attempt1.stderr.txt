import json

# Data from task.json
time_periods = ["t1", "t2", "t3"]
reserve_requirement_mw = 8.0

generators = {
    "g1": {"pmin_mw": 10.0, "pmax_mw": 70.0, "variable_cost_per_mwh": 18.0},
    "g2": {"pmin_mw": 5.0, "pmax_mw": 45.0, "variable_cost_per_mwh": 36.0}
}

scenarios = [
    {"id": "normal", "load_mw": {"t1": 78.0, "t2": 85.0, "t3": 75.0}, "renewable_mw": {"t1": 18.0, "t2": 24.0, "t3": 20.0}},
    {"id": "high_load", "load_mw": {"t1": 84.0, "t2": 92.0, "t3": 82.0}, "renewable_mw": {"t1": 12.0, "t2": 16.0, "t3": 13.0}},
    {"id": "low_renewable", "load_mw": {"t1": 80.0, "t2": 87.0, "t3": 77.0}, "renewable_mw": {"t1": 8.0, "t2": 10.0, "t3": 9.0}}
]

# Step 1: Calculate net load for each scenario and time period
print("Net load calculations (load_mw - renewable_mw):")
for scenario in scenarios:
    sid = scenario["id"]
    print(f"\nScenario: {sid}")
    for t in time_periods:
        net_load = scenario["load_mw"][t] - scenario["renewable_mw"][t]
        print(f"  {t}: {scenario['load_mw'][t]} - {scenario['renewable_mw'][t]} = {net_load} MW")

# Step 2: Find maximum net load across scenarios for each time period
print("\n\nMaximum net load across scenarios:")
max_net_load = {}
for t in time_periods:
    max_nl = max(s["load_mw"][t] - s["renewable_mw"][t] for s in scenarios)
    max_net_load[t] = max_nl
    print(f"  {t}: {max_nl} MW")

# Step 3: Calculate required thermal generation (max net load + reserve)
print("\n\nRequired thermal generation (max net load + reserve):")
required_gen = {}
for t in time_periods:
    req = max_net_load[t] + reserve_requirement_mw
    required_gen[t] = req
    print(f"  {t}: {max_net_load[t]} + {reserve_requirement_mw} = {req} MW")

# Step 4: Optimize dispatch
# g1 has lower cost (18 vs 36), so prioritize g1
# Check if g1 alone can cover the requirement
dispatch = {"g1": {}, "g2": {}}
total_cost = 0.0

for t in time_periods:
    req = required_gen[t]
    
    # Try to use g1 as much as possible (cheaper)
    g1_max = generators["g1"]["pmax_mw"]
    g1_min = generators["g1"]["pmin_mw"]
    g2_max = generators["g2"]["pmax_mw"]
    g2_min = generators["g2"]["pmin_mw"]
    
    # Use g1 first (cheaper)
    g1_dispatch = min(req, g1_max)
    remaining = req - g1_dispatch
    
    # If remaining is less than g2_min, we need to adjust
    if remaining > 0 and remaining < g2_min:
        # Need to increase g1 to cover more
        g1_dispatch = req - g2_min
        if g1_dispatch < g1_min:
            # Need to use minimums for both
            g1_dispatch = g1_min
            g2_dispatch = g2_min
        else:
            g2_dispatch = g2_min
    elif remaining > 0:
        g2_dispatch = remaining
    else:
        g2_dispatch = 0.0
    
    # Ensure g1 is at least pmin if dispatching
    if g1_dispatch > 0 and g1_dispatch < g1_min:
        g1_dispatch = g1_min
    if g2_dispatch > 0 and g2_dispatch < g2_min:
        g2_dispatch = g2_min
    
    dispatch["g1"][t] = round(g1_dispatch, 6)
    dispatch["g2"][t] = round(g2_dispatch, 6)
    
    cost = g1_dispatch * generators["g1"]["variable_cost_per_mwh"] + \
           g2_dispatch * generators["g2"]["variable_cost_per_mwh"]
    total_cost += cost
    
    print(f"\n{t}: Required {req} MW")
    print(f"  g1: {g1_dispatch} MW (cost: {g1_dispatch * generators['g1']['variable_cost_per_mwh']})")
    print(f"  g2: {g2_dispatch} MW (cost: {g2_dispatch * generators['g2']['variable_cost_per_mwh']})")
    print(f"  Total: {g1_dispatch + g2_dispatch} MW")

print(f"\n\nTotal cost: {total_cost}")

# Calculate worst case margins
worst_case_margin = {}
for t in time_periods:
    total_gen = dispatch["g1"][t] + dispatch["g2"][t]
    margin = total_gen - max_net_load[t]
    worst_case_margin[t] = round(margin, 6)
    print(f"{t}: Total gen {total_gen} - Max net load {max_net_load[t]} = Margin {margin} MW")

print("\n\nFinal submission:")
result = {
    "dispatch_mw": dispatch,
    "worst_case_margin_mw": worst_case_margin,
    "total_cost": round(total_cost, 6)
}
print(json.dumps(result, indent=2))
