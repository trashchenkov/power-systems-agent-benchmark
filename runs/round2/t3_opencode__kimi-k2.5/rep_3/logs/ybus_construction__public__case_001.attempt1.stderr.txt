import json
import numpy as np

# Network data
buses = ["bus_1", "bus_2", "bus_3"]
branches = [
    {"id": "l12", "from_bus": "bus_1", "to_bus": "bus_2", "r_pu": 0.02, "x_pu": 0.06, "b_pu": 0.03},
    {"id": "l23", "from_bus": "bus_2", "to_bus": "bus_3", "r_pu": 0.025, "x_pu": 0.075, "b_pu": 0.02},
    {"id": "l13", "from_bus": "bus_1", "to_bus": "bus_3", "r_pu": 0.01, "x_pu": 0.04, "b_pu": 0.01}
]
shunts = [
    {"id": "sh3", "bus": "bus_3", "g_pu": 0.0, "b_pu": 0.04}
]

n = len(buses)
bus_idx = {bus: i for i, bus in enumerate(buses)}

# Initialize Ybus matrix (complex)
Ybus = np.zeros((n, n), dtype=complex)

# Process each branch
for branch in branches:
    i = bus_idx[branch["from_bus"]]
    j = bus_idx[branch["to_bus"]]
    r = branch["r_pu"]
    x = branch["x_pu"]
    b = branch["b_pu"]
    
    # Series admittance y = 1 / (r + jx)
    z = complex(r, x)
    y = 1 / z
    
    # Line charging: j*b/2 added to both diagonal entries
    y_charging = complex(0, b / 2)
    
    # Add to diagonal entries (self admittance)
    Ybus[i, i] += y + y_charging
    Ybus[j, j] += y + y_charging
    
    # Add to off-diagonal entries (mutual admittance)
    Ybus[i, j] -= y
    Ybus[j, i] -= y

# Process shunts (add to diagonal)
for shunt in shunts:
    i = bus_idx[shunt["bus"]]
    g = shunt["g_pu"]
    b = shunt["b_pu"]
    Ybus[i, i] += complex(g, b)

# Format output
ybus_matrix = []
for i in range(n):
    row = []
    for j in range(n):
        row.append({"re": round(Ybus[i, j].real, 10), "im": round(Ybus[i, j].imag, 10)})
    ybus_matrix.append(row)

submission = {
    "buses": buses,
    "ybus": ybus_matrix
}

print(json.dumps(submission, indent=2))
