from itertools import combinations

# Data from task.json
budget = 2
candidate_meters = ["m12", "m23", "m31", "pinj2"]
attack_vectors = [
    {"attack_id": "a_theta2", "risk_weight": 7, "detectable_by": ["m12", "pinj2"]},
    {"attack_id": "a_theta3", "risk_weight": 6, "detectable_by": ["m31"]},
    {"attack_id": "a_load_shift", "risk_weight": 4, "detectable_by": ["m23", "pinj2"]},
    {"attack_id": "a_stealth_pair", "risk_weight": 9, "detectable_by": ["m12", "m31"]},
]

# Check all combinations of 2 meters
best_combo = None
best_score = -1
best_covered = []

for combo in combinations(candidate_meters, budget):
    covered_attacks = set()
    total_risk = 0
    
    for attack in attack_vectors:
        # Check if any meter in combo can detect this attack
        if any(meter in attack["detectable_by"] for meter in combo):
            covered_attacks.add(attack["attack_id"])
            total_risk += attack["risk_weight"]
    
    print(f"Combo {combo}: covered={covered_attacks}, total_risk={total_risk}")
    
    # Maximize total risk; ties use lexicographic meter IDs
    if total_risk > best_score:
        best_score = total_risk
        best_combo = combo
        best_covered = covered_attacks
    elif total_risk == best_score:
        # Lexicographic tie-breaker
        if combo < best_combo:
            best_combo = combo
            best_covered = covered_attacks

print(f"\nBest combo: {best_combo}")
print(f"Best score: {best_score}")
print(f"Covered attacks: {sorted(best_covered)}")
