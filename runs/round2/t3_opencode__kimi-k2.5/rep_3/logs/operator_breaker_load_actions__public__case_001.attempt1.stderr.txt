# Initial flows
flows = {
    'l12': 118,
    'l23': 72,
    'l13': 65
}

# Limits
limits = {
    'l12': 100,
    'l23': 90,
    'l13': 80
}

# Actions
actions = {
    'open_tie_13': {
        'type': 'open_breaker',
        'flow_delta': {'l12': -8, 'l23': 12}
    },
    'shed_bus3_15': {
        'type': 'shed_load',
        'mw': 15,
        'flow_delta': {'l12': -14, 'l23': -6}
    },
    'shed_bus2_10': {
        'type': 'shed_load',
        'mw': 10,
        'flow_delta': {'l12': -9, 'l23': -2}
    }
}

# Check all combinations (up to 2 actions)
from itertools import combinations

def check_feasible(final_flows):
    for branch, flow in final_flows.items():
        if flow > limits[branch]:
            return False
    return True

def apply_action(flows, action_id):
    new_flows = flows.copy()
    for branch, delta in actions[action_id]['flow_delta'].items():
        new_flows[branch] += delta
    return new_flows

def get_shed_mw(action_ids):
    total = 0
    for aid in action_ids:
        if 'mw' in actions[aid]:
            total += actions[aid]['mw']
    return total

# Test all combinations
solutions = []

# No action
if check_feasible(flows):
    solutions.append(([], flows.copy(), 0))

# Single actions
for aid in actions:
    new_flows = apply_action(flows, aid)
    shed = get_shed_mw([aid])
    if check_feasible(new_flows) and shed <= 25:
        solutions.append(([aid], new_flows, shed))

# Two actions
for combo in combinations(actions.keys(), 2):
    new_flows = flows.copy()
    for aid in combo:
        new_flows = apply_action(new_flows, aid)
    shed = get_shed_mw(list(combo))
    if check_feasible(new_flows) and shed <= 25:
        solutions.append((list(combo), new_flows, shed))

# Sort by shed_mw (minimize), then by action count (minimize)
solutions.sort(key=lambda x: (x[2], len(x[0])))

print("All feasible solutions (sorted by shed_mw, then action count):")
for sol in solutions:
    print(f"Actions: {sol[0]}, Shed: {sol[2]} MW, Flows: {sol[1]}")

if solutions:
    best = solutions[0]
    print(f"\nBest solution:")
    print(f"Actions: {best[0]}")
    print(f"Load shed: {best[2]} MW")
    print(f"Final flows: {best[1]}")
    print(f"All limits satisfied: True")
