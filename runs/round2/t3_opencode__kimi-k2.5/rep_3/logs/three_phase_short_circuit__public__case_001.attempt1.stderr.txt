import math

# Given data
base_mva = 100.0
voltage_kv = 10.5
voltage_factor_cmax = 1.1
voltage_factor_cmin = 1.0
fault_resistance_pu = 0.005

# Positive sequence impedances
impedances = [
    {"r": 0.018, "x": 0.155},   # grid_source
    {"r": 0.012, "x": 0.095},   # station_transformer
    {"r": 0.002, "x": 0.011},   # bus_duct
]

# Sum resistances and reactances
r_total = sum(z["r"] for z in impedances)
x_total = sum(z["x"] for z in impedances)

print(f"R_total = {r_total}")
print(f"X_total = {x_total}")

# Network impedance for bolted fault (no fault resistance)
z_bolted_mag_pu = math.sqrt(r_total**2 + x_total**2)
print(f"Z_bolted_mag_pu = {z_bolted_mag_pu}")

# Base current calculation
i_base = base_mva / (math.sqrt(3) * voltage_kv)
print(f"I_base = {i_base} kA")

# Maximum current (bolted fault, c_max)
ik_max_ka = (voltage_factor_cmax / z_bolted_mag_pu) * i_base
print(f"ik_max_ka = {ik_max_ka}")

# Maximum apparent power
sk_max_mva = math.sqrt(3) * voltage_kv * ik_max_ka
print(f"sk_max_mva = {sk_max_mva}")

# Minimum current (with fault resistance, c_min)
r_with_fault = r_total + fault_resistance_pu
z_min_mag_pu = math.sqrt(r_with_fault**2 + x_total**2)
print(f"Z_min_mag_pu = {z_min_mag_pu}")

ik_min_ka = (voltage_factor_cmin / z_min_mag_pu) * i_base
print(f"ik_min_ka = {ik_min_ka}")

# Final values
print("\n=== Final Values ===")
print(f"ik_max_ka: {ik_max_ka:.6f}")
print(f"sk_max_mva: {sk_max_mva:.6f}")
print(f"ik_min_ka: {ik_min_ka:.6f}")
print(f"z_bolted_mag_pu: {z_bolted_mag_pu:.6f}")
print(f"z_min_mag_pu: {z_min_mag_pu:.6f}")
