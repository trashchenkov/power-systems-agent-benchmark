import math
import json

# Input data
voltage_kv = 10.5
source_r = 0.045
source_x = 0.42
transformer_r = 0.018
transformer_x = 0.16
feeder_r_per_km = 0.18
feeder_x_per_km = 0.32
max_load_current_a = 360.0
breaker_safety_factor = 1.1
relay_load_margin = 1.25
minimum_fault_sensitivity = 2.0

# Fault locations
fault_locations = [
    {"id": "near_bus_fault", "distance_km": 0.2},
    {"id": "mid_feeder_fault", "distance_km": 4.0},
    {"id": "remote_end_fault", "distance_km": 8.0}
]

# Available breaker ratings
available_breaker_ratings_ka = [8.0, 12.5, 16.0, 20.0, 25.0, 31.5]

# Calculate impedances and fault currents
results = []

for fault in fault_locations:
    distance_km = fault["distance_km"]
    
    # Total impedance to fault point
    total_r = source_r + transformer_r + (feeder_r_per_km * distance_km)
    total_x = source_x + transformer_x + (feeder_x_per_km * distance_km)
    
    # Magnitude of total impedance
    z_total = math.sqrt(total_r**2 + total_x**2)
    
    # Three-phase fault current: V_ll / (sqrt(3) * |Z_total|)
    # V_ll is in kV, so convert to V: V_ll * 1000
    v_ll = voltage_kv * 1000  # volts
    i_3ph = v_ll / (math.sqrt(3) * z_total)  # amperes
    i_3ph_ka = i_3ph / 1000  # convert to kA
    
    # Two-phase fault current: sqrt(3)/2 * I_3ph
    i_2ph = (math.sqrt(3) / 2) * i_3ph
    i_2ph_ka = i_2ph / 1000  # convert to kA
    
    results.append({
        "id": fault["id"],
        "distance_km": distance_km,
        "z_total_ohm": z_total,
        "i_3ph_a": i_3ph,
        "i_3ph_ka": i_3ph_ka,
        "i_2ph_a": i_2ph,
        "i_2ph_ka": i_2ph_ka
    })
    
    print(f"Fault: {fault['id']} at {distance_km} km")
    print(f"  Z_total = {z_total:.4f} ohm")
    print(f"  I_3ph = {i_3ph:.2f} A = {i_3ph_ka:.4f} kA")
    print(f"  I_2ph = {i_2ph:.2f} A = {i_2ph_ka:.4f} kA")
    print()

# Find maximum 3-phase fault current and minimum 2-phase fault current
max_3ph_fault_ka = max(r["i_3ph_ka"] for r in results)
min_2ph_fault_ka = min(r["i_2ph_ka"] for r in results)
min_2ph_fault_a = min(r["i_2ph_a"] for r in results)

print(f"Maximum 3-phase fault current: {max_3ph_fault_ka:.4f} kA")
print(f"Minimum 2-phase fault current: {min_2ph_fault_ka:.4f} kA ({min_2ph_fault_a:.2f} A)")

# Select breaker interrupting rating
# Must be at least max_3ph_fault_ka * breaker_safety_factor
required_breaker_rating = max_3ph_fault_ka * breaker_safety_factor
print(f"\nRequired breaker rating (with safety factor {breaker_safety_factor}): {required_breaker_rating:.4f} kA")

# Find smallest available rating above required
breaker_interrupting_rating_ka = None
for rating in sorted(available_breaker_ratings_ka):
    if rating >= required_breaker_rating:
        breaker_interrupting_rating_ka = rating
        break

print(f"Selected breaker rating: {breaker_interrupting_rating_ka} kA")

# Calculate relay pickup constraints
# Must be at least max_load_current_a * relay_load_margin
relay_pickup_min = max_load_current_a * relay_load_margin
print(f"\nRelay pickup minimum (load margin): {relay_pickup_min:.2f} A")

# Must be no greater than min_2ph_fault_a / minimum_fault_sensitivity
relay_pickup_max = min_2ph_fault_a / minimum_fault_sensitivity
print(f"Relay pickup maximum (sensitivity): {relay_pickup_max:.2f} A")

# Select a relay pickup between min and max
# Choose a value in the middle or close to min
relay_pickup_a_primary = (relay_pickup_min + relay_pickup_max) / 2
print(f"Selected relay pickup: {relay_pickup_a_primary:.2f} A")

# Output results
output = {
    "max_3ph_fault_ka": round(max_3ph_fault_ka, 4),
    "min_2ph_fault_ka": round(min_2ph_fault_ka, 4),
    "breaker_interrupting_rating_ka": breaker_interrupting_rating_ka,
    "relay_pickup_a_primary": round(relay_pickup_a_primary, 2),
    "rationale": f"Calculated short-circuit currents for three fault locations. Max 3-phase fault: {max_3ph_fault_ka:.4f} kA (at near bus). Min 2-phase fault: {min_2ph_fault_ka:.4f} kA (at remote end). Breaker selected: {breaker_interrupting_rating_ka} kA (smallest standard rating above {required_breaker_rating:.4f} kA required). Relay pickup: {relay_pickup_a_primary:.2f} A (between {relay_pickup_min:.2f} A load margin min and {relay_pickup_max:.2f} A sensitivity max)."
}

print("\n--- Final Output ---")
print(json.dumps(output, indent=2))

# Save to submission.json
with open('<sandbox_a1>/submission.json', 'w') as f:
    json.dump(output, f, indent=2)

print("\nSaved to submission.json")
