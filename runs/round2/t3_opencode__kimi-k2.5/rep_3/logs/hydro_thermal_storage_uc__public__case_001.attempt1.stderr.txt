# Load submission
with open('submission.json') as f:
    sub = json.load(f)

# Verify constraints
hours = [
    {'hour_id': 'h1', 'load_mw': 120, 'hydro_mw': 35, 'storage_discharge_mw': 10},
    {'hour_id': 'h2', 'load_mw': 150, 'hydro_mw': 40, 'storage_discharge_mw': 20},
    {'hour_id': 'h3', 'load_mw': 130, 'hydro_mw': 30, 'storage_discharge_mw': 15},
]

coal = {'pmin': 40, 'pmax': 100, 'cost': 22, 'emission': 0.75}
gas = {'pmin': 20, 'pmax': 80, 'cost': 38, 'emission': 0.35}

print('Verification:')
for h in hours:
    hid = h['hour_id']
    residual = h['load_mw'] - h['hydro_mw'] - h['storage_discharge_mw']
    coal_d = sub['thermal_dispatch_mw'][hid]['coal']
