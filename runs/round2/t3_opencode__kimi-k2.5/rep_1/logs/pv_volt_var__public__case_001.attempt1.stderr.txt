def get_q_fraction(v, curve):
    """Linear interpolation of q_fraction for given voltage"""
    if v <= curve[0]["voltage_pu"]:
        return curve[0]["q_fraction"]
    if v >= curve[-1]["voltage_pu"]:
        return curve[-1]["q_fraction"]
    
    for i in range(len(curve)-1):
        v1, q1 = curve[i]["voltage_pu"], curve[i]["q_fraction"]
        v2, q2 = curve[i+1]["voltage_pu"], curve[i+1]["q_fraction"]
        if v1 <= v <= v2:
            # Linear interpolation
            if v2 == v1:
                return q1
            t = (v - v1) / (v2 - v1)
            return q1 + t * (q2 - q1)
    return curve[-1]["q_fraction"]
