import math

# Load task data
turbine = {
    "count": 8,
    "rated_mw_each": 2.5
}

    "cut_in_m_s": 3.0,
    "rated_m_s": 12.0,
    "cut_out_m_s": 25.0
}

reference_air_density = 1.225
farm_capacity = turbine["count"] * turbine["rated_mw_each"]

history = [
    {"timestamp": "h-4", "wind_speed_m_s": 6.2, "air_density_kg_m3": 1.21, "actual_power_mw": 0.98},
    {"timestamp": "h-3", "wind_speed_m_s": 7.4, "air_density_kg_m3": 1.2, "actual_power_mw": 2.35},
    {"timestamp": "h-2", "wind_speed_m_s": 8.6, "air_density_kg_m3": 1.19, "actual_power_mw": 4.55},
    {"timestamp": "h-1", "wind_speed_m_s": 9.1, "air_density_kg_m3": 1.18, "actual_power_mw": 5.85}
]

weather_forecast = [
    {"timestamp": "t01", "wind_speed_m_s": 8.0, "air_density_kg_m3": 1.19},
    {"timestamp": "t02", "wind_speed_m_s": 9.5, "air_density_kg_m3": 1.18},
    {"timestamp": "t03", "wind_speed_m_s": 11.0, "air_density_kg_m3": 1.17},
    {"timestamp": "t04", "wind_speed_m_s": 13.2, "air_density_kg_m3": 1.16}
]

def compute_power(wind_speed, air_density):
    """Compute power using the power curve formula."""
    # Corrected speed
    v_c = wind_speed * (air_density / reference_air_density) ** (1/3)
    
    # Check cut-in and cut-out
    if v_c < power_curve["cut_in_m_s"] or v_c >= power_curve["cut_out_m_s"]:
        return 0.0
    
    # Check rated
    if v_c >= power_curve["rated_m_s"]:
        return turbine["count"] * turbine["rated_mw_each"]
    
    # Cubic region
    rated_power = turbine["count"] * turbine["rated_mw_each"]
    power = rated_power * ((v_c - power_curve["cut_in_m_s"]) / (power_curve["rated_m_s"] - power_curve["cut_in_m_s"])) ** 3
    return power

def clip(value, min_val, max_val):
    """Clip value to range [min_val, max_val]."""
    return max(min_val, min(max_val, value))

# Calculate bias from history
biases = []
print("Computing bias from history:")
for record in history:
    power_estimate = compute_power(record["wind_speed_m_s"], record["air_density_kg_m3"])
    bias = record["actual_power_mw"] - power_estimate
    print(f"  {record['timestamp']}: v={record['wind_speed_m_s']}, rho={record['air_density_kg_m3']}, v_c={record['wind_speed_m_s'] * (record['air_density_kg_m3'] / reference_air_density) ** (1/3):.4f}, power_estimate={power_estimate:.4f}, actual={record['actual_power_mw']}, bias={bias:.4f}")
    biases.append(bias)

mean_bias = sum(biases) / len(biases)
print(f"\nMean bias: {mean_bias:.6f}")

# Compute forecasts
forecast_mw_by_timestamp = {}
print("\nComputing forecasts:")
for record in weather_forecast:
    power_estimate = compute_power(record["wind_speed_m_s"], record["air_density_kg_m3"])
    v_c = record["wind_speed_m_s"] * (record["air_density_kg_m3"] / reference_air_density) ** (1/3)
    forecast = clip(power_estimate + mean_bias, 0, farm_capacity)
    forecast_mw_by_timestamp[record["timestamp"]] = round(forecast, 6)
    print(f"  {record['timestamp']}: v={record['wind_speed_m_s']}, rho={record['air_density_kg_m3']}, v_c={v_c:.4f}, power_estimate={power_estimate:.4f}, forecast={forecast:.6f}")

# Calculate total energy (sum of hourly forecasts * forecast_step_hours)
forecast_step_hours = 1.0
total_forecast_mwh = sum(forecast_mw_by_timestamp.values()) * forecast_step_hours

# Find peak timestamp
peak_timestamp = max(forecast_mw_by_timestamp.keys(), key=lambda x: forecast_mw_by_timestamp[x])

print(f"\nResults:")
print(f"forecast_mw_by_timestamp: {forecast_mw_by_timestamp}")
print(f"total_forecast_mwh: {total_forecast_mwh:.6f}")
print(f"peak_timestamp: {peak_timestamp}")

# Prepare submission
submission = {
    "forecast_mw_by_timestamp": forecast_mw_by_timestamp,
    "total_forecast_mwh": round(total_forecast_mwh, 6),
    "peak_timestamp": peak_timestamp
}

print(f"\nSubmission JSON:")
print(json.dumps(submission, indent=2))
