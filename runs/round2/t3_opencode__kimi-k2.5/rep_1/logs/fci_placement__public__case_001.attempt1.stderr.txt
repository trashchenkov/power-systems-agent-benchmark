import itertools

# Data from task.json
fault_sections = {
    "s12": {"annual_fault_rate": 0.18, "customers_interrupted": 120},
    "s23": {"annual_fault_rate": 0.14, "customers_interrupted": 70},
    "s24": {"annual_fault_rate": 0.12, "customers_interrupted": 90},
    "s45": {"annual_fault_rate": 0.09, "customers_interrupted": 40}
}

candidate_locations = {
    "n1": {"s12": 18, "s23": 16, "s24": 15, "s45": 14},
    "n2": {"s12": 6, "s23": 28, "s24": 26, "s45": 23},
    "n4": {"s12": 0, "s23": 4, "s24": 18, "s45": 34},
    "n3": {"s12": 0, "s23": 24, "s24": 2, "s45": 2}
}

budget = 2

def calculate_objective(selected):
    """Calculate expected customer-minutes saved for selected locations"""
    total = 0
    section_gains = {}
    
    for section, data in fault_sections.items():
        # Get max gain among selected locations for this section
        max_gain = max(candidate_locations[loc][section] for loc in selected)
        section_gains[section] = max_gain
        
        # Calculate contribution
        contribution = data["annual_fault_rate"] * data["customers_interrupted"] * max_gain
        total += contribution
    
    return total, section_gains

# Enumerate all combinations of size budget
candidates = list(candidate_locations.keys())
best_value = -1
best_selection = None
best_gains = None

print("Evaluating all combinations of 2 locations:")
print("-" * 60)

for combo in itertools.combinations(candidates, budget):
    value, gains = calculate_objective(list(combo))
    print(f"{combo}: total={value:.2f}, gains={gains}")
    
    if value > best_value:
        best_value = value
        best_selection = combo
        best_gains = gains

print("-" * 60)
print(f"\nBest selection: {best_selection}")
print(f"Expected customer-minutes saved: {best_value}")
print(f"Section gains: {best_gains}")

# Prepare submission
import json
submission = {
    "selected_locations": list(best_selection),
    "expected_customer_minutes_saved": best_value,
    "section_patrol_time_reduction_min": best_gains
}

print("\nSubmission JSON:")
print(json.dumps(submission, indent=2))
