import cmath
import math

# Given data
base_mva = 50.0  # MVA
voltage_kv = 13.8  # kV
prefault_v_pu = 1.0
fault_resistance_pu = 0.04

# Sequence impedances in pu
Z1 = complex(0.042, 0.31)
Z2 = complex(0.045, 0.29)
Z0 = complex(0.12, 0.82)

# Calculate total impedance for SLG fault
# I1 = I2 = I0 = V_prefault / (Z1 + Z2 + Z0 + 3*Zf)
Z_total = Z1 + Z2 + Z0 + 3 * fault_resistance_pu
print(f"Z1 = {Z1}")
print(f"Z2 = {Z2}")
print(f"Z0 = {Z0}")
print(f"3*Zf = {3 * fault_resistance_pu}")
print(f"Z_total = {Z_total}")

# Calculate sequence currents in pu
I1_pu = prefault_v_pu / Z_total
I2_pu = I1_pu
I0_pu = I1_pu

print(f"\nI1_pu = {I1_pu}")
print(f"I2_pu = {I2_pu}")
print(f"I0_pu = {I0_pu}")

# Calculate base current
I_base = base_mva / (math.sqrt(3) * voltage_kv)
print(f"\nI_base = {I_base} kA")

# Convert to kA
I1_ka = I1_pu * I_base
I2_ka = I2_pu * I_base
I0_ka = I0_pu * I_base

print(f"\nI1_ka = {I1_ka}")
print(f"I2_ka = {I2_ka}")
print(f"I0_ka = {I0_ka}")

# Calculate magnitudes and angles
