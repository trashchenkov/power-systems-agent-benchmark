import math

# Given data
base_mva = 100.0
voltage_kv = 10.5
voltage_factor_cmax = 1.1
voltage_factor_cmin = 1.0
fault_resistance_pu = 0.005

# Positive sequence impedances
impedances = [
    {"r_pu": 0.018, "x_pu": 0.155},  # grid_source
    {"r_pu": 0.012, "x_pu": 0.095},  # station_transformer
    {"r_pu": 0.002, "x_pu": 0.011},  # bus_duct
]

# Sum impedances for Z_network (bolted fault)
r_total = sum(z["r_pu"] for z in impedances)
x_total = sum(z["x_pu"] for z in impedances)

print(f"R_total = {r_total}")
print(f"X_total = {x_total}")

# Z_network magnitude (bolted fault)
z_bolted_mag_pu = math.sqrt(r_total**2 + x_total**2)
print(f"|Z_network| (bolted) = {z_bolted_mag_pu}")

# Z with fault resistance (for min current)
r_total_with_fault = r_total + fault_resistance_pu
z_min_mag_pu = math.sqrt(r_total_with_fault**2 + x_total**2)
print(f"|Z_min| (with fault resistance) = {z_min_mag_pu}")

# Base current calculation
i_base = base_mva / (math.sqrt(3) * voltage_kv)
print(f"I_base = {i_base} kA")

# Maximum current (bolted fault, c_max = 1.1)

print(f"Ik_max = {ik_max_ka} kA")

# Maximum short-circuit apparent power

print(f"Sk_max = {sk_max_mva} MVA")

# Minimum current (with fault resistance, c_min = 1.0)

print(f"Ik_min = {ik_min_ka} kA")

print(f"z_min_mag_pu: {z_min_mag_pu}")
