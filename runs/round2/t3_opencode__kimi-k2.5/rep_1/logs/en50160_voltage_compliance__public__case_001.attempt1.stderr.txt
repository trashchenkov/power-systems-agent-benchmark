        above_normal.append(ts)

for v in voltage_series:
    val = v["voltage_pu"]
    if val < absolute_lower or val > absolute_upper:
        absolute_violations.append((v["timestamp"], val))

best_tap = None

    # Check for absolute violations
    has_absolute_violation = False
    for av in adjusted_voltages:
        if av < absolute_lower or av > absolute_upper:
            has_absolute_violation = True
            break
    
    # Count within normal band
    count_within = sum(1 for av in adjusted_voltages if normal_lower <= av <= normal_upper)
    
    print(f"\nTap {tap:+.3f}: adjusted range [{min(adjusted_voltages):.3f}, {max(adjusted_voltages):.3f}]")
    print(f"  Within normal: {count_within}/{total}")
    print(f"  Has absolute violation: {has_absolute_violation}")
    
    if not has_absolute_violation and count_within > best_count:
        best_count = count_within
        best_tap = tap

print(f"\n=== Best tap: {best_tap} (count: {best_count}) ===")

result = {
