import itertools

# Fault sections data
sections = {
    "s12": {"annual_fault_rate": 0.18, "customers_interrupted": 120},
    "s23": {"annual_fault_rate": 0.14, "customers_interrupted": 70},
    "s24": {"annual_fault_rate": 0.12, "customers_interrupted": 90},
    "s45": {"annual_fault_rate": 0.09, "customers_interrupted": 40}
}

# Candidate locations
candidates = {
    "n1": {"s12": 18, "s23": 16, "s24": 15, "s45": 14},
    "n2": {"s12": 6, "s23": 28, "s24": 26, "s45": 23},
    "n3": {"s12": 0, "s23": 24, "s24": 2, "s45": 2},
    "n4": {"s12": 0, "s23": 4, "s24": 18, "s45": 34}
}

budget = 2

best_score = -1
best_combo = None
best_reductions = None

for combo in itertools.combinations(candidates.keys(), budget):
    # For each section, find max gain among selected locations
    section_reductions = {}
    for section in sections:
        max_gain = max(candidates[loc][section] for loc in combo)
        section_reductions[section] = max_gain
    
    # Calculate objective
    score = 0
    for section, data in sections.items():
        contribution = data["annual_fault_rate"] * data["customers_interrupted"] * section_reductions[section]
        score += contribution
    
    print(f"Combo {combo}: score={score:.4f}, reductions={section_reductions}")
    
    if score > best_score:
        best_score = score
        best_combo = combo
        best_reductions = section_reductions

print(f"\nBest combination: {best_combo}")
print(f"Best score: {best_score:.4f}")
print(f"Section reductions: {best_reductions}")
