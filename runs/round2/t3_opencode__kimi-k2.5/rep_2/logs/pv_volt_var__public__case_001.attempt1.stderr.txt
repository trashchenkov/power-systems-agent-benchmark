import json
import math

# Load submission
with open('submission.json') as f:
    sub = json.load(f)

print("Volt-VAR Curve:")
for p in sub['volt_var_curve']:
    print(f"  V={p['voltage_pu']}, Q={p['q_fraction']}")

# Check monotonicity
print("\nMonotonicity check:")
voltages = [p['voltage_pu'] for p in sub['volt_var_curve']]
q_fractions = [p['q_fraction'] for p in sub['volt_var_curve']]

# Strictly increasing voltage
v_strict = all(voltages[i] < voltages[i+1] for i in range(len(voltages)-1))
print(f"  Voltage strictly increasing: {v_strict}")

# Non-increasing q_fraction
q_mono = all(q_fractions[i] >= q_fractions[i+1] for i in range(len(q_fractions)-1))
print(f"  Q fraction non-increasing: {q_mono}")

# Check each timestep
timesteps = [
    {"id": "t_low", "base_voltage_pu": 0.94, "pv_kw": 60.0, "inverter_kva": 100.0, "voltage_sensitivity_pu_per_kvar": 0.0007},
    {"id": "t_nom", "base_voltage_pu": 1.0, "pv_kw": 80.0, "inverter_kva": 100.0, "voltage_sensitivity_pu_per_kvar": 0.00055},
    {"id": "t_high", "base_voltage_pu": 1.07, "pv_kw": 60.0, "inverter_kva": 100.0, "voltage_sensitivity_pu_per_kvar": 0.0007}
]

voltage_limits = {"min": 0.95, "max": 1.05}

def get_q_command(v_base, curve):
    """Linear interpolation to get q_fraction from curve"""
    voltages = [p['voltage_pu'] for p in curve]
    q_values = [p['q_fraction'] for p in curve]
    
    if v_base <= voltages[0]:
        return q_values[0]
    if v_base >= voltages[-1]:
        return q_values[-1]
    
    for i in range(len(voltages)-1):
        if voltages[i] <= v_base <= voltages[i+1]:
            # Linear interpolation
            t = (v_base - voltages[i]) / (voltages[i+1] - voltages[i])
            return q_values[i] + t * (q_values[i+1] - q_values[i])
    return 0

print("\nTimestep analysis:")
all_feasible = True
for t in timesteps:
    kva = t['inverter_kva']
    kw = t['pv_kw']
    q_max = math.sqrt(kva**2 - kw**2)
    
    q_frac = get_q_command(t['base_voltage_pu'], sub['volt_var_curve'])
    q_kvar = q_frac * kva
    
    # Check capability constraint
    q_capable = abs(q_kvar) <= q_max
    
    # Calculate corrected voltage
    v_corrected = t['base_voltage_pu'] + q_kvar * t['voltage_sensitivity_pu_per_kvar']
    
    # Check voltage limits
    v_in_limits = voltage_limits['min'] <= v_corrected <= voltage_limits['max']
    
    print(f"\n  {t['id']}:")
    print(f"    Base V: {t['base_voltage_pu']}")
    print(f"    Q fraction: {q_frac:.4f}, Q kvar: {q_kvar:.2f}")
    print(f"    Q max: {q_max:.2f}, Q capable: {q_capable}")
    print(f"    Corrected V: {v_corrected:.4f}")
    print(f"    V in limits: {v_in_limits}")
    
    if not q_capable or not v_in_limits:
        all_feasible = False

print(f"\nAll constraints satisfied: {all_feasible}")
