with open('task.json') as f:

case=task['wind_forecast_case']
ref=case['reference_air_density_kg_m3']
count=case['turbine']['count']; rated_each=case['turbine']['rated_mw_each']; cap=count*rated_each

def est(ws,rho):
    vc=ws*((rho/ref)**(1/3))
    if vc < cut or vc >= out: return 0.0
    if vc >= rated: return cap
    return cap*((vc-cut)/(rated-cut))**3
hist=[]

    resid=h['actual_power_mw']-e
    hist.append((h['timestamp'],e,h['actual_power_mw'],resid))
bias=sum(r for *_,r in hist)/len(hist)
