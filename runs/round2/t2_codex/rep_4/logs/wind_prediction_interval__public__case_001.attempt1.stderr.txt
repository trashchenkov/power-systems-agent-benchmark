import json, math
with open('task.json') as f: t=json.load(f)
case=t['wind_interval_case']
ref=case['reference_air_density_kg_m3']
count=case['turbine']['count']; rated_each=case['turbine']['rated_mw_each']; cap=count*rated_each
pc=case['power_curve']; ci=pc['cut_in_m_s']; rated=pc['rated_m_s']; co=pc['cut_out_m_s']

def power(speed,dens):
    vc=speed*((dens/ref)**(1/3))
    if vc < ci or vc >= co: return 0.0
    if vc >= rated: return cap
    return cap*((vc-ci)/(rated-ci))**3

def quant(xs,q):
    xs=sorted(xs); n=len(xs); pos=q*(n-1); lo=math.floor(pos); hi=math.ceil(pos)
    if lo==hi: return xs[lo]
    return xs[lo]+(xs[hi]-xs[lo])*(pos-lo)
res={}
widths=[]

print('avg', sum(widths)/len(widths))
print(json.dumps({'prediction_interval_mw_by_timestamp':res,'average_interval_width_mw':sum(widths)/len(widths)}, indent=2))
