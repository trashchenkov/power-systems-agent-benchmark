import json, math
from pathlib import Path
p=Path('task.json')
task=json.loads(p.read_text())
case=task['wind_interval_case']
ref=case['reference_air_density_kg_m3']
count=case['turbine']['count']
rated_each=case['turbine']['rated_mw_each']
cap=count*rated_each
curve=case['power_curve']
cut_in=curve['cut_in_m_s']; rated=curve['rated_m_s']; cut_out=curve['cut_out_m_s']

def power(ws, rho):
    vc=ws*((rho/ref)**(1/3))
    if vc < cut_in or vc >= cut_out:
        return 0.0
    if vc >= rated:
        return cap
    return cap*((vc-cut_in)/(rated-cut_in))**3

def quantile(xs,q):
    xs=sorted(xs)
    n=len(xs)
    pos=q*(n-1)
    lo=math.floor(pos); hi=math.ceil(pos)
    if lo==hi:
        return xs[lo]
    frac=pos-lo
    return xs[lo]*(1-frac)+xs[hi]*frac

out={}
widths=[]
for h in case['horizons']:
    vals=[power(s['wind_speed_m_s'], s['air_density_kg_m3']) for s in h['scenarios']]
    p10=quantile(vals,0.10)
    p50=quantile(vals,0.50)
    p90=quantile(vals,0.90)
    out[h['timestamp']]={'p10':p10,'p50':p50,'p90':p90}
    widths.append(p90-p10)
res={
  'prediction_interval_mw_by_timestamp': out,
  'average_interval_width_mw': sum(widths)/len(widths),
  'rationale': 'Applied the specified density-corrected turbine power curve to each scenario, then computed p10/p50/p90 using linear interpolation over sorted scenario powers.'
}
Path('submission.json').write_text(json.dumps(res, indent=2) + '\\n')
