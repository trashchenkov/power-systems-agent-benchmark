import json, math
with open('task.json') as f:
    t=json.load(f)['wind_forecast_case']

def est(ws,rho):
    vc=ws*((rho/ref)**(1/3))
    if vc < pc['cut_in_m_s'] or vc >= pc['cut_out_m_s']:
        p=0.0
    elif vc >= pc['rated_m_s']:
        p=cap
    else:

    return p, vc
hist=[]
