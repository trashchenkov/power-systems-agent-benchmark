base_mva=100.0
voltage_kv=10.5
cmax=1.1
cmin=1.0
parts=[(0.018,0.155),(0.012,0.095),(0.002,0.011)]
r=sum(p[0] for p in parts)
x=sum(p[1] for p in parts)
z_b=math.hypot(r,x)

ik_max=cmax/z_b*i_base
