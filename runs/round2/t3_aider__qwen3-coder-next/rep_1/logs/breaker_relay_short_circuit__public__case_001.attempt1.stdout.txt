import math

# Given parameters
V_ll = 10.5  # kV
Z_s = complex(0.045, 0.42)  # Source Thevenin impedance (Ω)
Z_t = complex(0.018, 0.16)  # Transformer impedance (Ω)
z_f = complex(0.18, 0.32)   # Feeder impedance per km (Ω/km)

fault_locations = [
    {"id": "near_bus_fault", "distance_km": 0.2},
    {"id": "mid_feeder_fault", "distance_km": 4.0},
    {"id": "remote_end_fault", "distance_km": 8.0}
]

# Available breaker ratings
available_breakers = [8.0, 12.5, 16.0, 20.0, 25.0, 31.5]  # kA

# Selection rules
breaker_safety_factor = 1.1
relay_load_margin = 1.25
min_fault_sensitivity = 2.0
max_load_current = 360.0  # A

# Calculate fault currents for each location
fault_results = []

for loc in fault_locations:
    d = loc["distance_km"]
    # Total impedance up to fault location
    Z_total = Z_s + Z_t + z_f * d
    Z_total_mag = abs(Z_total)
    
    # Three-phase fault current (kA)
    I_3ph = (V_ll * 1000) / (math.sqrt(3) * Z_total_mag) / 1000  # Convert to kA
    # Two-phase fault current (kA)
    I_2ph = (math.sqrt(3)/2) * I_3ph
    
    fault_results.append({
        "id": loc["id"],
        "I_3ph_kA": I_3ph,
        "I_2ph_kA": I_2ph
    })

# Find max 3-phase and min 2-phase fault currents
max_3ph_fault = max(f["I_3ph_kA"] for f in fault_results)
min_2ph_fault = min(f["I_2ph_kA"] for f in fault_results)

# Breaker selection: must be >= max_3ph_fault * safety_factor
required_breaker_rating = max_3ph_fault * breaker_safety_factor
# Find smallest available rating that meets requirement
breaker_rating = min(r for r in available_breakers if r >= 
required_breaker_rating)

# Relay pickup calculation
# Must be >= max_load_current * relay_load_margin
min_relay_pickup = max_load_current * relay_load_margin
# Must be <= min_2ph_fault_current / min_fault_sensitivity
# Convert min_2ph_fault from kA to A
max_relay_pickup = (min_2ph_fault * 1000) / min_fault_sensitivity

# Choose relay pickup (between min and max)
relay_pickup = min_relay_pickup  # Use minimum required to maximize sensitivity

# Format results
submission = {
    "max_3ph_fault_ka": round(max_3ph_fault, 3),
    "min_2ph_fault_ka": round(min_2ph_fault, 3),
    "breaker_interrupting_rating_ka": breaker_rating,
    "relay_pickup_a_primary": round(relay_pickup, 1)
}

print(f"Fault results: {fault_results}")
print(f"Max 3-phase fault: {max_3ph_fault:.3f} kA")
print(f"Min 2-phase fault: {min_2ph_fault:.3f} kA")
