import numpy as np

# Network data
branches = [
    {"id": "l12", "from_bus": "b1", "to_bus": "b2", "x_pu": 0.1},
    {"id": "l13", "from_bus": "b1", "to_bus": "b3", "x_pu": 0.125},
    {"id": "l23", "from_bus": "b2", "to_bus": "b3", "x_pu": 0.2}
]

# Build H matrix for DC state estimation
# State variables: theta_2, theta_3 (theta_1 = 0 as reference)

H = np.zeros((5, 2))

# flow_l12: P12 = (theta_1 - theta_2)/x12 = -theta_2/0.1
# derivative w.r.t. theta_2: -1/0.1 = -10
H[0, 0] = -10.0  # d(flow_l12)/d(theta_2)
H[0, 1] = 0.0    # d(flow_l12)/d(theta_3)

# flow_l13: P13 = (theta_1 - theta_3)/x13 = -theta_3/0.125
H[1, 0] = 0.0    # d(flow_l13)/d(theta_2)
H[1, 1] = -8.0   # d(flow_l13)/d(theta_3) = -1/0.125

# flow_l23: P23 = (theta_2 - theta_3)/x23
H[2, 0] = 1.0/0.2  # d(flow_l23)/d(theta_2) = 1/0.2 = 5
H[2, 1] = -1.0/0.2 # d(flow_l23)/d(theta_3) = -1/0.2 = -5

# injection_b2: P2 = P12 + P23 = (theta_1-theta_2)/x12 + (theta_2-theta_3)/x23
# dP2/dtheta_2 = -1/0.1 + 1/0.2 = -10 + 5 = -5
# dP2/dtheta_3 = -1/0.2 = -5
H[3, 0] = -5.0
H[3, 1] = -5.0

# injection_b3: P3 = P13 + P23 = (theta_1-theta_3)/x13 + (theta_2-theta_3)/x23
# dP3/dtheta_2 = 1/0.2 = 5
# dP3/dtheta_3 = -1/0.125 - 1/0.2 = -8 - 5 = -13
H[4, 0] = 5.0
H[4, 1] = -13.0

# Measurement vector z
z = np.array([0.402, 0.557, 0.151, -0.1, -0.713])

# Sigma values
sigma = np.array([0.01, 0.01, 0.01, 0.01, 0.01])

# Weight matrix W (diagonal with 1/sigma^2)
W = np.diag(1.0 / (sigma ** 2))

# G = H^T * W * H
G = H.T @ W @ H

# Solve for theta: G * theta = H^T * W * z
theta = np.linalg.solve(G, H.T @ W @ z)

theta_2 = theta[0]
theta_3 = theta[1]

# Compute residuals: r = z - H * theta
residuals = z - H @ theta

# Compute variance of residuals: Var(r) = sigma^2 - h * G^-1 * h^T
G_inv = np.linalg.inv(G)

# For each measurement, compute normalized residual
normalized_residuals = []
for i in range(5):
    # variance of residual i
    var_r_i = sigma[i]**2 - H[i] @ G_inv @ H[i].T
    if var_r_i > 0:
        norm_res = abs(residuals[i]) / np.sqrt(var_r_i)
    else:
        norm_res = float('inf')
    normalized_residuals.append(norm_res)

# Iteratively remove measurements with largest normalized residual > 3.0
flagged = []
remaining_indices = list(range(5))

while True:
    if not remaining_indices:
        break
    
    # Compute current estimate with remaining measurements
    H_remaining = H[remaining_indices, :]
    z_remaining = z[remaining_indices]
    sigma_remaining = sigma[remaining_indices]
    W_remaining = np.diag(1.0 / (sigma_remaining ** 2))
    
    G_remaining = H_remaining.T @ W_remaining @ H_remaining
    theta_remaining = np.linalg.solve(G_remaining, H_remaining.T @ W_remaining @
z_remaining)
    
    # Compute residuals for remaining measurements
    residuals_remaining = z_remaining - H_remaining @ theta_remaining
    
    # Compute normalized residuals for remaining measurements
    G_remaining_inv = np.linalg.inv(G_remaining)
    norm_residuals_remaining = []
    for idx in remaining_indices:
        i = remaining_indices.index(idx)
        var_r = sigma[idx]**2 - H[idx] @ G_remaining_inv @ H[idx].T
        if var_r > 0:
            norm_res = abs(residuals_remaining[i]) / np.sqrt(var_r)
        else:
            norm_res = float('inf')
        norm_residuals_remaining.append((idx, norm_res))
    
    # Find largest normalized residual
    max_idx, max_res = max(norm_residuals_remaining, key=lambda x: x[1])
    
    if max_res <= 3.0:
        break
    
    flagged.append(max_idx)
    remaining_indices.remove(max_idx)

# Final state estimate
H_final = H[remaining_indices, :]
z_final = z[remaining_indices]
sigma_final = sigma[remaining_indices]
W_final = np.diag(1.0 / (sigma_final ** 2))

G_final = H_final.T @ W_final @ H_final
theta_final = np.linalg.solve(G_final, H_final.T @ W_final @ z_final)

# Map measurement IDs to indices
measurement_ids = ["flow_l12", "flow_l13", "flow_l23", "inj_b2", "inj_b3"]
flagged_ids = [measurement_ids[i] for i in flagged]

print("Initial estimate:")
print(f"theta_2 = {theta_2:.10f}")
print(f"theta_3 = {theta_3:.10f}")
print("\nNormalized residuals (initial):")
for i, (mid, nr) in enumerate(zip(measurement_ids, normalized_residuals)):
    print(f"{mid}: {nr:.10f}")
print(f"\nFlagged measurements: {flagged_ids}")
print(f"Remaining indices: {remaining_indices}")
print(f"\nFinal corrected state:")
print(f"theta_2 = {theta_final[0]:.10f}")
print(f"theta_3 = {theta_final[1]:.10f}")
