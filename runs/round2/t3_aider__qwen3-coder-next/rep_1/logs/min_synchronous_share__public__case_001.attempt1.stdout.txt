```python
import itertools

# Given data
demand_mw = 1000.0
ibr_capacity_mva = 720.0
nominal_frequency_hz = 50.0
largest_loss_mw = 180.0
max_rocof_hz_per_s = 2.0
min_scr = 2.25
grid_short_circuit_mva_base = 1180.0

units = [
    {"unit_id": "g1", "p_mw": 120.0, "h_s": 4.5, "short_circuit_mva": 280.0},
    {"unit_id": "g2", "p_mw": 180.0, "h_s": 5.0, "short_circuit_mva": 360.0},
    {"unit_id": "g3", "p_mw": 240.0, "h_s": 3.8, "short_circuit_mva": 440.0},
    {"unit_id": "g4", "p_mw": 90.0, "h_s": 6.0, "short_circuit_mva": 190.0}
]

best_solution = None
min_share = float('inf')

# Try all combinations of units
for r in range(1, len(units) + 1):
    for combo in itertools.combinations(units, r):
        # Calculate total synchronous power
        total_sync_power = sum(u["p_mw"] for u in combo)
        sync_share_pct = (total_sync_power / demand_mw) * 100
        
        # Calculate system inertia (GWh)
        inertia_gws = sum(u["h_s"] * u["p_mw"] for u in combo) / 1000
        
        # Calculate RoCoF
        if inertia_gws > 0:
            rocof = largest_loss_mw * nominal_frequency_hz / (2 * inertia_gws * 
1000)
        else:
            rocof = float('inf')
        
        # Calculate SCR
        total_short_circuit = grid_short_circuit_mva_base + 
sum(u["short_circuit_mva"] for u in combo)
        scr = total_short_circuit / ibr_capacity_mva
        
        # Check constraints
        if rocof <= max_rocof_hz_per_s and scr >= min_scr:
            if sync_share_pct < min_share:
                min_share = sync_share_pct
                best_solution = {
                    "online_units": [u["unit_id"] for u in combo],
                    "min_synchronous_share_pct": round(sync_share_pct, 1),
                    "system_inertia_gws": round(inertia_gws, 2),
                    "rocof_hz_per_s": round(rocof, 4),
                    "short_circuit_ratio": round(scr, 2)
                }

print(f"Best solution: {best_solution}")
print(f"Minimum share: {min_share}%")
```
