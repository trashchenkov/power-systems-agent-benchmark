import sys
sys.setrecursionlimit(10000)

time_periods = ['t00', 't05', 't10', 't15']
interval_hours = 0.0833333333
load_mw = {'t00': 38.0, 't05': 42.0, 't10': 48.0, 't15': 44.0}
pv_mw = {'t00': 18.0, 't05': 15.0, 't10': 11.0, 't15': 9.0}
diesel = {'pmin': 12.0, 'pmax': 55.0, 'cost': 92.0}
battery = {
    'initial_soc': 5.0, 'soc_min': 2.5, 'soc_max': 8.0,
    'charge_pmax': 6.0, 'discharge_pmax': 10.0,
    'eta_charge': 0.94, 'eta_discharge': 0.9
}

def simulate(dispatch):
    soc = battery['initial_soc']
    total_cost = 0.0
    
    for t in time_periods:
        load = load_mw[t]
        pv = pv_mw[t]
        batt_mw = dispatch[t]['battery_mw']
        pv_curtailment = dispatch[t].get('pv_curtailment_mw', 0.0)
        dt = interval_hours
        
        if batt_mw >= 0:
            soc -= batt_mw * dt
            diesel_mw = load + pv_curtailment - pv - batt_mw
        else:
            soc += (-batt_mw) * dt * battery['eta_charge']
            diesel_mw = load + pv_curtailment - pv + (-batt_mw)
        
        if diesel_mw < diesel['pmin'] - 1e-6 or diesel_mw > diesel['pmax'] + 1e-6:
            return None, None
        if soc < battery['soc_min'] - 1e-6 or soc > battery['soc_max'] + 1e-6:
            return None, None
        
        total_cost += diesel_mw * dt * diesel['cost']
        dispatch[t]['diesel_mw'] = diesel_mw
        dispatch[t]['soc_mwh'] = soc
    
    return total_cost, dispatch

battery_actions = list(range(-6, 11, 2))
curtail_actions = list(range(0, 9))

best_cost = float('inf')
best_dispatch = None

for b0 in battery_actions:
    for b1 in battery_actions:

                for c0 in curtail_actions:
