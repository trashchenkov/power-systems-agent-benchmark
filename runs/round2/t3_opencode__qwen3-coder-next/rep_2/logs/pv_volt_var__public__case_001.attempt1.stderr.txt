# Design Volt-VAR curve
# Voltage breakpoints: strictly increasing
# q_fraction: monotonic non-increasing

# Curve: [(v1, q1), (v2, q2), ...]
# t_low uses voltage ~0.94, needs positive Q
# t_nom uses voltage ~1.0, needs minimal Q  
# t_high uses voltage ~1.07, needs negative Q

# Let's define points that will work:
# At low voltage (0.94), use full injection (q=0.8)
# At nominal voltage (1.0), use zero (q=0)
# At high voltage (1.07), use full absorption (q=-0.8)

curve = [
    {"voltage_pu": 0.92, "q_fraction": 0.8},
    {"voltage_pu": 0.98, "q_fraction": 0.0},
    {"voltage_pu": 1.04, "q_fraction": -0.8},
    {"voltage_pu": 1.08, "q_fraction": -1.0}
]

# Verify monotonicity
voltages = [p["voltage_pu"] for p in curve]
qfracs = [p["q_fraction"] for p in curve]

print("Voltage points (should be strictly increasing):", voltages)
print("q_fractions (should be non-increasing):", qfracs)
print("Voltages strictly increasing:", all(voltages[i] < voltages[i+1] for i in range(len(voltages)-1)))
print("q_fractions non-increasing:", all(qfracs[i] >= qfracs[i+1] for i in range(len(qfracs)-1)))

# Now test against timesteps
timesteps = [
    {"id": "t_low", "base_voltage": 0.94, "pv_kw": 60.0, "kva": 100.0, "sensitivity": 0.0007},
    {"id": "t_nom", "base_voltage": 1.0, "pv_kw": 80.0, "kva": 100.0, "sensitivity": 0.00055},
    {"id": "t_high", "base_voltage": 1.07, "pv_kw": 60.0, "kva": 100.0, "sensitivity": 0.0007}
]

def interpolate_q(v, curve):
    """Interpolate q_fraction for a given voltage"""
    if v <= curve[0]["voltage_pu"]:
        return curve[0]["q_fraction"]
    if v >= curve[-1]["voltage_pu"]:
        return curve[-1]["q_fraction"]
    for i in range(len(curve)-1):
        if curve[i]["voltage_pu"] <= v <= curve[i+1]["voltage_pu"]:
            # Linear interpolation
            t = (v - curve[i]["voltage_pu"]) / (curve[i+1]["voltage_pu"] - curve[i]["voltage_pu"])
            return curve[i]["q_fraction"] + t * (curve[i+1]["q_fraction"] - curve[i]["q_fraction"])
    return curve[-1]["q_fraction"]

print("\nTesting curve against timesteps:")
for ts in timesteps:
    q_frac = interpolate_q(ts["base_voltage"], curve)
    q_command = q_frac * ts["kva"]
    v_corrected = ts["base_voltage"] + ts["sensitivity"] * q_command
    q_max = (ts["kva"]**2 - ts["pv_kw"]**2)**0.5
    
    print(f"{ts['id']}: v_base={ts['base_voltage']}, q_frac={q_frac:.2f}, q_cmd={q_command:.2f}, v_corr={v_corrected:.4f}")
    print(f"  q_max={q_max:.2f}, within limits? {abs(q_command) <= q_max}")
    print(f"  voltage in [0.95, 1.05]? {0.95 <= v_corrected <= 1.05}")
