from itertools import combinations

budget = 2
candidates = ["m12", "m23", "m31", "pinj2"]

attacks = [
    {"attack_id": "a_theta2", "risk_weight": 7, "detectable_by": ["m12", "pinj2"]},
    {"attack_id": "a_theta3", "risk_weight": 6, "detectable_by": ["m31"]},
    {"attack_id": "a_load_shift", "risk_weight": 4, "detectable_by": ["m23", "pinj2"]},
    {"attack_id": "a_stealth_pair", "risk_weight": 9, "detectable_by": ["m12", "m31"]}
]

best_score = -1
best_combo = None
best_covered = None

for combo in combinations(candidates, budget):
    covered_attacks = []
    total_risk = 0
    for attack in attacks:
        if any(m in combo for m in attack["detectable_by"]):
            covered_attacks.append(attack["attack_id"])
            total_risk += attack["risk_weight"]
    
    print(f"{combo}: covered={covered_attacks}, risk={total_risk}")
    
    if total_risk > best_score or (total_risk == best_score and combo < best_combo):
        best_score = total_risk
        best_combo = combo
        best_covered = covered_attacks

print(f"\nBest: {best_combo}, covered={best_covered}, risk={best_score}")

submission = {
    "protected_meters": sorted(list(best_combo)),
    "covered_attack_vectors": sorted(best_covered),
    "covered_risk_weight": best_score,
    "rationale": f"Selected meters {best_combo} to cover attacks {best_covered} with total risk weight {best_score}"
}

import json
with open("<packets>/protected_meter_placement__public__case_001/submission.json", "w") as f:
    json.dump(submission, f, indent=2)
