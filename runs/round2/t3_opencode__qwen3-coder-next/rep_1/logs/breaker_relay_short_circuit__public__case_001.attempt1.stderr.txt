import math

# Given impedances
V_ll = 10.5  # kV
Z_source = complex(0.045, 0.42)  # ohms

r_km = 0.18  # ohm/km
x_km = 0.32  # ohm/km
length_km = 8.0

# Fault locations
fault_locations = [0.2, 4.0, 8.0]

# Calculate feeder impedance per location
results = []
for d in fault_locations:
    Z_feeder = complex(r_km * d, x_km * d)
    Z_total = Z_source + Z_transformer + Z_feeder
    Z_mag = abs(Z_total)
    
    # Three-phase fault: I_3ph = V_ll / (sqrt(3) * Z_total)
    I_3ph = (V_ll * 1000) / (math.sqrt(3) * Z_mag)  # amps
    I_3ph_ka = I_3ph / 1000  # kA
    
    # Two-phase fault: I_2ph = sqrt(3)/2 * I_3ph
    I_2ph = (math.sqrt(3)/2) * I_3ph
    I_2ph_ka = I_2ph / 1000  # kA
    
    results.append({
        'location': d,
        'Z_total': Z_total,
        'I_3ph_ka': I_3ph_ka,
        'I_2ph_ka': I_2ph_ka
    })
    print(f"Distance {d} km: Z={Z_total}, I_3ph={I_3ph_ka:.4f} kA, I_2ph={I_2ph_ka:.4f} kA")
