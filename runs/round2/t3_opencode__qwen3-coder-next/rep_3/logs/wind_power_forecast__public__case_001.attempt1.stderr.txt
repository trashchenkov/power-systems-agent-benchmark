import json

# Given parameters
farm_capacity = 8 * 2.5  # 20 MW
cut_in = 3.0
rated = 12.0
cut_out = 25.0
ref_density = 1.225

# Power curve function
def calc_power(v_wind, rho, count=8, rated_mw=2.5):
    v_c = v_wind * (rho / ref_density) ** (1/3)
    if v_c < cut_in or v_c >= cut_out:
        return 0.0
    if v_c >= rated:
        return count * rated_mw
    return count * rated_mw * ((v_c - cut_in) / (rated - cut_in)) ** 3

# History data
history = [
    {"timestamp": "h-4", "wind_speed_m_s": 6.2, "air_density_kg_m3": 1.21, "actual_power_mw": 0.98},
    {"timestamp": "h-3", "wind_speed_m_s": 7.4, "air_density_kg_m3": 1.2, "actual_power_mw": 2.35},
    {"timestamp": "h-2", "wind_speed_m_s": 8.6, "air_density_kg_m3": 1.19, "actual_power_mw": 4.55},
    {"timestamp": "h-1", "wind_speed_m_s": 9.1, "air_density_kg_m3": 1.18, "actual_power_mw": 5.85}
]

# Calculate bias: mean of (actual - power_curve_estimate)
bias_sum = 0
for h in history:
    est = calc_power(h["wind_speed_m_s"], h["air_density_kg_m3"])
    bias_sum += h["actual_power_mw"] - est
bias = bias_sum / len(history)
print(f"Bias: {bias}")

# Weather forecast
weather = [
    {"timestamp": "t01", "wind_speed_m_s": 8.0, "air_density_kg_m3": 1.19},
    {"timestamp": "t02", "wind_speed_m_s": 9.5, "air_density_kg_m3": 1.18},
    {"timestamp": "t03", "wind_speed_m_s": 11.0, "air_density_kg_m3": 1.17},
    {"timestamp": "t04", "wind_speed_m_s": 13.2, "air_density_kg_m3": 1.16}
]

# Calculate forecasts
forecasts = {}
for w in weather:
    est = calc_power(w["wind_speed_m_s"], w["air_density_kg_m3"])
    fc = max(0, min(est + bias, farm_capacity))
    forecasts[w["timestamp"]] = round(fc, 2)

print(f"Forecasts: {forecasts}")

# Total MWh
step_hours = 1.0
total = sum(forecasts.values()) * step_hours
print(f"Total: {total}")

# Peak timestamp
peak_ts = max(forecasts, key=forecasts.get)
print(f"Peak: {peak_ts} @ {forecasts[peak_ts]}")

submission = {
    "forecast_mw_by_timestamp": forecasts,
    "total_forecast_mwh": round(total, 2),
    "peak_timestamp": peak_ts
}

print(json.dumps(submission, indent=2))

with open("submission.json", "w") as f:
    json.dump(submission, f, indent=2)
