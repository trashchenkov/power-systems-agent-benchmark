import numpy as np

# Given values
base_mva = 50.0
voltage_kv = 13.8
fault_resistance_pu = 0.04
prefault_voltage_pu = 1.0

# Sequence impedances
Z1 = complex(0.042, 0.31)
Z2 = complex(0.045, 0.29)
Z0 = complex(0.12, 0.82)

# Fault impedance
Zf = complex(fault_resistance_pu, 0)

# Sequence currents for single-line-to-ground fault
Z_total = Z1 + Z2 + Z0 + 3*Zf
I_seq = prefault_voltage_pu / Z_total

print(f"Z_total = {Z_total}")
print(f"I_seq (pu) = {I_seq}")

# Convert to kA
# I_base = S_base / (sqrt(3) * V_base)
I_base = base_mva / (np.sqrt(3) * voltage_kv)
print(f"I_base = {I_base} kA")

# Sequence currents in kA
I1_ka = I_seq * I_base
I2_ka = I_seq * I_base
I0_ka = I_seq * I_base

print(f"I1 (kA) = {I1_ka}")
print(f"I2 (kA) = {I2_ka}")
print(f"I0 (kA) = {I0_ka}")

# Magnitudes and angles
I1_mag = np.abs(I1_ka)
I1_angle = np.angle(I1_ka, deg=True)
I2_mag = np.abs(I2_ka)
I2_angle = np.angle(I2_ka, deg=True)
I0_mag = np.abs(I0_ka)
I0_angle = np.angle(I0_ka, deg=True)
