import json

# Case parameters
f0 = 50.0  # Hz

postfault_pmax = 1.5  # pu

# Step 1: Calculate delta0 (pre-fault rotor angle)
delta0_rad = math.asin(Pm / prefault_pmax)
delta0_deg = math.degrees(delta0_rad)

# Step 2: Calculate delta_max (maximum post-fault angle)
delta_max_rad = math.pi - math.asin(Pm / postfault_pmax)
delta_max_deg = math.degrees(delta_max_rad)

# Step 3: Calculate delta_cr (critical clearing angle) using equal-area criterion
# cos(delta_cr) = (Pm*(delta_max - delta0) + postfault_pmax*cos(delta_max)) / postfault_pmax

delta_cr_rad = math.acos(cos_delta_cr)
