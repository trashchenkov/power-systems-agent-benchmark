base_mva = 50.0

z0 = complex(0.12, 0.82)
zf = complex(fault_resistance_pu, 0)

# I1 = I2 = I0 = V / (Z1 + Z2 + Z0 + 3*Zf)
z_total = z1 + z2 + z0 + 3 * zf
i_seq_pu = prefault_voltage_pu / z_total

print(f"Z1 = {z1}")
print(f"Z2 = {z2}")
print(f"Z0 = {z0}")
print(f"Zf = {zf}")

# Convert to kA: Ibase = Sbase / (sqrt(3) * Vbase)
sqrt3 = math.sqrt(3)
i_base_ka = base_mva / (sqrt3 * voltage_kv)
print(f"\nIbase = {base_mva} / ({sqrt3} * {voltage_kv}) = {i_base_ka} kA")

i1_ka = i_seq_pu * i_base_ka
