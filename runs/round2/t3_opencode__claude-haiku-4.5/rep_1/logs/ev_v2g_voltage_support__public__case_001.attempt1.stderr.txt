import json
import math

# Input data
target_voltage_pu = 0.98
base_voltage_pu = {
    "bus_2": 0.944,
    "bus_3": 0.952
}
q_sensitivity_pu_per_kvar = {
    "bus_2": 0.0016,
    "bus_3": 0.0012
}
evs = {
    "evse_1": {
        "bus": "bus_2",
        "s_rating_kva": 18.0,
        "active_power_kw": 10.0
    },
    "evse_2": {
        "bus": "bus_2",
        "s_rating_kva": 12.0,
        "active_power_kw": 6.0
    },
    "evse_3": {
        "bus": "bus_3",
        "s_rating_kva": 16.0,
        "active_power_kw": 8.0
    }
}

# Step 1: Calculate aggregate reactive capability for each bus
# For each EV, reactive capability = sqrt(s_rating^2 - active_power^2)
bus_reactive_capability = {}
for bus in base_voltage_pu.keys():
    bus_reactive_capability[bus] = 0.0

for evse_id, evse_data in evs.items():
    bus = evse_data["bus"]
    s_rating = evse_data["s_rating_kva"]
    p_active = evse_data["active_power_kw"]
    
    # Calculate reactive capability
    q_capability = math.sqrt(s_rating**2 - p_active**2)
    bus_reactive_capability[bus] += q_capability
    
    print(f"{evse_id} at {bus}: S={s_rating} kVA, P={p_active} kW, Q_capability={q_capability:.3f} kVAr")

print(f"\nAggregate reactive capability by bus:")
for bus, q_cap in bus_reactive_capability.items():
    print(f"  {bus}: {q_cap:.3f} kVAr")

# Step 2: Calculate reactive command needed to reach target voltage
# Using linear relationship: dV = q_sensitivity * q_dispatch
# Rearranging: q_dispatch = (V_target - V_base) / q_sensitivity
q_dispatch_kvar = {}
corrected_voltage_pu = {}

print(f"\nVoltage support calculations:")
for bus in base_voltage_pu.keys():
    v_base = base_voltage_pu[bus]
    v_target = target_voltage_pu
    sensitivity = q_sensitivity_pu_per_kvar[bus]
    capability = bus_reactive_capability[bus]
    
    # Calculate required reactive power to reach target
    dv_needed = v_target - v_base
    q_required = dv_needed / sensitivity
    
    # Clip by capability (can only inject positive reactive power for voltage support)
    q_dispatch = max(min(q_required, capability), -capability)
    
    # Calculate actual voltage correction
    dv_actual = sensitivity * q_dispatch
    v_corrected = v_base + dv_actual
    
    q_dispatch_kvar[bus] = q_dispatch
    corrected_voltage_pu[bus] = v_corrected
    
    print(f"  {bus}:")
    print(f"    V_base: {v_base} pu, V_target: {v_target} pu")
    print(f"    V_gap: {dv_needed} pu, Sensitivity: {sensitivity} pu/kVAr")
    print(f"    Q_required: {q_required:.3f} kVAr, Capability: {capability:.3f} kVAr")
    print(f"    Q_dispatch (clipped): {q_dispatch:.3f} kVAr")
    print(f"    V_corrected: {v_corrected:.6f} pu")

# Step 3: Calculate total absolute reactive effort
total_abs_q_kvar = sum(abs(q) for q in q_dispatch_kvar.values())

print(f"\nTotal absolute Q: {total_abs_q_kvar:.3f} kVAr")

# Step 4: Prepare submission JSON
submission = {
    "q_dispatch_kvar": q_dispatch_kvar,
    "corrected_voltage_pu": corrected_voltage_pu,
    "total_abs_q_kvar": round(total_abs_q_kvar, 6),
    "rationale": "Linear voltage sensitivity used to compute reactive dispatch needed to reach target voltage. Each bus reactive command clipped by aggregate EV inverter reactive capability."
}

print("\nFinal submission:")
print(json.dumps(submission, indent=2))

# Save to file
with open('submission.json', 'w') as f:
    json.dump(submission, f, indent=2)

print("\nSubmission saved to submission.json")
