def get_reachable(start, active_branches):
    reachable = set()
    queue = [start]
    reachable.add(start)
    
    while queue:
        current = queue.pop(0)
        for branch_id in active_branches:
            if branch_id not in branches:
                continue
            branch = branches[branch_id]
            next_bus = None
            if branch['from_bus'] == current:
                next_bus = branch['to_bus']
            elif branch['to_bus'] == current:
                next_bus = branch['from_bus']

                queue.append(next_bus)
