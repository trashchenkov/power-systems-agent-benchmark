import json

# Load task data
with open('task.json', 'r') as f:
    task = json.load(f)

case = task['dispatch_uncertainty_case']
generators = case['generators']
scenarios = case['scenarios']
time_periods = case['time_periods']
reserve_requirement = case['reserve_requirement_mw']

# Compute worst-case net load for each time period
worst_case_net_load = {}
for t in time_periods:
    max_net_load = 0
    for scenario in scenarios:
        load = scenario['load_mw'][t]
        renewable = scenario['renewable_mw'][t]
        net_load = load - renewable
        max_net_load = max(max_net_load, net_load)
    worst_case_net_load[t] = max_net_load

print("Worst-case net load by time period:")
print(worst_case_net_load)

# For each time period, we need to dispatch generators such that:
# - Total thermal generation >= worst_case_net_load + reserve_requirement
# - Each generator respects pmin/pmax
# - Minimize total variable cost

# This is a linear programming problem
# For now, let's use a greedy approach: use cheapest generators first

dispatch_mw = {}
for gen in generators:
    dispatch_mw[gen['id']] = {}

total_cost = 0
worst_case_margins = {}

for t in time_periods:
    required_generation = worst_case_net_load[t] + reserve_requirement
    
    # Sort generators by cost (cheapest first)
    sorted_gens = sorted(generators, key=lambda g: g['variable_cost_per_mwh'])
    
    # Greedy dispatch: use cheapest generators first
    current_generation = 0
    for gen in sorted_gens:
        if current_generation >= required_generation:
            # Set to pmin to stay online if already committed
            dispatch_mw[gen['id']][t] = gen['pmin_mw']
            current_generation += gen['pmin_mw']
        else:
            # Dispatch as much as needed from this generator
            needed = required_generation - current_generation
            dispatch = min(needed + gen['pmin_mw'], gen['pmax_mw'])
            dispatch_mw[gen['id']][t] = dispatch
            current_generation += dispatch
    
    # Calculate actual margin for this time period
    actual_generation = sum(dispatch_mw[gen['id']][t] for gen in generators)
    margin = actual_generation - worst_case_net_load[t]
    worst_case_margins[t] = margin
    
    # Add cost for this time period
    for gen in generators:
        total_cost += dispatch_mw[gen['id']][t] * gen['variable_cost_per_mwh']
    
    print(f"\nTime period {t}:")
    print(f"  Required generation: {required_generation} MW")
    print(f"  Actual generation: {actual_generation} MW")
    print(f"  Margin: {margin} MW")
    for gen in generators:
        print(f"    {gen['id']}: {dispatch_mw[gen['id']][t]} MW @ ${gen['variable_cost_per_mwh']}/MWh")

print(f"\nTotal cost: ${total_cost}")

# Prepare submission
submission = {
    "dispatch_mw": dispatch_mw,
    "worst_case_margin_mw": worst_case_margins,
    "total_cost": total_cost,
    "rationale": "Greedy dispatch using cheapest generators first to meet worst-case net load plus reserve requirement"
}

print("\nSubmission:")
print(json.dumps(submission, indent=2))

# Write submission.json
with open('submission.json', 'w') as f:
    json.dump(submission, f, indent=2)

print("\nSubmission written to submission.json")

import json

with open('task.json', 'r') as f:
    task = json.load(f)

with open('submission.json', 'r') as f:
    submission = json.load(f)

case = task['dispatch_uncertainty_case']
generators = case['generators']
time_periods = case['time_periods']
reserve_requirement = case['reserve_requirement_mw']

# Verify constraints
print("=== Constraint Verification ===\n")

# Check pmin/pmax
print("1. Generator pmin/pmax limits:")
for gen in generators:
    gen_id = gen['id']
    pmin = gen['pmin_mw']
    pmax = gen['pmax_mw']
    for t in time_periods:
        dispatch = submission['dispatch_mw'][gen_id][t]
        if dispatch < pmin or dispatch > pmax:
            print(f"  VIOLATION: {gen_id} at {t}: {dispatch} MW (limits: {pmin}-{pmax})")
        else:
            print(f"  OK: {gen_id} at {t}: {dispatch} MW (limits: {pmin}-{pmax})")

# Check coverage of worst-case net load + reserve
print("\n2. Coverage of worst-case net load + reserve:")
worst_case_net_load = {}
for t in time_periods:
    max_net_load = 0
    for scenario in case['scenarios']:
        load = scenario['load_mw'][t]
        renewable = scenario['renewable_mw'][t]
        net_load = load - renewable
        max_net_load = max(max_net_load, net_load)
    worst_case_net_load[t] = max_net_load

for t in time_periods:
    total_dispatch = sum(submission['dispatch_mw'][gen['id']][t] for gen in generators)
    required = worst_case_net_load[t] + reserve_requirement
    margin = total_dispatch - worst_case_net_load[t]
    
    if total_dispatch < required:
        print(f"  VIOLATION: {t}: dispatch {total_dispatch} < required {required}")
    else:
        print(f"  OK: {t}: dispatch {total_dispatch} >= required {required}")
    
    if abs(margin - submission['worst_case_margin_mw'][t]) > 0.01:
        print(f"    WARNING: margin mismatch - calculated {margin}, reported {submission['worst_case_margin_mw'][t]}")

    gen_id = gen['id']
    cost_per_mwh = gen['variable_cost_per_mwh']

        calculated_cost += dispatch * cost_per_mwh

else:

print("\n=== Summary ===")
print(json.dumps(submission, indent=2))
