with open('task.json', 'r') as f:

deployed_fci_locations = case['deployed_fci_locations']
fault_sections = case['fault_sections']
fci_locations = case['fci_locations']

# Calculate baseline metrics
baseline_customer_minutes = 0
baseline_customer_interruptions = 0

for section_id, section_data in fault_sections.items():
    annual_fault_rate = section_data['annual_fault_rate']
    customers_interrupted = section_data['customers_interrupted']
    base_restoration_minutes = section_data['base_restoration_minutes']
    
    customer_minutes = annual_fault_rate * customers_interrupted * base_restoration_minutes
    customer_interruptions = annual_fault_rate * customers_interrupted
    
    baseline_customer_minutes += customer_minutes
    baseline_customer_interruptions += customer_interruptions

baseline_saidi = baseline_customer_minutes / total_customers
baseline_caidi = baseline_customer_minutes / baseline_customer_interruptions if baseline_customer_interruptions > 0 else 0
baseline_saifi = baseline_customer_interruptions / total_customers

print(f"Baseline CAIDI: {baseline_caidi}")
print(f"Baseline SAIFI: {baseline_saifi}")

with_fci_customer_minutes = 0
with_fci_customer_interruptions = baseline_customer_interruptions  # Same as baseline

for section_id, section_data in fault_sections.items():

    customers_interrupted = section_data['customers_interrupted']
    base_restoration_minutes = section_data['base_restoration_minutes']
    
    # Find the best (maximum) restoration time reduction from deployed FCI locations
    max_reduction = 0
    for fci_location in deployed_fci_locations:

            max_reduction = max(max_reduction, reduction)
    
    # Apply reduction but respect minimum restoration time
    reduced_restoration_minutes = max(base_restoration_minutes - max_reduction, minimum_restoration_minutes)
