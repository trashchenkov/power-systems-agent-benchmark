import json

with open('task.json', 'r') as f:
    task = json.load(f)

with open('submission.json', 'r') as f:
    submission = json.load(f)

case = task['v2g_outage_case']
evs = case['evs']
interval_hours = case['interval_hours']
critical_load_kw = case['critical_load_kw']
dispatch_kw = submission['dispatch_kw']

print("Constraint Verification:")
print("=" * 60)

# Track SOC over time
current_soc = {ev_id: ev['initial_soc_kwh'] for ev_id, ev in evs.items()}
total_energy_served = 0.0

for t in case['time_periods']:
    load = critical_load_kw[t]
    print(f"\n{t}: Load = {load} kW")
    
    power_provided = 0.0
    for ev_id, ev in evs.items():
        dispatch = dispatch_kw[t][ev_id]
        power_provided += dispatch
        
        # Check power limits
        if dispatch > 0:  # Discharging
            if dispatch > ev['discharge_max_kw']:
                print(f"  ✗ {ev_id}: discharge {dispatch} > max {ev['discharge_max_kw']}")
            energy_out = dispatch * interval_hours * ev['discharge_efficiency']
            new_soc = current_soc[ev_id] - energy_out
            if new_soc < ev['reserve_soc_kwh']:
                print(f"  ✗ {ev_id}: SOC would be {new_soc:.2f} < reserve {ev['reserve_soc_kwh']}")
            else:
                print(f"  ✓ {ev_id}: dispatch {dispatch} kW, SOC: {current_soc[ev_id]:.2f} → {new_soc:.2f}")
            current_soc[ev_id] = new_soc
        elif dispatch < 0:  # Charging
            if -dispatch > ev['charge_max_kw']:
                print(f"  ✗ {ev_id}: charge {-dispatch} > max {ev['charge_max_kw']}")
            energy_in = -dispatch * interval_hours / ev['charge_efficiency']
            new_soc = current_soc[ev_id] + energy_in
            if new_soc > ev['capacity_kwh']:
                print(f"  ✗ {ev_id}: SOC would be {new_soc:.2f} > capacity {ev['capacity_kwh']}")
            else:
                print(f"  ✓ {ev_id}: dispatch {dispatch} kW, SOC: {current_soc[ev_id]:.2f} → {new_soc:.2f}")
            current_soc[ev_id] = new_soc
        else:
            print(f"  ✓ {ev_id}: no dispatch, SOC: {current_soc[ev_id]:.2f}")
    
    energy_served = min(power_provided, load) * interval_hours
    total_energy_served += energy_served
    print(f"  Power provided: {power_provided} kW → Energy served: {energy_served} kWh")

print("\n" + "=" * 60)
print(f"Total energy served: {total_energy_served} kWh (submitted: {submission['served_energy_kwh']} kWh)")
print(f"Final SOC: {current_soc}")
print(f"Submitted: {submission['final_soc_kwh']}")

# Check if matches
eps = 1e-5
if abs(total_energy_served - submission['served_energy_kwh']) < eps:
    print("✓ Served energy matches")
else:
    print(f"✗ Served energy mismatch: {total_energy_served} vs {submission['served_energy_kwh']}")
