    is_decreasing_q = q1 >= q2 - monotonic_tolerance
    is_monotonic = is_increasing_v and is_decreasing_q
    if not is_monotonic:
        all_monotonic = False
    status = "✓" if is_monotonic else "✗"

    base_v = ts['base_voltage_pu']

        v2, q2 = curve_points[i+1]
