import json
from itertools import combinations

# Load task
with open('task.json', 'r') as f:
    task = json.load(f)

inputs = task['public_inputs']['synchronous_share_case']

# Extract parameters
demand_mw = inputs['demand_mw']
ibr_capacity_mva = inputs['ibr_capacity_mva']
nominal_frequency_hz = inputs['nominal_frequency_hz']
largest_loss_mw = inputs['largest_loss_mw']
max_rocof_hz_per_s = inputs['max_rocof_hz_per_s']
min_scr = inputs['min_scr']
grid_short_circuit_mva_base = inputs['grid_short_circuit_mva_base']
sync_units = inputs['sync_units']

# All possible units
all_units = [(u['unit_id'], u['p_mw'], u['h_s'], u['short_circuit_mva']) for u in sync_units]

# Generate all possible non-empty subsets and check feasibility
best_solution = None
best_share = float('inf')

for r in range(1, len(all_units) + 1):
    for combo in combinations(all_units, r):
        unit_ids = [c[0] for c in combo]
        total_p_mw = sum(c[1] for c in combo)
        
        # Calculate inertia
        total_h_p = sum(c[1] * c[2] for c in combo)
        inertia_gws = total_h_p / 1000.0
        
        # Calculate RoCoF
        rocof_hz_per_s = (largest_loss_mw * nominal_frequency_hz) / (2 * inertia_gws * 1000)
        
        # Calculate SCR
        total_scr_mva = grid_short_circuit_mva_base + sum(c[3] for c in combo)
        scr = total_scr_mva / ibr_capacity_mva
        
        # Check constraints
        rocof_ok = rocof_hz_per_s <= max_rocof_hz_per_s
        scr_ok = scr >= min_scr
        
        if rocof_ok and scr_ok:
            share = total_p_mw / demand_mw
            if share < best_share:
                best_share = share
                best_solution = {
                    'online_units': unit_ids,
                    'p_mw': total_p_mw,
                    'inertia_gws': inertia_gws,
                    'rocof_hz_per_s': rocof_hz_per_s,
                    'scr': scr,
                    'share_pct': share * 100
                }

if best_solution:
    print("Best solution found:")
    print(f"  Units: {best_solution['online_units']}")
    print(f"  Total P: {best_solution['p_mw']} MW")
    print(f"  Inertia: {best_solution['inertia_gws']} GWs")
    print(f"  RoCoF: {best_solution['rocof_hz_per_s']} Hz/s")
    print(f"  SCR: {best_solution['scr']}")
    print(f"  Share: {best_solution['share_pct']}%")
else:
    print("No feasible solution found!")
