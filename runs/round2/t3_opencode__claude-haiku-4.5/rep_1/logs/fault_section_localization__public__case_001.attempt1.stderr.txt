import json

# Load the task
with open('<sandbox_a1>/task.json') as f:
    task = json.load(f)

case = task['fault_localization_case']
sections = case['sections']
fci_locations = case['fci_locations']
observed_fci_state = case['observed_fci_state']
outage_complaints = case['outage_complaint_nodes']
customer_nodes = case['customer_nodes']
source_node = case['source_node']

print("=== Network Structure ===")
print(f"Source: {source_node}")
print(f"Sections: {sections}")
print(f"FCI Locations: {fci_locations}")
print(f"Observed FCI States: {observed_fci_state}")
print(f"Outage Complaints: {outage_complaints}")
print()

# Build a graph representation
# For each section, determine upstream and downstream nodes
def get_upstream_path(section_id):
    """Get all nodes and sections upstream (toward source) from the start of a section"""
    section = sections[section_id]
    path = []
    current = section['from']
    
    # Walk backward from the start of the section to source
    visited = set()
    while current != source_node and current not in visited:
        path.append(current)
        visited.add(current)
        # Find section that leads to current
        found = False
        for sid, s in sections.items():
            if s['to'] == current and s['from'] not in visited:
                current = s['from']
                found = True
                break
        if not found:
            break
    path.append(source_node)
    return path

def get_downstream_customers(section_id):
    """Get all customer nodes downstream from the end of a section"""
    section = sections[section_id]
    end_node = section['to']
    
    # BFS to find all reachable nodes from end_node
    reachable = set()
    queue = [end_node]
    visited = set()
    
    while queue:
        node = queue.pop(0)
        if node in visited:
            continue
        visited.add(node)
        reachable.add(node)
        
        # Find sections that start from this node
        for sid, s in sections.items():
            if s['from'] == node and s['to'] not in visited:
                queue.append(s['to'])
    
    # Return only customer nodes
    return [n for n in reachable if n in customer_nodes]

print("=== Analysis for Each Section ===\n")

results = []

for section_id in sorted(sections.keys()):
    section = sections[section_id]
    print(f"--- Section {section_id}: {section['from']} → {section['to']} ---")
    
    # Get upstream nodes (before the fault)
    upstream = get_upstream_path(section_id)
    print(f"Upstream nodes: {upstream}")
    
    # FCIs that should detect fault (those at upstream nodes)
    expected_fci_trips = {}
    for fci_id, fci_node in fci_locations.items():
        expected_fci_trips[fci_id] = fci_node in upstream
    print(f"Expected FCI states: {expected_fci_trips}")
    
    # Downstream customer nodes
    downstream_customers = get_downstream_customers(section_id)
    print(f"Downstream customer nodes: {downstream_customers}")
    
    # Calculate mismatch
    fci_mismatch = 0
    for fci_id, expected in expected_fci_trips.items():
        observed = observed_fci_state[fci_id]
        if expected != observed:
            fci_mismatch += 1
    
    complaint_mismatch = 0
    for complaint_node in outage_complaints:
        if complaint_node not in downstream_customers:
            complaint_mismatch += 1
    
    for customer in downstream_customers:
        if customer not in outage_complaints:
            complaint_mismatch += 1
    
    total_mismatch = fci_mismatch + complaint_mismatch
    print(f"FCI mismatch: {fci_mismatch}, Complaint mismatch: {complaint_mismatch}, Total: {total_mismatch}")
    
    results.append({
        'section_id': section_id,
        'upstream': upstream,
        'expected_fci': expected_fci_trips,
        'downstream_customers': downstream_customers,
        'fci_mismatch': fci_mismatch,
        'complaint_mismatch': complaint_mismatch,
        'total_mismatch': total_mismatch
    })
    print()

print("=== Results ===")
best = min(results, key=lambda x: x['total_mismatch'])
print(f"Best match: Section {best['section_id']} with total mismatch {best['total_mismatch']}")
print(f"Faulted section: {best['section_id']}")
print(f"Affected customer nodes: {best['downstream_customers']}")
