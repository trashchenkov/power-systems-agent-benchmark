import json
from pathlib import Path

# Read task.json
with open('task.json', 'r') as f:
    task = json.load(f)

network = task['network']
buses = network['buses']
branches = network['branches']
shunts = network['shunts']

n_buses = len(buses)
bus_index = {bus: i for i, bus in enumerate(buses)}

# Initialize Ybus as complex matrix (n x n)
ybus = [[complex(0, 0) for _ in range(n_buses)] for _ in range(n_buses)]

# Process branches
for branch in branches:
    from_bus = branch['from_bus']
    to_bus = branch['to_bus']
    r = branch['r_pu']
    x = branch['x_pu']
    b = branch['b_pu']
    
    i = bus_index[from_bus]
    j = bus_index[to_bus]
    
    # Compute branch admittance: y = 1 / (r + jx)
    z = complex(r, x)
    y = 1 / z
    
    # Add to diagonal entries: y + j*b/2
    ybus[i][i] += y + 1j * (b / 2)
    ybus[j][j] += y + 1j * (b / 2)
    
    # Add to off-diagonal entries: -y
    ybus[i][j] += -y
    ybus[j][i] += -y

# Process shunts
for shunt in shunts:
    bus = shunt['bus']
    g = shunt['g_pu']
    b = shunt['b_pu']
    
    i = bus_index[bus]
    ybus[i][i] += complex(g, b)

# Convert to submission format
submission = {
    "buses": buses,
    "ybus": []
}

for i in range(n_buses):
    row = []
    for j in range(n_buses):
        entry = {
            "re": ybus[i][j].real,
            "im": ybus[i][j].imag
        }
        row.append(entry)
    submission["ybus"].append(row)

# Write submission.json
with open('submission.json', 'w') as f:
    json.dump(submission, f, indent=2)

print("Submission written successfully")
print(json.dumps(submission, indent=2))
