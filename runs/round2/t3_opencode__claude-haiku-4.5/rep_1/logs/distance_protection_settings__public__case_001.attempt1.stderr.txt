import json
import math

# Load task data
with open('task.json', 'r') as f:
    task = json.load(f)

settings = protection['setting_rules']

# Calculate protected line impedance magnitude
protected_line = protection['protected_line']
R_line = protected_line['r1_ohm_per_km'] * protected_line['length_km']
X_line = protected_line['x1_ohm_per_km'] * protected_line['length_km']
Z_line = math.sqrt(R_line**2 + X_line**2)

print(f"Protected line (L_AB):")
print(f"  Length: {protected_line['length_km']} km")
print(f"  R: {R_line} Ω")
print(f"  X: {X_line} Ω")
print(f"  Z magnitude: {Z_line} Ω")
print()

# Find shortest and longest backup lines
backup_lines = protection['remote_backup_lines']
print(f"Backup lines:")
for line in backup_lines:
    R = line['r1_ohm_per_km'] * line['length_km']
    X = line['x1_ohm_per_km'] * line['length_km']
    Z = math.sqrt(R**2 + X**2)
    print(f"  {line['id']}: {line['length_km']} km, Z={Z:.2f} Ω")
print()

# Calculate backup line impedances
backup_impedances = []
for line in backup_lines:
    R = line['r1_ohm_per_km'] * line['length_km']
    X = line['x1_ohm_per_km'] * line['length_km']
    Z = math.sqrt(R**2 + X**2)
    backup_impedances.append((line['id'], line['length_km'], Z))

# Sort by length to find shortest and longest
backup_impedances_sorted = sorted(backup_impedances, key=lambda x: x[1])
shortest_backup = backup_impedances_sorted[0]
longest_backup = backup_impedances_sorted[-1]

print(f"Shortest backup: {shortest_backup[0]} ({shortest_backup[1]} km), Z={shortest_backup[2]:.2f} Ω")
print(f"Longest backup: {longest_backup[0]} ({longest_backup[1]} km), Z={longest_backup[2]:.2f} Ω")
print()

# Zone 1: underreach within 78-85% of protected line
zone1_min_fraction = settings['zone1_fraction'][0]
zone1_max_fraction = settings['zone1_fraction'][1]
zone1_reach = Z_line * (zone1_min_fraction + zone1_max_fraction) / 2  # Middle of range
print(f"Zone 1 reach (target {zone1_min_fraction*100}-{zone1_max_fraction*100}%): {zone1_reach:.2f} Ω")

# Zone 2: line + fraction of shortest backup
zone2_min_fraction = settings['zone2_backup_fraction'][0]

zone3_max_fraction = settings['zone3_backup_fraction'][1]

apparent_power = protection['maximum_load']['apparent_power_mva']
pf = protection['maximum_load']['power_factor_lagging']
P_load = apparent_power * pf
Q_load = apparent_power * math.sqrt(1 - pf**2)
V_kv = protection['voltage_kv']
V_V = V_kv * 1000

# Load impedance: Z_load = V^2 / (P + jQ)

print(f"  Max Z3 (80% of load impedance): {max_zone3_load:.2f} Ω")

# Zone 3 reach: middle of range but capped by load encroachment
zone3_nominal = Z_line + longest_backup[2] * (zone3_min_fraction + zone3_max_fraction) / 2
zone3_reach = min(zone3_nominal, max_zone3_load)
print(f"  Zone 3 nominal: {zone3_nominal:.2f} Ω")
print(f"  Zone 3 actual (capped by load): {zone3_reach:.2f} Ω")

}
