import json
import numpy as np

# Load task data
with open('task.json', 'r') as f:
    task = json.load(f)

# Extract inputs
case = task['pv_lcoe_case']
system_size_kw = case['system_size_kw_dc']
hard_cost_usd_per_kw = case['hard_cost_usd_per_kw']
soft_cost_components = case['soft_cost_components_usd_per_kw']
capital_recovery_factor = case['capital_recovery_factor']
fixed_om_usd_per_kw_year = case['fixed_om_usd_per_kw_year']
annual_energy_mwh_year1 = case['annual_energy_mwh_year1']
soft_cost_scenarios = case['soft_cost_scenarios']

# Calculate LCOE for each scenario
scenario_lcoe = {}
lcoe_values = []

for scenario_name, multipliers in soft_cost_scenarios.items():
    # Calculate soft cost for this scenario
    soft_cost_total_usd_per_kw = 0
    for component, base_cost in soft_cost_components.items():
        soft_cost_total_usd_per_kw += base_cost * multipliers[component]
    
    # Calculate total CAPEX
    total_cost_per_kw = hard_cost_usd_per_kw + soft_cost_total_usd_per_kw
    capex_usd = system_size_kw * total_cost_per_kw
    
    # Calculate LCOE
    numerator = capex_usd * capital_recovery_factor + fixed_om_usd_per_kw_year * system_size_kw
    lcoe = numerator / annual_energy_mwh_year1
    
    scenario_lcoe[scenario_name] = round(lcoe, 2)
    lcoe_values.append(lcoe)

# Sort LCOE values for quantile calculation
lcoe_values_sorted = sorted(lcoe_values)

# Calculate empirical quantiles with linear interpolation
def percentile_linear(data, p):
    """Calculate percentile using linear interpolation (NIST method)"""
    n = len(data)
    h = (n - 1) * p / 100.0
    h_floor = int(np.floor(h))
    h_frac = h - h_floor
    
    if h_floor >= n - 1:
        return data[n - 1]
    if h_floor < 0:
        return data[0]
    
    return data[h_floor] + h_frac * (data[h_floor + 1] - data[h_floor])

p10 = percentile_linear(lcoe_values_sorted, 10)
p50 = percentile_linear(lcoe_values_sorted, 50)
p90 = percentile_linear(lcoe_values_sorted, 90)

# Calculate mean
mean_lcoe = np.mean(lcoe_values)

# Build submission
submission = {
    "scenario_lcoe_usd_per_mwh": scenario_lcoe,
    "lcoe_quantiles_usd_per_mwh": {
        "p10": round(p10, 2),
        "p50": round(p50, 2),
        "p90": round(p90, 2)
    },
    "expected_lcoe_usd_per_mwh": round(mean_lcoe, 2)
}

print("Scenario LCOE values:")
for scenario, lcoe in scenario_lcoe.items():
    print(f"  {scenario}: ${lcoe:.2f}/MWh")

print(f"\nSorted LCOE values: {[f'{x:.2f}' for x in lcoe_values_sorted]}")
print(f"\nQuantiles:")
print(f"  p10: ${p10:.2f}/MWh")
print(f"  p50: ${p50:.2f}/MWh")
print(f"  p90: ${p90:.2f}/MWh")
print(f"\nMean: ${mean_lcoe:.2f}/MWh")

print("\nFinal submission:")
print(json.dumps(submission, indent=2))

# Write submission.json
with open('submission.json', 'w') as f:
    json.dump(submission, f)

print("\n✓ submission.json written")
