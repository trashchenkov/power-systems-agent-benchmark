import math
import json

# Input data
scheduled_current_a = 520.0
T_max = 75.0  # max_conductor_temperature_c

# Conductor properties
diameter_mm = 22.4
resistance_ohm_per_km_at_20c = 0.118
resistance_temp_coeff_per_c = 0.00403
emissivity = 0.7
solar_absorptivity = 0.7

# Model constants
convective_w_per_m_c_at_1ms = 0.72
stefan_boltzmann = 5.670374419e-8

# Weather series
weather_series = [
    {
        "timestamp": "t00",
        "ambient_temperature_c": 18.0,
        "wind_speed_m_s": 2.8,
        "solar_w_per_m2": 150.0
    },
    {
        "timestamp": "t01",
        "ambient_temperature_c": 31.0,
        "wind_speed_m_s": 0.7,
        "solar_w_per_m2": 850.0
    },
    {
        "timestamp": "t02",
        "ambient_temperature_c": 27.0,
        "wind_speed_m_s": 1.4,
        "solar_w_per_m2": 600.0
    },
    {
        "timestamp": "t03",
        "ambient_temperature_c": 22.0,
        "wind_speed_m_s": 3.2,
        "solar_w_per_m2": 100.0
    }
]

# Calculate ampacity for each timestamp
ampacity_by_timestamp = {}
ampacities = []

for weather in weather_series:
    timestamp = weather["timestamp"]
    ambient = weather["ambient_temperature_c"]
    wind_speed = weather["wind_speed_m_s"]
    solar = weather["solar_w_per_m2"]
    
    # Calculate R_hot (resistance at max temperature)
    R_hot = (resistance_ohm_per_km_at_20c / 1000) * (1 + resistance_temp_coeff_per_c * (T_max - 20))
    
    # Calculate convection
    convection = convective_w_per_m_c_at_1ms * max(wind_speed, 0.05) ** 0.6 * (T_max - ambient)
    
    # Calculate radiation (note: radiating surface is pi*D per metre)
    radiation = emissivity * stefan_boltzmann * (math.pi * diameter_mm / 1000) * \
                ((T_max + 273.15) ** 4 - (ambient + 273.15) ** 4)
    
    # Calculate solar gain (projected width D per metre)
    solar_gain = solar_absorptivity * solar * (diameter_mm / 1000)
    
    # Calculate ampacity
    numerator = max(convection + radiation - solar_gain, 0)
    ampacity = math.sqrt(numerator / R_hot)
    
    ampacity_by_timestamp[timestamp] = ampacity
    ampacities.append((timestamp, ampacity))
    
    print(f"{timestamp}:")
    print(f"  R_hot: {R_hot}")
    print(f"  convection: {convection}")
    print(f"  radiation: {radiation}")
    print(f"  solar_gain: {solar_gain}")
    print(f"  numerator: {numerator}")
    print(f"  ampacity: {ampacity}")

# Find minimum ampacity
minimum_ampacity = min(ampacities, key=lambda x: x[1])
limiting_timestamp = minimum_ampacity[0]
minimum_ampacity_value = minimum_ampacity[1]

# Find overload timestamps
overload_timestamps = []
for timestamp, ampacity in ampacities:
    if scheduled_current_a > ampacity:
        overload_timestamps.append(timestamp)

print(f"\nMinimum ampacity: {minimum_ampacity_value} A at {limiting_timestamp}")
print(f"Scheduled current: {scheduled_current_a} A")
print(f"Overload timestamps: {overload_timestamps}")

# Create submission
submission = {
    "ampacity_by_timestamp_a": ampacity_by_timestamp,
    "minimum_ampacity_a": minimum_ampacity_value,
    "limiting_timestamp": limiting_timestamp,
    "overload_timestamps": overload_timestamps
}

print("\n" + json.dumps(submission, indent=2))
