import json
import cmath

# Load task data
with open('task.json', 'r') as f:
    task = json.load(f)

network = task['network']
buses = network['buses']
branches = network['branches']
shunts = network['shunts']

n_buses = len(buses)
bus_idx = {bus: i for i, bus in enumerate(buses)}

# Initialize Ybus as complex matrix
Ybus = [[0+0j for _ in range(n_buses)] for _ in range(n_buses)]

# Process branches
for branch in branches:
    from_bus = branch['from_bus']
    to_bus = branch['to_bus']
    r = branch['r_pu']
    x = branch['x_pu']
    b = branch['b_pu']
    
    # Calculate branch admittance y = 1 / (r + jx)
    z = complex(r, x)
    y = 1 / z
    
    i = bus_idx[from_bus]
    j = bus_idx[to_bus]
    
    # Add y + j*b/2 to diagonal entries
    Ybus[i][i] += y + 1j * b / 2
    Ybus[j][j] += y + 1j * b / 2
    
    # Add -y to off-diagonal entries
    Ybus[i][j] += -y
    Ybus[j][i] += -y

# Process shunts
for shunt in shunts:
    bus = shunt['bus']
    g = shunt['g_pu']
    b = shunt['b_pu']
    
    i = bus_idx[bus]
    Ybus[i][i] += complex(g, b)

# Convert to output format
ybus_output = []
for i in range(n_buses):
    row = []
    for j in range(n_buses):
        entry = {
            "re": round(Ybus[i][j].real, 10),
            "im": round(Ybus[i][j].imag, 10)
        }
        row.append(entry)
    ybus_output.append(row)

# Create submission
submission = {
    "buses": buses,
    "ybus": ybus_output,
    "rationale": "Computed Ybus from branches and shunts following nodal admittance matrix construction."
}

# Save submission
with open('submission.json', 'w') as f:
    json.dump(submission, f, indent=2)

# Display for verification
print("Ybus matrix (complex values):")
for i in range(n_buses):
    for j in range(n_buses):
        print(f"Y[{i}][{j}] = {Ybus[i][j]:.6f}")
print("\nSubmission saved to submission.json")
