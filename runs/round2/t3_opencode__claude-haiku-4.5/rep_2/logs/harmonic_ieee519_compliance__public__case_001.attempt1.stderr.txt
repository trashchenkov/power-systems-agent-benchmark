import json
import math

# Read task.json
with open('task.json', 'r') as f:
    task = json.load(f)

harmonic_case = task['public_inputs']['harmonic_case']

# Extract data
max_demand_current_a = harmonic_case['pcc']['max_demand_current_a']
short_circuit_current_ka = harmonic_case['pcc']['short_circuit_current_ka']
short_circuit_current_a = short_circuit_current_ka * 1000

fundamental_current_a = harmonic_case['measured']['fundamental_current_a']
harmonic_currents_a = harmonic_case['measured']['harmonic_currents_a']
harmonic_voltages_pct = harmonic_case['measured']['harmonic_voltages_pct']

# IL is the max demand current
IL = max_demand_current_a

# Calculate Isc/IL ratio
isc_over_il = short_circuit_current_a / IL
print(f"Isc/IL ratio: {isc_over_il}")

# Select the appropriate limit row - first row where isc_il_below > isc_over_il
limit_table = harmonic_case['current_limit_table']
selected_row = None
for row in limit_table:
    if row['isc_il_below'] > isc_over_il:
        selected_row = row
        print(f"Selected limit row with isc_il_below: {row['isc_il_below']}")
        break

if selected_row is None:
    print("ERROR: No suitable limit row found!")
else:
    # Calculate TDD = sqrt(sum of squared harmonic currents)/IL*100
    sum_squared = sum(h**2 for h in harmonic_currents_a.values())
    tdd_pct = (math.sqrt(sum_squared) / IL) * 100
    print(f"TDD: {tdd_pct:.4f}%")
    
    # Check for current limit violations
    current_violations = []
    
    # Check TDD
    if tdd_pct > selected_row['tdd_pct']:
        current_violations.append('tdd')
        print(f"TDD violation: {tdd_pct:.4f}% > {selected_row['tdd_pct']}%")
    
    # Check individual current harmonics
    for harmonic_order_str, harmonic_current in harmonic_currents_a.items():
        harmonic_order = int(harmonic_order_str)
        harmonic_pct = (harmonic_current / IL) * 100
        
        # Determine which limit applies based on harmonic order
        if harmonic_order < 11:
            limit = selected_row['h_below_11_pct']
        elif harmonic_order <= 17:
            limit = selected_row['h_11_to_17_pct']
        elif harmonic_order <= 23:
            limit = selected_row['h_17_to_23_pct']
        elif harmonic_order <= 35:
            limit = selected_row['h_23_to_35_pct']
        else:
            limit = selected_row['h_35_up_pct']
        
        if harmonic_pct > limit:
            current_violations.append(harmonic_order_str)
            print(f"Current harmonic {harmonic_order}: {harmonic_pct:.4f}% > {limit}%")
    
    # Calculate Voltage THD = root-sum-square of harmonic_voltages_pct
    sum_squared_v = sum(h**2 for h in harmonic_voltages_pct.values())
    voltage_thd_pct = math.sqrt(sum_squared_v)
    print(f"Voltage THD: {voltage_thd_pct:.4f}%")
    
    # Check for voltage limit violations
    voltage_violations = []
    voltage_limits = harmonic_case['voltage_limits']
    
    # Check voltage THD
    if voltage_thd_pct > voltage_limits['thd_pct']:
        voltage_violations.append('thd')
        print(f"Voltage THD violation: {voltage_thd_pct:.4f}% > {voltage_limits['thd_pct']}%")
    
    # Check individual voltage harmonics
    for harmonic_order_str, harmonic_voltage_pct in harmonic_voltages_pct.items():
        if harmonic_voltage_pct > voltage_limits['individual_pct']:
            voltage_violations.append(harmonic_order_str)
            print(f"Voltage harmonic {harmonic_order_str}: {harmonic_voltage_pct}% > {voltage_limits['individual_pct']}%")
    
    # Determine compliance
    compliant = (len(current_violations) == 0) and (len(voltage_violations) == 0)
    print(f"Compliant: {compliant}")
    
    # Build submission
    submission = {
        "isc_over_il": isc_over_il,
        "tdd_pct": round(tdd_pct, 2),
        "current_violations": current_violations,
        "voltage_thd_pct": round(voltage_thd_pct, 2),
        "voltage_violations": voltage_violations,
        "compliant": compliant
    }
    
    print("\n=== SUBMISSION ===")
    print(json.dumps(submission, indent=2))
    
    # Write submission.json
    with open('submission.json', 'w') as f:
        json.dump(submission, f, indent=2)
    print("\nSubmission written to submission.json")
