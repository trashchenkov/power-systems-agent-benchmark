import json
import math

# Public inputs
angle_limit_deg = 180.0
voltage_floor_pu = 0.7
min_damping_pct = 3.0

cases = [
    {
        "case_id": "fault_a",
        "fault_clearing_time_s": 0.1,
        "estimated_cct_s": 0.17,
        "max_rotor_angle_deg": 142.0,
        "min_voltage_pu": 0.76,
        "damping_ratio_pct": 5.0
    },
    {
        "case_id": "fault_b",
        "fault_clearing_time_s": 0.16,
        "estimated_cct_s": 0.14,
        "max_rotor_angle_deg": 188.0,
        "min_voltage_pu": 0.68,
        "damping_ratio_pct": 2.2
    },
    {
        "case_id": "fault_c",
        "fault_clearing_time_s": 0.08,
        "estimated_cct_s": 0.11,
        "max_rotor_angle_deg": 171.0,
        "min_voltage_pu": 0.72,
        "damping_ratio_pct": 3.4
    }
]

case_results = {}
stable_count = 0
highest_risk_case = None
highest_risk_score = -1

for case in cases:
    case_id = case["case_id"]
    
    # Calculate margins
    angle_margin = angle_limit_deg - case["max_rotor_angle_deg"]
    voltage_margin = case["min_voltage_pu"] - voltage_floor_pu
    clearing_margin_s = case["estimated_cct_s"] - case["fault_clearing_time_s"]
    damping_margin = case["damping_ratio_pct"] - min_damping_pct
    
    # Determine stability
    is_stable = (angle_margin >= 0 and voltage_margin >= 0 and 
                 clearing_margin_s >= 0 and damping_margin >= 0)
    
    # Calculate risk score
    risk_score = (max(0, -angle_margin/60) + 
                  max(0, -voltage_margin/0.1) + 
                  max(0, -clearing_margin_s/0.1) + 
                  max(0, -damping_margin/5) + 
                  max(0, 1 - clearing_margin_s/0.2)*0.1)
    
    risk_score = round(risk_score, 3)
    
    label = "stable" if is_stable else "unstable"
    
    case_results[case_id] = {
        "label": label,
        "risk_score": risk_score,
        "clearing_margin_s": clearing_margin_s
    }
    
    if is_stable:
        stable_count += 1
    
    # Track highest risk case
    if risk_score > highest_risk_score or (risk_score == highest_risk_score and (highest_risk_case is None or case_id > highest_risk_case)):
        highest_risk_score = risk_score
        highest_risk_case = case_id

print("Case Results:")
for case_id, result in case_results.items():
    print(f"{case_id}: {result}")

print(f"\nStable case count: {stable_count}")
print(f"Highest risk case: {highest_risk_case}")

# Create submission
submission = {
    "case_results": case_results,
    "stable_case_count": stable_count,
    "highest_risk_case": highest_risk_case
}

print("\nSubmission JSON:")
print(json.dumps(submission, indent=2))
