# Create bus index
bus_ids = [b["id"] for b in task_data["buses"]]

n_buses = len(bus_ids)

# DC power flow solver
def solve_dc_pf(branches_active, buses):
    """Solve DC power flow for given active branches"""
    # Build admittance matrix (B matrix)
    B = np.zeros((n_buses, n_buses))
    
    for branch in branches_active:
        i = bus_idx[branch["from_bus"]]
        j = bus_idx[branch["to_bus"]]
        b_ij = 1.0 / branch["x_pu"]
        B[i, i] += b_ij
        B[j, j] += b_ij
        B[i, j] -= b_ij
        B[j, i] -= b_ij
    
    # Power injections (slack bus is first)
    p_inj = np.array([b["p_mw"] for b in buses])
    
    # Remove slack bus row/column
    B_reduced = B[1:, 1:]
    p_reduced = p_inj[1:]
    
    # Solve for voltage angles (slack is at 0)
    try:
        theta_reduced = solve(B_reduced, -p_reduced)
        theta = np.concatenate([[0.0], theta_reduced])
    except np.linalg.LinAlgError:
        return None  # Singular matrix, network not connected
    
    # Calculate flows
    flows = {}
    for branch in branches_active:
        i = bus_idx[branch["from_bus"]]
        j = bus_idx[branch["to_bus"]]
        flow = (theta[i] - theta[j]) / branch["x_pu"]
        flows[branch["id"]] = flow
    
    return theta, flows

def test_scenario(close_branches, open_branches):
    """Test a switching scenario"""
    # Create branch state
    branches_state = {}
    for b in task_data["branches"]:
        branches_state[b["id"]] = b["closed"]
    
    # Apply switches
    for bid in close_branches:
        branches_state[bid] = True
    for bid in open_branches:
        branches_state[bid] = False
    
    # Get active branches
    active_branches = [b for b in task_data["branches"] if branches_state[b["id"]]]
    
    # Solve power flow
    result = solve_dc_pf(active_branches, task_data["buses"])
    if result is None:

    theta, flows = result
    
    # Check loading constraints
    max_loading = 0.0

    for branch in active_branches:
        flow_mw = abs(flows[branch["id"]])
        loading_pct = (flow_mw / branch["rating_mw"]) * 100
        if loading_pct > 100:
            violations.append(f"{branch['id']}: {loading_pct:.2f}%")
        max_loading = max(max_loading, loading_pct)
    
    connected = True  # If solve didn't fail, network is connected
    
    return flows, max_loading, len(violations) == 0 and connected, violations
