import json
import math

# Read task.json
with open('task.json', 'r') as f:
    task = json.load(f)

case = task['v2g_voltage_support_case']
target_voltage = case['target_voltage_pu']
base_voltages = case['base_voltage_pu']
q_sensitivities = case['q_sensitivity_pu_per_kvar']
evs = case['evs']

# Group EVs by bus and calculate aggregate reactive capability
bus_evs = {}
for evse_name, evse_data in evs.items():
    bus = evse_data['bus']
    if bus not in bus_evs:
        bus_evs[bus] = []
    bus_evs[bus].append(evse_data)

# Calculate results for each bus
q_dispatch = {}
corrected_voltage = {}

for bus in base_voltages.keys():
    base_v = base_voltages[bus]
    target_v = target_voltage
    sensitivity = q_sensitivities[bus]
    
    # Calculate voltage deficit
    voltage_deficit = target_v - base_v
    
    # Calculate reactive power needed (from linear sensitivity: ΔV = sensitivity * Q)
    # So: Q = ΔV / sensitivity
    if sensitivity != 0:
        q_needed = voltage_deficit / sensitivity
    else:
        q_needed = 0
    
    # Calculate aggregate capability for EVs at this bus
    aggregate_q_capability = 0
    if bus in bus_evs:
        for evse in bus_evs[bus]:
            s_rating = evse['s_rating_kva']
            p_active = evse['active_power_kw']
            
            # Calculate available reactive capability (from S² = P² + Q²)
            # Q_max = sqrt(S² - P²)
            q_max = math.sqrt(max(0, s_rating**2 - p_active**2))
            aggregate_q_capability += q_max
    
    # Clip the reactive command by aggregate capability
    # We can inject (+Q) or absorb (-Q) reactive power up to the limit
    q_clipped = max(-aggregate_q_capability, min(aggregate_q_capability, q_needed))
    
    q_dispatch[bus] = q_clipped
    
    # Calculate corrected voltage
    corrected_v = base_v + sensitivity * q_clipped
    corrected_voltage[bus] = corrected_v

# Calculate total absolute reactive effort
total_abs_q = sum(abs(q) for q in q_dispatch.values())

# Prepare submission
submission = {
    "q_dispatch_kvar": q_dispatch,
    "corrected_voltage_pu": corrected_voltage,
    "total_abs_q_kvar": total_abs_q
}

# Write submission.json
with open('submission.json', 'w') as f:
    json.dump(submission, f, indent=2)

print(json.dumps(submission, indent=2))
print("\nSubmission written to submission.json")
