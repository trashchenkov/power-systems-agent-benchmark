import json
import math

# Load task data
with open('task.json', 'r') as f:
    task = json.load(f)

case = task['bess_response_case']

# Extract parameters
nominal_freq = case['nominal_frequency_hz']
target_voltage = case['target_voltage_pu']
interval_sec = case['interval_seconds']
initial_soc = case['initial_soc_mwh']
soc_min = case['soc_min_mwh']
soc_max = case['soc_max_mwh']
p_discharge_limit = case['p_discharge_limit_mw']
p_charge_limit = case['p_charge_limit_mw']
s_rating = case['s_rating_mva']
freq_droop = case['frequency_droop_mw_per_hz']
volt_droop = case['voltage_droop_mvar_per_pu']
charge_eff = case['charge_efficiency']
discharge_eff = case['discharge_efficiency']

measurements = case['measurements']

# Initialize results
p_response = {}
q_response = {}
current_soc = initial_soc

# Process each measurement
for timestamp in sorted(measurements.keys()):
    meas = measurements[timestamp]
    freq = meas['frequency_hz']
    voltage = meas['voltage_pu']
    
    # Calculate P response from frequency droop
    # Positive P means discharge (frequency is below nominal)
    p_unclamped = (nominal_freq - freq) * freq_droop
    
    # Clip P by charge/discharge limits
    # Positive P (discharge) is limited by p_discharge_limit
    # Negative P (charge) is limited by -p_charge_limit
    p_response_mw = max(-p_charge_limit, min(p_discharge_limit, p_unclamped))
    p_response[timestamp] = p_response_mw
    
    # Calculate Q response from voltage droop
    # Positive Q means reactive discharge (voltage is above target)
    q_unclamped = (target_voltage - voltage) * volt_droop
    
    # Apparent power capability: S^2 = P^2 + Q^2
    # So Q is limited by sqrt(S^2 - P^2)
    max_q = math.sqrt(s_rating**2 - p_response_mw**2)
    q_response_mvar = max(-max_q, min(max_q, q_unclamped))
    q_response[timestamp] = q_response_mvar
    
    # Update SOC over the interval
    # Positive P discharges the battery (decreases SOC)
    # Energy change in MWh = P * interval_seconds / 3600 (convert seconds to hours)
    energy_mwh = p_response_mw * interval_sec / 3600.0
    
    # Apply efficiency
    if energy_mwh > 0:  # Discharging
        soc_change = -energy_mwh / discharge_eff
    else:  # Charging
        soc_change = -energy_mwh * charge_eff
    
    current_soc = current_soc + soc_change
    
    # Clamp SOC to limits (though the problem says update over interval)
    current_soc = max(soc_min, min(soc_max, current_soc))
    
    print(f"{timestamp}: P={p_response_mw:.4f} MW, Q={q_response_mvar:.4f} Mvar, SOC={current_soc:.4f} MWh")

final_soc = current_soc

# Create submission
submission = {
    "p_response_mw": p_response,
    "q_response_mvar": q_response,
    "final_soc_mwh": final_soc,
    "rationale": "Applied frequency droop to calculate active power response, voltage droop for reactive power, clipped by inverter capability constraints and updated SOC with charge/discharge efficiencies."
}

print("\nSubmission:")
print(json.dumps(submission, indent=2))

# Write submission
with open('submission.json', 'w') as f:
    json.dump(submission, f, indent=2)

print("\nSubmission written to submission.json")
