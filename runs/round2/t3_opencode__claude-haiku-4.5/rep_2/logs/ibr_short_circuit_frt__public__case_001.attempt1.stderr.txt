import json
import math

# Load task data
with open('task.json', 'r') as f:
    task = json.load(f)

inputs = task['public_inputs']['ibr_frt_case']
samples = inputs['samples']

# Parameters
current_limit = inputs['current_limit_pu']
prefault_active = inputs['prefault_active_current_pu']
reactive_deadband = inputs['reactive_deadband_pu']
k_reactive = inputs['k_reactive_current_pu_per_pu']
frt_voltage_floor = inputs['frt_voltage_floor_pu']
max_low_voltage_duration = inputs['max_low_voltage_duration_s']

# Calculate current profile for each sample
current_profile = {}
min_reactive = float('inf')
max_total = 0
frt_compliant = True

# Check FRT voltage compliance
low_voltage_duration = 0
for sample in samples:
    voltage = sample['voltage_pu']
    time = sample['time_s']
    
    if voltage < frt_voltage_floor:
        # This is a low voltage event - need to track duration
        # For now, we'll check if any sample is below floor outside acceptable duration
        pass

# Calculate currents for each sample
sample_times = [s['time_s'] for s in samples]
for i, sample in enumerate(samples):
    sample_id = sample['sample_id']
    voltage = sample['voltage_pu']
    time = sample['time_s']
    
    # Calculate reactive current requirement
    if voltage < reactive_deadband:
        # Reactive current needed: k * (deadband - voltage)
        iq_required = k_reactive * (reactive_deadband - voltage)
        iq = min(iq_required, current_limit)
    else:
        iq = 0.0
    
    # Calculate active current (limited by remaining capability)
    # S_apparent = I_total (in normalized units)
    # I_total^2 = id^2 + iq^2, so id^2 = I_total^2 - iq^2
    # Active current is also limited by prefault value and remaining capability
    remaining_current_capability = math.sqrt(current_limit**2 - iq**2)
    id_pu = min(prefault_active, remaining_current_capability)
    
    # Total current magnitude
    total_current = math.sqrt(id_pu**2 + iq**2)
    
    current_profile[sample_id] = {
        'id_pu': round(id_pu, 6),
        'iq_pu': round(iq, 6),
        'total_current_pu': round(total_current, 6)
    }
    
    min_reactive = min(min_reactive, iq)
    max_total = max(max_total, total_current)
    
    # Check FRT compliance
    if voltage < frt_voltage_floor:
        frt_compliant = False

# Additional FRT check: voltage must not exceed duration below floor
low_voltage_times = [s['time_s'] for s in samples if s['voltage_pu'] < frt_voltage_floor]
if low_voltage_times:
    # Check if low voltage duration fits within max allowed
    duration = max(low_voltage_times) - min(low_voltage_times)
    if duration > max_low_voltage_duration:
        frt_compliant = False

# Prepare submission
submission = {
    'current_profile_pu': current_profile,
    'min_reactive_current_pu': round(min_reactive, 6),
    'max_total_current_pu': round(max_total, 6),
    'frt_compliant': frt_compliant
}

print(json.dumps(submission, indent=2))

# Write submission
with open('submission.json', 'w') as f:
    json.dump(submission, f, indent=2)

print("\nSubmission written to submission.json")
