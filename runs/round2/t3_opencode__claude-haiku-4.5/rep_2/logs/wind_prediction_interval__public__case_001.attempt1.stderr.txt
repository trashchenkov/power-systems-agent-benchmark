import json
import math

# Load task data
with open('task.json', 'r') as f:
    task = json.load(f)

# Extract parameters
case = task['wind_interval_case']
ref_density = case['reference_air_density_kg_m3']
turbine_count = case['turbine']['count']
rated_mw = case['turbine']['rated_mw_each']
cut_in = case['power_curve']['cut_in_m_s']
rated_speed = case['power_curve']['rated_m_s']
cut_out = case['power_curve']['cut_out_m_s']
horizons = case['horizons']

def calculate_power(wind_speed_m_s, air_density_kg_m3):
    """Calculate power using the specified power curve convention."""
    # Correct the wind speed
    v_c = wind_speed_m_s * ((air_density_kg_m3 / ref_density) ** (1/3))
    
    # Apply power curve
    if v_c < cut_in or v_c >= cut_out:
        power = 0
    elif v_c >= rated_speed:
        power = turbine_count * rated_mw
    else:
        # Linear interpolation between cut-in and rated
        normalized = (v_c - cut_in) / (rated_speed - cut_in)
        power = turbine_count * rated_mw * (normalized ** 3)
    
    return power

def calculate_quantile(values, q):
    """Calculate quantile using specified convention."""
    sorted_vals = sorted(values)
    n = len(sorted_vals)
    pos = q * (n - 1)
    
    lower_idx = math.floor(pos)
    upper_idx = math.ceil(pos)
    
    if lower_idx == upper_idx:
        return sorted_vals[lower_idx]
    
    # Linear interpolation
    frac = pos - lower_idx
    return sorted_vals[lower_idx] * (1 - frac) + sorted_vals[upper_idx] * frac

# Process each horizon
prediction_intervals = {}
interval_widths = []

for horizon in horizons:
    timestamp = horizon['timestamp']
    scenarios = horizon['scenarios']
    
    # Calculate power for each scenario
    powers = []
    for scenario in scenarios:
        wind_speed = scenario['wind_speed_m_s']
        air_density = scenario['air_density_kg_m3']
        power = calculate_power(wind_speed, air_density)
        powers.append(power)
    
    # Calculate quantiles
    p10 = calculate_quantile(powers, 0.10)
    p50 = calculate_quantile(powers, 0.50)
    p90 = calculate_quantile(powers, 0.90)
    
    prediction_intervals[timestamp] = {
        "p10": round(p10, 2),
        "p50": round(p50, 2),
        "p90": round(p90, 2)
    }
    
    interval_widths.append(p90 - p10)

# Calculate average interval width
avg_width = sum(interval_widths) / len(interval_widths)

# Create submission
submission = {
    "prediction_interval_mw_by_timestamp": prediction_intervals,
    "average_interval_width_mw": round(avg_width, 2)
}

print(json.dumps(submission, indent=2))

# Verify constraints
print("\n--- Verification ---")
for ts, vals in prediction_intervals.items():
    print(f"{ts}: p10={vals['p10']}, p50={vals['p50']}, p90={vals['p90']}")
    if vals['p10'] <= vals['p50'] <= vals['p90']:
        print(f"  ✓ Ordering satisfied")
    else:
        print(f"  ✗ Ordering violated")
