import json

# Load task data
with open('task.json', 'r') as f:
    task = json.load(f)

case = task['dispatch_uncertainty_case']
generators = case['generators']
scenarios = case['scenarios']
time_periods = case['time_periods']
reserve_req = case['reserve_requirement_mw']

# Calculate maximum net load for each time period across all scenarios
max_net_load = {}
for t in time_periods:
    max_net_load[t] = 0
    for scenario in scenarios:
        load = scenario['load_mw'][t]
        renewable = scenario['renewable_mw'][t]
        net_load = load - renewable
        max_net_load[t] = max(max_net_load[t], net_load)

print("Maximum net load per time period:")
for t, val in max_net_load.items():
    print(f"  {t}: {val} MW")

# Required thermal generation: max_net_load + reserve
required_thermal = {}
for t in time_periods:
    required_thermal[t] = max_net_load[t] + reserve_req
    print(f"Required thermal generation at {t}: {required_thermal[t]} MW")

# Optimization: minimize cost while meeting requirements
# We need to dispatch generators to meet required_thermal[t] for each t
# This is a simple least-cost dispatch problem per time period

dispatch_mw = {gen['id']: {} for gen in generators}
total_cost = 0

for t in time_periods:
    required = required_thermal[t]
    
    # Sort generators by cost (ascending)
    sorted_gens = sorted(generators, key=lambda g: g['variable_cost_per_mwh'])
    
    remaining_needed = required
    dispatch = {}
    
    for gen in sorted_gens:
        gen_id = gen['id']
        pmin = gen['pmin_mw']
        pmax = gen['pmax_mw']
        
        if remaining_needed <= 0:
            dispatch[gen_id] = 0
        else:
            # Dispatch as much as needed from this generator
            dispatch_amount = min(pmax, max(pmin, remaining_needed))
            dispatch[gen_id] = dispatch_amount
            remaining_needed -= dispatch_amount
    
    # Update dispatch for this time period
    for gen_id, amount in dispatch.items():
        dispatch_mw[gen_id][t] = amount
    
    # Add cost for this time period
    total_generation = sum(dispatch.values())
    for gen in sorted_gens:
        gen_id = gen['id']
        amount = dispatch[gen_id]
        cost = amount * gen['variable_cost_per_mwh']
        total_cost += cost
        print(f"{t}: {gen_id} dispatches {amount} MW at ${gen['variable_cost_per_mwh']}/MWh = ${cost}")

print(f"\nTotal cost: ${total_cost}")
