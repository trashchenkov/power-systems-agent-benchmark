source_node = case['source_node']  # n0
sections = case['sections']
fci_locations = case['fci_locations']  # {fci_id: node}
observed_fci_state = case['observed_fci_state']  # {fci_id: bool}
outage_complaints = case['outage_complaint_nodes']  # [n5]

# Build network graph from sections

children = {}
parents = {}
for sec_id, sec_info in sections.items():
    from_node = sec_info['from']
    to_node = sec_info['to']
    if from_node not in children:
        children[from_node] = []
    children[from_node].append(to_node)
    parents[to_node] = from_node

print("Children:", children)

# Function to get all downstream nodes from a node
def get_downstream(node, children):
    downstream = set()
    stack = [node]
    while stack:
        curr = stack.pop()
        if curr in children:
            for child in children[curr]:
                downstream.add(child)
                stack.append(child)
    return downstream

# Function to get all upstream nodes from a node (towards source)
def get_upstream(node, parents):
    upstream = set()
    curr = node
    while curr in parents:
        curr = parents[curr]
        upstream.add(curr)
    return upstream

# For each candidate faulted section, compute mismatch

results = []

    # - All downstream customer nodes from to_node should be outaged
    
    # Downstream nodes from the faulted section's "to" node

    upstream_and_from = get_upstream(from_node, parents) | {from_node}

        # More precisely: FCI is upstream if its node is <= from_node in the path
        if fci_node in upstream_and_from or fci_node == from_node:

            complaint_mismatch += 1

    for customer in affected_customers:
        if customer not in outage_complaints:
            complaint_mismatch += 1
    
    total_mismatch = fci_mismatch + complaint_mismatch

        'expected_fci_active': sorted(list(expected_fci_active)),
        'affected_customers': sorted(list(affected_customers)),
        'fci_mismatch': fci_mismatch,
        'complaint_mismatch': complaint_mismatch,
        'total_mismatch': total_mismatch

    print(f"  Expected FCIs active: {result['expected_fci_active']}")
    print(f"  FCI mismatch: {fci_mismatch}, Complaint mismatch: {complaint_mismatch}")
    print(f"  Total mismatch: {total_mismatch}")
