from itertools import combinations

# Define meters and attack vectors
candidates = ["m12", "m23", "m31", "pinj2"]
attacks = [
    {"attack_id": "a_theta2", "risk_weight": 7, "detectable_by": ["m12", "pinj2"]},
    {"attack_id": "a_theta3", "risk_weight": 6, "detectable_by": ["m31"]},
    {"attack_id": "a_load_shift", "risk_weight": 4, "detectable_by": ["m23", "pinj2"]},
    {"attack_id": "a_stealth_pair", "risk_weight": 9, "detectable_by": ["m12", "m31"]}
]

budget = 2

# Generate all combinations of 2 meters
best_combination = None
best_risk = -1
best_covered = []

for meter_combo in combinations(sorted(candidates), budget):
    meter_set = set(meter_combo)
    covered_attacks = []
    total_risk = 0
    
    for attack in attacks:
        # Check if any meter in the combination can detect this attack
        if any(meter in meter_set for meter in attack["detectable_by"]):
            covered_attacks.append(attack["attack_id"])
            total_risk += attack["risk_weight"]
    
    print(f"Meters: {meter_combo}, Covered attacks: {covered_attacks}, Total risk: {total_risk}")
    
    # Keep track of best (ties broken lexicographically by meter IDs)
    if total_risk > best_risk or (total_risk == best_risk and meter_combo < best_combination):
        best_risk = total_risk
        best_combination = meter_combo
        best_covered = covered_attacks

print(f"\nBest combination: {best_combination}")
print(f"Best risk weight: {best_risk}")
print(f"Covered attacks: {best_covered}")
