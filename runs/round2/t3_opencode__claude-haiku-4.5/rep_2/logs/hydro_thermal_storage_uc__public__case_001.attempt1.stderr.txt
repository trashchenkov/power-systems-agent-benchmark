import json
from itertools import product

# Read task
with open('task.json') as f:
    task = json.load(f)

uc_case = task['public_inputs']['uc_case']
emission_cap = uc_case['emission_cap_ton']
hours = uc_case['hours']
thermal_units = uc_case['thermal_units']

# Calculate residual load for each hour
residual_loads = []
for hour in hours:
    residual = hour['load_mw'] - hour['hydro_mw'] - hour['storage_discharge_mw']
    residual_loads.append((hour['hour_id'], residual))
    print(f"Hour {hour['hour_id']}: Load={hour['load_mw']}, Hydro={hour['hydro_mw']}, Storage={hour['storage_discharge_mw']}, Residual={residual}")

print(f"\nEmission cap: {emission_cap} tons")
print(f"Thermal units: {[(u['unit_id'], u['pmin_mw'], u['pmax_mw'], u['cost_usd_per_mwh'], u['emission_ton_per_mwh']) for u in thermal_units]}")

# Try to find feasible solution
# Strategy: use greedy approach - dispatch lowest cost first, respecting constraints
best_dispatch = None
best_cost = float('inf')
best_emissions = None

# For each hour, we need to dispatch coal and gas to meet residual load
# Subject to: pmin <= dispatch <= pmax for each unit, total emissions <= cap

def solve_hour(residual_load, emission_cap_remaining):
    """Find minimum cost dispatch for one hour that meets load with emission constraint"""
    coal = thermal_units[0]
    gas = thermal_units[1]
    
    best_hour_dispatch = None
    best_hour_cost = float('inf')
    
    # Try all combinations of coal and gas dispatch within limits
    for coal_dispatch in [i * 0.1 for i in range(int(coal['pmin_mw']*10), int(coal['pmax_mw']*10) + 1)]:
        for gas_dispatch in [i * 0.1 for i in range(int(gas['pmin_mw']*10), int(gas['pmax_mw']*10) + 1)]:
            # Check if meets load
            if abs(coal_dispatch + gas_dispatch - residual_load) < 0.01:
                # Check emissions
                emissions = coal_dispatch * coal['emission_ton_per_mwh'] + gas_dispatch * gas['emission_ton_per_mwh']
                if emissions <= emission_cap_remaining + 0.01:
                    cost = coal_dispatch * coal['cost_usd_per_mwh'] + gas_dispatch * gas['cost_usd_per_mwh']
                    if cost < best_hour_cost:
                        best_hour_cost = cost
                        best_hour_dispatch = {'coal': coal_dispatch, 'gas': gas_dispatch, 'emissions': emissions}
    
    return best_hour_dispatch, best_hour_cost

# Solve each hour
total_dispatch = {}
total_cost = 0
total_emissions = 0

for hour_id, residual_load in residual_loads:
    if residual_load < 0:
        print(f"ERROR: Hour {hour_id} has negative residual load {residual_load}")
        exit(1)
    
    remaining_cap = emission_cap - total_emissions
    dispatch, cost = solve_hour(residual_load, remaining_cap)
    
    if dispatch is None:
        print(f"ERROR: Cannot solve hour {hour_id} with residual load {residual_load}")
        exit(1)
    
    total_dispatch[hour_id] = dispatch
    total_cost += cost
    total_emissions += dispatch['emissions']
    print(f"Hour {hour_id}: Coal={dispatch['coal']:.1f} MW, Gas={dispatch['gas']:.1f} MW, Emissions={dispatch['emissions']:.2f} tons, Cost=${cost:.2f}")

print(f"\nTotal cost: ${total_cost:.2f}")
print(f"Total emissions: {total_emissions:.2f} tons (cap: {emission_cap})")
print(f"Feasible: {total_emissions <= emission_cap}")

# Format output
submission = {
    "thermal_dispatch_mw": {hour_id: {"coal": dispatch['coal'], "gas": dispatch['gas']} for hour_id, dispatch in total_dispatch.items()},
    "total_cost_usd": total_cost,
    "emissions_ton": total_emissions,
    "emission_cap_satisfied": total_emissions <= emission_cap
}

print("\nSubmission:")
print(json.dumps(submission, indent=2))

with open('submission.json', 'w') as f:
