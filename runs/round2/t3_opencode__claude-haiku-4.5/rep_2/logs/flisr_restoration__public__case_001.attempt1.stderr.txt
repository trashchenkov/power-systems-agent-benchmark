def is_connected(bus, branches_state, source):
    """Check if a bus is reachable from source given branch states"""
    edges = {}
    for bid, branch in branches_dict.items():
        if branches_state.get(bid, branch['normally_closed']):  # branch is closed
            u, v = branch['from_bus'], branch['to_bus']
            if u not in edges:
                edges[u] = []
            if v not in edges:
                edges[v] = []
            edges[u].append(v)
            edges[v].append(u)
    
    # BFS from source
    visited = set()
    queue = [source]
    visited.add(source)
    while queue:
        current = queue.pop(0)
        if current == bus:
            return True
        if current in edges:
            for neighbor in edges[current]:
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append(neighbor)
    return False

def is_radial(branches_state):
    """Check if network is radial (tree, no cycles)"""
    edges = {}
    edge_count = 0
    for bid, branch in branches_dict.items():
        if branches_state.get(bid, branch['normally_closed']):
            u, v = branch['from_bus'], branch['to_bus']
            if u not in edges:
                edges[u] = []
            if v not in edges:
                edges[v] = []
            edges[u].append(v)
            edges[v].append(u)
            edge_count += 1
    
    # For a tree: E = V - 1 where E is edges and V is nodes
    # But we only count connected components
    visited = set()
    def dfs(node):
        visited.add(node)
        for neighbor in edges.get(node, []):
            if neighbor not in visited:
                dfs(neighbor)
    
    components = 0
    for node in edges:
        if node not in visited:
            components += 1
            dfs(node)
    
    num_nodes = len(edges)
    return edge_count == num_nodes - components
