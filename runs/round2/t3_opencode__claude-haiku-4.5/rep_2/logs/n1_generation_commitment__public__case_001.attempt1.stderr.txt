import json
from itertools import combinations

# Load task data
with open('task.json', 'r') as f:
    task = json.load(f)

inputs = task['public_inputs']['commitment_case']
demand = inputs['demand_mw']
reserve = inputs['reserve_mw']
total_required = demand + reserve
units = inputs['units']

# Function to calculate commitment cost
def get_cost(unit_ids):
    cost = 0
    for uid in unit_ids:
        unit = next(u for u in units if u['unit_id'] == uid)
        cost += unit['startup_cost_usd'] + unit['no_load_cost_usd']
    return cost

# Function to check if a commitment is feasible
def is_feasible(unit_ids):
    if not unit_ids:
        return False
    
    # Get total available capacity
    total_capacity = sum(next(u['pmax_mw'] for u in units if u['unit_id'] == uid) for uid in unit_ids)
    
    # Check if it meets demand + reserve requirement
    if total_capacity < total_required:
        return False
    
    # Get capacity of largest unit
    capacities = [next(u['pmax_mw'] for u in units if u['unit_id'] == uid) for uid in unit_ids]
    largest_capacity = max(capacities)
    
    # Check N-1: after losing largest unit, still meet requirement
    n1_capacity = total_capacity - largest_capacity
    if n1_capacity < total_required:
        return False
    
    return True

# Try all combinations, starting from smallest
best_commitment = None
best_cost = float('inf')

for r in range(1, len(units) + 1):
    for combo in combinations([u['unit_id'] for u in units], r):
        if is_feasible(combo):
            cost = get_cost(combo)
            if cost < best_cost:
                best_cost = cost
                best_commitment = combo

if best_commitment:
    total_capacity = sum(next(u['pmax_mw'] for u in units if u['unit_id'] == uid) for uid in best_commitment)
    capacities = [next(u['pmax_mw'] for u in units if u['unit_id'] == uid) for uid in best_commitment]
    largest_capacity = max(capacities)
    n1_capacity = total_capacity - largest_capacity
    reserve_available = total_capacity - demand
    
    result = {
        "committed_units": list(best_commitment),
        "available_capacity_mw": total_capacity,
        "n1_secure_capacity_mw": n1_capacity,
        "reserve_mw": reserve_available,
        "commitment_cost_usd": best_cost,
        "rationale": f"Selected {','.join(best_commitment)} to meet demand {demand}MW + reserve {reserve}MW = {total_required}MW total, with N-1 survivability requiring {total_required}MW after loss of largest unit ({largest_capacity}MW)."
    }
    
    print(json.dumps(result, indent=2))
    
    # Verify constraints
    print("\n=== Verification ===")

    print(f"Total committed capacity: {total_capacity} MW")
    print(f"Largest unit: {largest_capacity} MW")
    print(f"N-1 capacity (after loss): {n1_capacity} MW")

    print(f"Meets demand + reserve? {total_capacity >= total_required}")
    print(f"N-1 secure? {n1_capacity >= total_required}")
else:
    print("No feasible solution found")

from itertools import combinations

units_data = [
    {"unit_id": "g1", "pmax_mw": 120, "startup_cost_usd": 400, "no_load_cost_usd": 80},
    {"unit_id": "g2", "pmax_mw": 160, "startup_cost_usd": 550, "no_load_cost_usd": 100},
    {"unit_id": "g3", "pmax_mw": 180, "startup_cost_usd": 700, "no_load_cost_usd": 120},
    {"unit_id": "g4", "pmax_mw": 90, "startup_cost_usd": 220, "no_load_cost_usd": 60}
]

demand = 260
reserve = 40
total_required = 300

def get_cost(unit_ids):
    return sum(next(u['startup_cost_usd'] + u['no_load_cost_usd'] for u in units_data if u['unit_id'] == uid) for uid in unit_ids)

def check_feasibility(unit_ids):
    if not unit_ids:
        return False, "Empty"
    
    capacity_map = {u['unit_id']: u['pmax_mw'] for u in units_data}
    total_capacity = sum(capacity_map[uid] for uid in unit_ids)
    
    if total_capacity < total_required:
        return False, f"Total {total_capacity} < required {total_required}"
    
    largest = max(capacity_map[uid] for uid in unit_ids)
    n1_capacity = total_capacity - largest
    
    if n1_capacity < total_required:
        return False, f"N-1: {n1_capacity} < required {total_required} (after losing {largest})"
    
    return True, "Feasible"

print("Testing all combinations:\n")
all_combos = []

for r in range(1, 5):
    print(f"=== {r} units ===")
    for combo in combinations(['g1', 'g2', 'g3', 'g4'], r):
        feasible, msg = check_feasibility(combo)
        cost = get_cost(combo)
        if feasible:
            capacity_map = {u['unit_id']: u['pmax_mw'] for u in units_data}
            total_cap = sum(capacity_map[uid] for uid in combo)
            largest = max(capacity_map[uid] for uid in combo)
            n1_cap = total_cap - largest
            print(f"  {combo}: FEASIBLE, cost=${cost}, capacity={total_cap}, N-1={n1_cap}")
            all_combos.append((cost, combo, total_cap, n1_cap))
        else:
            print(f"  {combo}: INFEASIBLE - {msg}")
    print()

if all_combos:
    print("=== BEST SOLUTION ===")
    best_cost, best_combo, best_cap, best_n1 = min(all_combos, key=lambda x: x[0])
    print(f"Units: {best_combo}")
    print(f"Cost: ${best_cost}")
    print(f"Total capacity: {best_cap} MW")
    print(f"N-1 capacity: {best_n1} MW")
