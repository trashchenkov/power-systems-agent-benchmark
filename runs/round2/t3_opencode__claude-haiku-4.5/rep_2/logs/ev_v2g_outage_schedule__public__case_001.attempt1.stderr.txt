        discharge_eff = ev['discharge_efficiency']

import json
from itertools import product

with open('task.json', 'r') as f:
    task = json.load(f)

case = task['v2g_outage_case']
evs = case['evs']

def get_feasible_dispatch(ev_id, ev, current_soc, interval_hours):
    """Get all feasible dispatch values for an EV at current SOC"""
    reserve = ev['reserve_soc_kwh']
    capacity = ev['capacity_kwh']
    discharge_max = ev['discharge_max_kw']
    discharge_eff = ev['discharge_efficiency']
    charge_max = ev['charge_max_kw']
    charge_eff = ev['charge_efficiency']
    dispatch_step = ev['dispatch_step_kw']
    
    feasible = []
    
    # Try discharge values (positive)
    kw = 0
    while kw <= discharge_max:
        if kw > 0:
            energy_drawn = kw * interval_hours / discharge_eff
            new_soc = current_soc - energy_drawn
            if new_soc >= reserve:
                feasible.append(kw)
        else:
            feasible.append(0)  # zero discharge
        kw += dispatch_step
    
    # Try charging (negative)
    kw = dispatch_step
    while kw <= charge_max:
        energy_added = kw * interval_hours * charge_eff
        new_soc = current_soc + energy_added
        if new_soc <= capacity:
            feasible.append(-kw)
        kw += dispatch_step
    
    if not feasible:
        feasible = [0]
    
    # Remove duplicates and sort
    feasible = sorted(list(set(feasible)))
    return feasible

# Solve with dynamic planning to maximize total energy served
soc_state = {ev_id: ev['initial_soc_kwh'] for ev_id, ev in evs.items()}
dispatch_results = {}
total_served = 0

for t_idx, t in enumerate(case['time_periods']):
    load_kw = case['critical_load_kw'][t]
    interval = case['interval_hours']
    
    # Get feasible options for each EV
    options = {}
    for ev_id, ev in evs.items():
        options[ev_id] = get_feasible_dispatch(ev_id, ev, soc_state[ev_id], interval)
    
    # Enumerate all combinations and pick best by served energy
    ev_ids = list(evs.keys())
    dispatch_choices = [options[ev_id] for ev_id in ev_ids]
    
    best_dispatch = {}
    best_served = 0
    best_new_soc = soc_state.copy()
    
    for combination in product(*dispatch_choices):
        dispatch_dict = dict(zip(ev_ids, combination))
        
        # Calculate total dispatch available
        total_dispatch = sum(v for v in dispatch_dict.values() if v > 0)
        served = min(total_dispatch, load_kw)
        
        # Only consider if this serves more or equal
        if served >= best_served or (served == best_served and total_dispatch > sum(v for v in best_dispatch.values() if v > 0)):
            # Validate
            new_soc_test = {}
            valid = True
            for ev_id, disp in dispatch_dict.items():
                ev = evs[ev_id]
                curr_soc = soc_state[ev_id]
                if disp > 0:
                    energy_drawn = disp * interval / ev['discharge_efficiency']
                    new_s = curr_soc - energy_drawn
                elif disp < 0:
                    energy_added = -disp * interval * ev['charge_efficiency']
                    new_s = curr_soc + energy_added
                else:
                    new_s = curr_soc
                new_soc_test[ev_id] = new_s
                if new_s < ev['reserve_soc_kwh']:
                    valid = False
                    break
            
            if valid:
                best_dispatch = dispatch_dict
                best_served = served
                best_new_soc = new_soc_test
    
    dispatch_results[t] = best_dispatch
    soc_state = best_new_soc
    total_served += best_served

print("Optimal dispatch (maximize served energy):")
for t, disp in dispatch_results.items():
    print(f"{t}: {disp}")

print(f"\nTotal served: {total_served}")
print(f"Final SOC: {soc_state}")

# Save to submission
submission = {
    "dispatch_kw": dispatch_results,
    "served_energy_kwh": round(total_served, 2),
    "final_soc_kwh": {ev_id: round(soc, 2) for ev_id, soc in soc_state.items()},
    "rationale": "Optimal enumeration of feasible dispatch values maximizing total energy served to critical load while respecting all power limits and reserve SOC constraints."
}

    json.dump(submission, f, indent=2)
