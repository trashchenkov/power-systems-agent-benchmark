import math

# Given data
base_mva = 100.0
voltage_kv = 10.5
voltage_factor_cmax = 1.1
voltage_factor_cmin = 1.0
fault_resistance_pu = 0.005

# Positive sequence impedances
impedances = [
    {"r_pu": 0.018, "x_pu": 0.155},  # grid_source
    {"r_pu": 0.012, "x_pu": 0.095},  # station_transformer
    {"r_pu": 0.002, "x_pu": 0.011}   # bus_duct
]

# Calculate network impedance (sum of all impedances)
R_network = sum(z["r_pu"] for z in impedances)
X_network = sum(z["x_pu"] for z in impedances)

# Magnitude of network impedance (bolted fault case)
Z_bolted_mag = math.sqrt(R_network**2 + X_network**2)

# Calculate base current: I_base = base_mva / (sqrt(3) * voltage_kv)
I_base = base_mva / (math.sqrt(3) * voltage_kv)

# Maximum current: Ik_max = voltage_factor_cmax / |Z_network| * I_base (bolted fault)
Ik_max = (voltage_factor_cmax / Z_bolted_mag) * I_base

# Maximum short-circuit power: Sk_max = sqrt(3) * voltage_kv * Ik_max
Sk_max = math.sqrt(3) * voltage_kv * Ik_max

# For minimum current, we need network impedance + fault resistance
Z_min_R = R_network + fault_resistance_pu
Z_min_mag = math.sqrt(Z_min_R**2 + X_network**2)

# Minimum current: Ik_min = voltage_factor_cmin / |Z_network + fault_resistance| * I_base
Ik_min = (voltage_factor_cmin / Z_min_mag) * I_base
