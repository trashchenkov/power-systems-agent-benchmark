import json

# Load task
with open('<sandbox_a1>/task.json', 'r') as f:
    task = json.load(f)

case = task['wind_forecast_case']
turbine = case['turbine']
power_curve = case['power_curve']
history = case['history']
weather_forecast = case['weather_forecast']

reference_density = case['reference_air_density_kg_m3']
forecast_step_hours = case['forecast_step_hours']
farm_capacity = turbine['count'] * turbine['rated_mw_each']

def calculate_power(wind_speed_m_s, air_density_kg_m3):
    """Calculate power using the exact formula from instructions"""
    # Corrected speed formula
    v_c = wind_speed_m_s * ((air_density_kg_m3 / reference_density) ** (1/3))
    
    cut_in = power_curve['cut_in_m_s']
    rated = power_curve['rated_m_s']
    cut_out = power_curve['cut_out_m_s']
    rated_power = turbine['count'] * turbine['rated_mw_each']
    
    # Power calculation logic
    if v_c < cut_in or v_c >= cut_out:
        power = 0.0
    elif v_c >= rated:
        power = rated_power
    else:
        # Linear interpolation in the transition zone (cut-in to rated)
        power = rated_power * (((v_c - cut_in) / (rated - cut_in)) ** 3)
    
    return power

# Calculate bias from history
bias_list = []
for record in history:
    estimated_power = calculate_power(record['wind_speed_m_s'], record['air_density_kg_m3'])
    residual = record['actual_power_mw'] - estimated_power
    bias_list.append(residual)

bias = sum(bias_list) / len(bias_list)

print(f"Bias list: {bias_list}")
print(f"Calculated bias: {bias}")

# Calculate forecasts
forecast_mw_by_timestamp = {}
forecast_values = []

for record in weather_forecast:
    estimated_power = calculate_power(record['wind_speed_m_s'], record['air_density_kg_m3'])
    forecast_power = estimated_power + bias
    # Clip to [0, farm_capacity]
    forecast_power = max(0.0, min(forecast_power, farm_capacity))
    forecast_mw_by_timestamp[record['timestamp']] = forecast_power
    forecast_values.append(forecast_power)

# Calculate total energy in MWh (sum of hourly forecasts times forecast_step_hours)
total_forecast_mwh = sum(forecast_values) * forecast_step_hours

# Find peak timestamp
peak_timestamp = max(weather_forecast, key=lambda x: forecast_mw_by_timestamp[x['timestamp']])['timestamp']

print("\nForecast by timestamp:")
for ts, mw in forecast_mw_by_timestamp.items():
    print(f"  {ts}: {mw:.4f} MW")
print(f"\nTotal forecast: {total_forecast_mwh:.4f} MWh")
print(f"Peak timestamp: {peak_timestamp}")

# Create submission
submission = {
    "forecast_mw_by_timestamp": forecast_mw_by_timestamp,
    "total_forecast_mwh": total_forecast_mwh,
    "peak_timestamp": peak_timestamp,
    "rationale": "Calculated using power curve formula with air density correction and historical bias adjustment"
}

# Save submission
with open('<sandbox_a1>/submission.json', 'w') as f:
    json.dump(submission, f, indent=2)

print("\nSubmission saved to submission.json")
