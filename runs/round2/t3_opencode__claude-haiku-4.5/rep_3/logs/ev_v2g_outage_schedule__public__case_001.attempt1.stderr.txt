import json

# Load task data
with open('task.json', 'r') as f:
    task = json.load(f)

# Extract case data
case = task['v2g_outage_case']
time_periods = case['time_periods']
interval_hours = case['interval_hours']
critical_load = case['critical_load_kw']
evs_spec = case['evs']

# Initialize state
ev_states = {}
for ev_id, spec in evs_spec.items():
    ev_states[ev_id] = {
        'soc_kwh': spec['initial_soc_kwh'],
        'capacity_kwh': spec['capacity_kwh'],
        'reserve_soc_kwh': spec['reserve_soc_kwh'],
    }

dispatch_kw = {}
total_served = 0

# For each time period, optimize dispatch
for t in time_periods:
    load_kw = critical_load[t]
    dispatch_kw[t] = {}
    
    # Determine what each EV can discharge
    available_discharge = {}
    for ev_id, spec in evs_spec.items():
        soc = ev_states[ev_id]['soc_kwh']
        reserve = spec['reserve_soc_kwh']
        # Maximum energy available above reserve
        available_energy = soc - reserve
        # Maximum power we can discharge
        max_discharge_kw = spec['discharge_max_kw']
        
        # Apply dispatch step quantization
        dispatch_step = spec['dispatch_step_kw']
        
        # Energy that can be discharged in this interval
        max_energy_available = max(0, available_energy)
        max_kw_for_interval = min(max_discharge_kw, max_energy_available / interval_hours)
        
        # Quantize to dispatch step
        quantized_kw = (int(max_kw_for_interval / dispatch_step)) * dispatch_step
        available_discharge[ev_id] = quantized_kw
    
    # Greedily dispatch to meet load
    remaining_load = load_kw
    period_served = 0
    
    # Sort by discharge capacity (largest first) for greedy allocation
    ev_ids_sorted = sorted(available_discharge.keys(), 
                          key=lambda x: available_discharge[x], 
                          reverse=True)
    
    for ev_id in ev_ids_sorted:
        if remaining_load <= 0:
            dispatch_kw[t][ev_id] = 0
            continue
        
        max_kw = available_discharge[ev_id]
        dispatch_amount = min(max_kw, remaining_load)
        
        # Quantize to dispatch step
        spec = evs_spec[ev_id]
        dispatch_step = spec['dispatch_step_kw']
        dispatch_amount = (int(dispatch_amount / dispatch_step)) * dispatch_step
        
        dispatch_kw[t][ev_id] = dispatch_amount
        remaining_load -= dispatch_amount
        period_served += dispatch_amount
        
        # Update SOC: discharge reduces it (with efficiency)
        energy_discharged = dispatch_amount * interval_hours
        actual_energy_loss = energy_discharged / spec['discharge_efficiency']
        ev_states[ev_id]['soc_kwh'] -= actual_energy_loss
    
    total_served += period_served * interval_hours

# Final SOC
final_soc = {}
for ev_id in evs_spec.keys():
    final_soc[ev_id] = ev_states[ev_id]['soc_kwh']

# Create submission
submission = {
    "dispatch_kw": dispatch_kw,
    "served_energy_kwh": total_served,
    "final_soc_kwh": final_soc,
    "rationale": "Greedy discharge strategy prioritizing EVs with highest available discharge capacity to maximize energy served while respecting power limits, reserve SOC constraints, and efficiency losses."
}

print(json.dumps(submission, indent=2))

# Save submission
with open('submission.json', 'w') as f:
    json.dump(submission, f, indent=2)

print("\n--- Submission saved ---")

import json

with open('task.json', 'r') as f:
    task = json.load(f)

with open('submission.json', 'r') as f:
    submission = json.load(f)

case = task['v2g_outage_case']
evs_spec = case['evs']

# Verify constraints
print("=== CONSTRAINT VERIFICATION ===\n")

# Initial states
ev_states = {}
for ev_id, spec in evs_spec.items():
    ev_states[ev_id] = {
        'soc_kwh': spec['initial_soc_kwh'],
    }

valid = True

# Check each time period
for t in case['time_periods']:
    print(f"\n--- Time period {t} ---")
    load = case['critical_load_kw'][t]
    dispatch_t = submission['dispatch_kw'].get(t, {})
    
    total_dispatch = 0
    for ev_id, dispatch_kw_amount in dispatch_t.items():
        spec = evs_spec[ev_id]
        soc_before = ev_states[ev_id]['soc_kwh']
        
        print(f"{ev_id}: dispatch {dispatch_kw_amount} kW (SOC before: {soc_before:.2f} kWh)")
        
        # Check power limit
        if dispatch_kw_amount > 0:  # Discharging
            if dispatch_kw_amount > spec['discharge_max_kw']:
                print(f"  ERROR: Exceeds discharge max ({spec['discharge_max_kw']} kW)")
                valid = False
            
            # Check reserve SOC constraint
            energy_discharged = dispatch_kw_amount * case['interval_hours']
            actual_loss = energy_discharged / spec['discharge_efficiency']
            soc_after = soc_before - actual_loss
            
            if soc_after < spec['reserve_soc_kwh']:
                print(f"  ERROR: SOC {soc_after:.2f} < reserve {spec['reserve_soc_kwh']}")
                valid = False
            
            ev_states[ev_id]['soc_kwh'] = soc_after
            print(f"  SOC after: {soc_after:.2f} kWh (≥ {spec['reserve_soc_kwh']}? {soc_after >= spec['reserve_soc_kwh']})")
        
        elif dispatch_kw_amount < 0:  # Charging
            if abs(dispatch_kw_amount) > spec['charge_max_kw']:
                print(f"  ERROR: Exceeds charge max ({spec['charge_max_kw']} kW)")
                valid = False
        
        # Check dispatch step
        if dispatch_kw_amount % spec['dispatch_step_kw'] != 0 and dispatch_kw_amount != 0:
            print(f"  WARNING: Not multiple of dispatch_step {spec['dispatch_step_kw']}")
        
        total_dispatch += dispatch_kw_amount
    
    print(f"Total dispatch: {total_dispatch} kW, Load: {load} kW")

print(f"\n--- Final States ---")
for ev_id in evs_spec:
    spec = evs_spec[ev_id]
    final_soc = ev_states[ev_id]['soc_kwh']
    submitted_final = submission['final_soc_kwh'][ev_id]
    match = abs(final_soc - submitted_final) < 0.01
    print(f"{ev_id}: Final SOC {final_soc:.2f} kWh (submitted: {submitted_final:.2f}, match: {match})")
    if not match:
        valid = False

print(f"\n--- Energy Verification ---")
print(f"Submitted served energy: {submission['served_energy_kwh']} kWh")

print(f"\nAll constraints satisfied: {valid}")
