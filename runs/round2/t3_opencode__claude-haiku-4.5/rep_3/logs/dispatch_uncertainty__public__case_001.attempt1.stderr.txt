import json

# Load task data
with open('task.json', 'r') as f:
    task = json.load(f)

case = task['dispatch_uncertainty_case']
generators = case['generators']
scenarios = case['scenarios']
time_periods = case['time_periods']
reserve_req = case['reserve_requirement_mw']

# For each time period, compute maximum net load across scenarios
max_net_load = {}
for t in time_periods:
    max_nl = 0
    for scenario in scenarios:
        load = scenario['load_mw'][t]
        renewable = scenario['renewable_mw'][t]
        net_load = load - renewable
        max_nl = max(max_nl, net_load)
    max_net_load[t] = max_nl

print("Maximum net load per time period:")
for t in time_periods:
    print(f"  {t}: {max_net_load[t]} MW")

# Minimum generation needed per time period
min_generation_needed = {}
for t in time_periods:
    min_generation_needed[t] = max_net_load[t] + reserve_req
    print(f"\n{t}: Max net load {max_net_load[t]} + reserve {reserve_req} = {min_generation_needed[t]} MW needed")

# Generator info
g1 = next(g for g in generators if g['id'] == 'g1')
g2 = next(g for g in generators if g['id'] == 'g2')

print(f"\nGenerator constraints:")
print(f"  g1: {g1['pmin_mw']}-{g1['pmax_mw']} MW, cost ${g1['variable_cost_per_mwh']}/MWh")
print(f"  g2: {g2['pmin_mw']}-{g2['pmax_mw']} MW, cost ${g2['variable_cost_per_mwh']}/MWh")

# Linear programming approach: minimize cost
# For each time period, we need to decide dispatch for g1 and g2
# Objective: minimize sum of (g1_dispatch * g1_cost + g2_dispatch * g2_cost)
# Constraints:
#   g1_dispatch + g2_dispatch >= min_generation_needed[t]
#   g1_pmin <= g1_dispatch <= g1_pmax
#   g2_pmin <= g2_dispatch <= g2_pmax

dispatch = {}
total_cost = 0.0
margins = {}

for t in time_periods:
    needed = min_generation_needed[t]
    
    # Greedy approach: use the cheaper generator first
    g1_cost = g1['variable_cost_per_mwh']
    g2_cost = g2['variable_cost_per_mwh']
    
    # g1 is cheaper (18 < 36), so use it as much as possible
    g1_dispatch = min(g1['pmax_mw'], needed - g2['pmin_mw'])
    g1_dispatch = max(g1['pmin_mw'], g1_dispatch)
    
    # Then use g2 to meet the remaining requirement
    remaining = needed - g1_dispatch
    g2_dispatch = max(g2['pmin_mw'], remaining)
    g2_dispatch = min(g2['pmax_mw'], g2_dispatch)
    
    # Check if we can meet the requirement
    total_dispatch = g1_dispatch + g2_dispatch
    
    if total_dispatch < needed:
        print(f"\nERROR: Cannot meet requirement for {t}")
        print(f"  Need: {needed} MW, can provide: {total_dispatch} MW")
    else:
        print(f"\n{t}:")
        print(f"  g1: {g1_dispatch} MW")
        print(f"  g2: {g2_dispatch} MW")
        print(f"  Total: {total_dispatch} MW (need: {needed} MW)")
        print(f"  Margin: {total_dispatch - max_net_load[t]} MW")
        
        dispatch[t] = {
            'g1': g1_dispatch,
            'g2': g2_dispatch
        }
        margins[t] = total_dispatch - max_net_load[t]
        total_cost += g1_dispatch * g1_cost + g2_dispatch * g2_cost

print(f"\nTotal cost: ${total_cost}")

# Create submission
submission = {
    "dispatch_mw": {
        "g1": {
            "t1": dispatch['t1']['g1'],
            "t2": dispatch['t2']['g1'],
            "t3": dispatch['t3']['g1']
        },
        "g2": {
            "t1": dispatch['t1']['g2'],
            "t2": dispatch['t2']['g2'],
            "t3": dispatch['t3']['g2']
        }
    },
    "worst_case_margin_mw": margins,
    "total_cost": total_cost
}

print("\nSubmission:")
print(json.dumps(submission, indent=2))

with open('submission.json', 'w') as f:
    json.dump(submission, f, indent=2)
