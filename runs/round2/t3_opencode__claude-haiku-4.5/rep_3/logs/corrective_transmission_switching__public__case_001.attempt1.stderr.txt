def build_network(branch_status):
    """Build network from branch status dict (branch_id -> True/False for closed/open)"""
    graph = defaultdict(list)
    for branch_id, branch_data in branches_data.items():
        is_open = not branch_status.get(branch_id, branch_data['closed'])
        if not is_open:  # if branch is closed
            from_bus = branch_data['from_bus']
            to_bus = branch_data['to_bus']
            graph[from_bus].append(to_bus)
            graph[to_bus].append(from_bus)
    return graph

def is_connected(graph, all_buses):
    """Check if the network is connected via BFS"""
    if not all_buses:
        return True
    visited = set()
    queue = deque([all_buses[0]])
    visited.add(all_buses[0])
    while queue:
        node = queue.popleft()
        for neighbor in graph[node]:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append(neighbor)
    return len(visited) == len(all_buses)

def solve_dc_powerflow(branch_status):
    """
    Solve DC power flow with fixed topology.
    Returns bus angles and branch flows, or None if infeasible.
    """
    bus_list = list(buses.keys())
    bus_idx = {b: i for i, b in enumerate(bus_list)}
    n = len(bus_list)
    
    # Build admittance matrix (susceptance matrix for DC flow)
    B = [[0.0] * n for _ in range(n)]
    
    for branch_id, branch_data in branches_data.items():
        if not branch_status.get(branch_id, branch_data['closed']):
            continue  # branch is open
        
        from_idx = bus_idx[branch_data['from_bus']]
        to_idx = bus_idx[branch_data['to_bus']]
        x = reactance_data[branch_id]
        b = 1.0 / x  # susceptance
        
        B[from_idx][from_idx] += b
        B[to_idx][to_idx] += b
        B[from_idx][to_idx] -= b
        B[to_idx][from_idx] -= b
    
    # DC power flow: B * theta = P (with slack bus at bus 0)
    # Set slack bus angle to 0, solve for other angles
    # We'll use a simple iterative approach or direct inversion
    
    # Build reduced system (remove slack bus row/col)
    B_red = [[B[i][j] for j in range(1, n)] for i in range(1, n)]
    P_red = [buses[bus_list[i]] for i in range(1, n)]
    
    # Solve B_red * theta_red = P_red using Gaussian elimination
    try:
        # Create augmented matrix
        aug = [B_red[i] + [P_red[i]] for i in range(n-1)]
        
        # Gaussian elimination with partial pivoting
        for col in range(n-1):
            # Find pivot
            max_row = col
            for row in range(col+1, n-1):
                if abs(aug[row][col]) > abs(aug[max_row][col]):
                    max_row = row
            aug[col], aug[max_row] = aug[max_row], aug[col]
            
            if abs(aug[col][col]) < 1e-10:
                return None  # singular matrix
            
            # Eliminate below
            for row in range(col+1, n-1):
                factor = aug[row][col] / aug[col][col]
                for j in range(col, n):
                    aug[row][j] -= factor * aug[col][j]
        
        # Back substitution
        theta_red = [0.0] * (n-1)
        for i in range(n-2, -1, -1):
            theta_red[i] = aug[i][n-1]
            for j in range(i+1, n-1):
                theta_red[i] -= aug[i][j] * theta_red[j]
            theta_red[i] /= aug[i][i]
        
        # Full angle vector (slack bus angle = 0)
        theta = [0.0] + theta_red
        
        # Compute flows
        flows = {}
        for branch_id, branch_data in branches_data.items():
            if not branch_status.get(branch_id, branch_data['closed']):
                flows[branch_id] = 0.0
                continue
            
            from_idx = bus_idx[branch_data['from_bus']]
            to_idx = bus_idx[branch_data['to_bus']]
            x = reactance_data[branch_id]
            flow = (theta[from_idx] - theta[to_idx]) / x
            flows[branch_id] = flow
        
        return theta, flows
    except:
        return None

def check_feasibility(branch_status):
    """Check if a topology is feasible (connected and all flows within limits)"""
    graph = build_network(branch_status)
    if not is_connected(graph, list(buses.keys())):
        return False, None, None
    
    result = solve_dc_powerflow(branch_status)
    if result is None:
        return False, None, None
    
    theta, flows = result
    
    # Check all branch flows
    max_loading = 0.0
    for branch_id, flow in flows.items():
        rating = rating_data[branch_id]
        loading = abs(flow) / rating
        if loading > 1.0 + 1e-6:  # flow_tolerance_pct = 1e-06
            return False, flows, loading
        max_loading = max(max_loading, loading)
    
    return True, flows, max_loading
