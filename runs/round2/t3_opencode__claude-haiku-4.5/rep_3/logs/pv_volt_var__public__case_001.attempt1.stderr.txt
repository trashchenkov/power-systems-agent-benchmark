import json
import math

# Load task parameters
with open('task.json', 'r') as f:
    task = json.load(f)

# Extract public inputs
voltage_limits = task['voltage_limits_pu']
timesteps = task['public_inputs']['timesteps']

print("=== Task Analysis ===")
print(f"Voltage limits: {voltage_limits['min']} - {voltage_limits['max']} pu")
print("\nTimesteps:")
for ts in timesteps:
    print(f"  {ts['id']}: V_base={ts['base_voltage_pu']} pu, "
          f"P={ts['pv_kw']} kW, "
          f"S={ts['inverter_kva']} kVA, "
          f"sens={ts['voltage_sensitivity_pu_per_kvar']} pu/kvar")

# For each timestep, calculate reactive power capability
print("\n=== Reactive Power Capability ===")
capabilities = {}
for ts in timesteps:
    p_kw = ts['pv_kw']
    s_kva = ts['inverter_kva']
    q_max = math.sqrt(s_kva**2 - p_kw**2)
    capabilities[ts['id']] = q_max
    print(f"{ts['id']}: Q_max = ±{q_max:.2f} kvar")

# Strategy: Design a 4-point Volt-VAR curve
# Points: (0.90, 1.0), (0.95, 0.5), (1.05, -0.5), (1.10, -1.0)
# This curve absorbs reactive power when voltage is low (to raise voltage)
# and injects reactive power when voltage is high (to lower voltage)

# However, we need to optimize the curve to stay within capabilities and limits

# Let's use a more conservative approach with 5 breakpoints
# The curve should be monotonically non-increasing in q_fraction as voltage increases

print("\n=== Generating Volt-VAR Curve ===")

# Define candidate breakpoints
curve_points = [
    (0.90, 1.0),    # Low voltage: inject max reactive power (raise voltage)
    (0.95, 0.5),    # Near min limit: reduce injection
    (1.00, 0.0),    # Nominal: no reactive power
    (1.05, -0.5),   # Near max limit: absorb reactive power (lower voltage)
    (1.10, -1.0),   # High voltage: absorb max reactive power
]

print("Initial curve points:")
for v, q in curve_points:
    print(f"  V={v:.2f} pu, Q_fraction={q:.2f}")

# Verify constraints
print("\n=== Verifying Constraints ===")

# 1. Check strictly increasing voltage
voltage_increasing = all(curve_points[i][0] < curve_points[i+1][0] for i in range(len(curve_points)-1))
print(f"1. Strictly increasing voltage: {voltage_increasing}")

# 2. Check monotonic non-increasing q_fraction
q_decreasing = all(curve_points[i][1] >= curve_points[i+1][1] for i in range(len(curve_points)-1))
print(f"2. Monotonic non-increasing q_fraction: {q_decreasing}")

# 3. Check inverter capability and voltage limits for each timestep
print("\n3. Checking inverter capability and voltage limits:")
feasible = True
for ts in timesteps:
    v_base = ts['base_voltage_pu']
    q_max = capabilities[ts['id']]
    sensitivity = ts['voltage_sensitivity_pu_per_kvar']
    
    # Interpolate q_fraction at this base voltage
    q_fraction = None
    for i in range(len(curve_points)-1):
        v1, q1 = curve_points[i]
        v2, q2 = curve_points[i+1]
        if v1 <= v_base <= v2:
            # Linear interpolation
            t = (v_base - v1) / (v2 - v1) if v2 != v1 else 0
            q_fraction = q1 + t * (q2 - q1)
            break
    
    if q_fraction is None:
        if v_base < curve_points[0][0]:
            q_fraction = curve_points[0][1]
        else:
            q_fraction = curve_points[-1][1]
    
    # Convert to actual reactive power
    q_kvar = q_fraction * q_max
    
    # Check capability
    if abs(q_kvar) > q_max + 1e-6:
        print(f"  {ts['id']}: EXCEEDS Q capability ({abs(q_kvar):.2f} > {q_max:.2f})")
        feasible = False
    else:
        print(f"  {ts['id']}: Q={q_kvar:.2f} kvar (OK, limit={q_max:.2f})")
    
    # Calculate corrected voltage
    v_corrected = v_base + sensitivity * q_kvar
    
    # Check voltage limits
    in_limits = voltage_limits['min'] <= v_corrected <= voltage_limits['max']
    status = "OK" if in_limits else "VIOLATION"
    print(f"    V_base={v_base:.3f}, Q={q_kvar:.2f}, V_corr={v_corrected:.4f} [{status}]")
    
    if not in_limits:
        feasible = False

print(f"\n4. Overall feasible: {feasible}")

# Generate submission
submission = {
    "volt_var_curve": [{"voltage_pu": v, "q_fraction": q} for v, q in curve_points],
    "rationale": "Monotonic Volt-VAR curve designed to mitigate voltage sag and swell. "
                 "Injects reactive power at low voltages to raise voltage, "
                 "and absorbs reactive power at high voltages to lower voltage. "
                 "Stays within inverter reactive power capability and voltage limits."
}

print("\n=== Final Submission ===")
print(json.dumps(submission, indent=2))

# Write submission
with open('submission.json', 'w') as f:
    json.dump(submission, f, indent=2)

print("\nSubmission written to submission.json")
