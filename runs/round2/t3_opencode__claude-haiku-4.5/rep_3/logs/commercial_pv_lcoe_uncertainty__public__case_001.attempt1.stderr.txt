import json
import numpy as np

# Load task.json
with open('task.json', 'r') as f:
    task = json.load(f)

pv_lcoe_case = task['pv_lcoe_case']

# Extract inputs
system_size_kw = pv_lcoe_case['system_size_kw_dc']
hard_cost_usd_per_kw = pv_lcoe_case['hard_cost_usd_per_kw']
soft_cost_components = pv_lcoe_case['soft_cost_components_usd_per_kw']
capital_recovery_factor = pv_lcoe_case['capital_recovery_factor']
fixed_om_usd_per_kw_year = pv_lcoe_case['fixed_om_usd_per_kw_year']
annual_energy_mwh_year1 = pv_lcoe_case['annual_energy_mwh_year1']
soft_cost_scenarios = pv_lcoe_case['soft_cost_scenarios']

# Dictionary to store LCOE for each scenario
scenario_lcoe = {}
lcoe_values = []

# For each scenario, calculate LCOE
for scenario_id, multipliers in soft_cost_scenarios.items():
    # Calculate total soft costs for this scenario
    total_soft_cost_per_kw = sum(
        soft_cost_components[component] * multipliers[component]
        for component in soft_cost_components
    )
    
    # Calculate CAPEX
    capex = system_size_kw * (hard_cost_usd_per_kw + total_soft_cost_per_kw)
    
    # Calculate LCOE
    numerator = (capex * capital_recovery_factor) + (fixed_om_usd_per_kw_year * system_size_kw)
    lcoe = numerator / annual_energy_mwh_year1
    
    scenario_lcoe[scenario_id] = round(lcoe, 2)
    lcoe_values.append(lcoe)

# Calculate quantiles using linear interpolation
lcoe_values_sorted = sorted(lcoe_values)
n = len(lcoe_values_sorted)

def calculate_quantile(values, p):
    """Calculate quantile with linear interpolation"""
    sorted_values = sorted(values)
    n = len(sorted_values)
    # Using the linear interpolation method (default in numpy)
    index = (p / 100) * (n - 1)
    lower_index = int(np.floor(index))
    upper_index = int(np.ceil(index))
    
    if lower_index == upper_index:
        return sorted_values[lower_index]
    
    # Linear interpolation
    fraction = index - lower_index
    lower_value = sorted_values[lower_index]
    upper_value = sorted_values[upper_index]
    return lower_value + fraction * (upper_value - lower_value)

p10 = calculate_quantile(lcoe_values, 10)
p50 = calculate_quantile(lcoe_values, 50)
p90 = calculate_quantile(lcoe_values, 90)

# Calculate mean
mean_lcoe = np.mean(lcoe_values)

# Create submission
submission = {
    "scenario_lcoe_usd_per_mwh": scenario_lcoe,
    "lcoe_quantiles_usd_per_mwh": {
        "p10": round(p10, 2),
        "p50": round(p50, 2),
        "p90": round(p90, 2)
    },
    "expected_lcoe_usd_per_mwh": round(mean_lcoe, 2)
}

# Write submission.json
with open('submission.json', 'w') as f:
    json.dump(submission, f, indent=2)

print(json.dumps(submission, indent=2))
