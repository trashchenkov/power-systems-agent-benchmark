# Build adjacency list for network
edges = []
for b in branches.values():
    edges.append((b['from_bus'], b['to_bus'], b['id']))

print("Network edges:")
for u, v, bid in edges:
    print(f"  {u} -- {v} ({bid})")

# Function to check if graph is connected after removing an edge
def is_connected_after_removal(edges, removed_branch_id):
    # Filter out the removed branch
    remaining_edges = [(u, v) for u, v, bid in edges if bid != removed_branch_id]
    
    # Find all unique nodes
    nodes = set()
    for u, v in remaining_edges:
        nodes.add(u)
        nodes.add(v)
    
    if len(nodes) <= 1:
        return True
    
    # BFS from first node
    start = next(iter(nodes))
    visited = {start}
    queue = [start]
    
    adj = {n: [] for n in nodes}
    for u, v in remaining_edges:
        adj[u].append(v)
        adj[v].append(u)
    
    while queue:
        node = queue.pop(0)
        for neighbor in adj[node]:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append(neighbor)
    
    return len(visited) == len(nodes)
