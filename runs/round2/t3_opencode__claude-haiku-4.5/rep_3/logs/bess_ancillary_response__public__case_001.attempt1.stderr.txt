import json
import math

# Load task data
with open('task.json', 'r') as f:
    task = json.load(f)

case = task['bess_response_case']
measurements = case['measurements']

# Extract parameters
nominal_freq = case['nominal_frequency_hz']
target_voltage = case['target_voltage_pu']
interval_sec = case['interval_seconds']
initial_soc = case['initial_soc_mwh']
soc_min = case['soc_min_mwh']
soc_max = case['soc_max_mwh']
p_discharge_limit = case['p_discharge_limit_mw']
p_charge_limit = case['p_charge_limit_mw']
s_rating = case['s_rating_mva']
freq_droop = case['frequency_droop_mw_per_hz']
volt_droop = case['voltage_droop_mvar_per_pu']
charge_eff = case['charge_efficiency']
discharge_eff = case['discharge_efficiency']

# Initialize results
p_response = {}
q_response = {}
current_soc = initial_soc

# Process each measurement
timestamps = sorted(measurements.keys())

for ts in timestamps:
    meas = measurements[ts]
    freq_hz = meas['frequency_hz']
    volt_pu = meas['voltage_pu']
    
    # Step 1: Compute P response from frequency droop
    p_raw = (nominal_freq - freq_hz) * freq_droop
    
    # Clip P by charge/discharge limits
    # Positive P = discharge (withdraw from battery)
    # Negative P = charge (inject into battery)
    p_clipped = max(-p_charge_limit, min(p_discharge_limit, p_raw))
    p_response[ts] = p_clipped
    
    # Step 2: Compute Q response from voltage droop
    q_raw = (target_voltage - volt_pu) * volt_droop
    
    # Compute apparent-power capability after P
    # S^2 = P^2 + Q^2, so Q_max = sqrt(S^2 - P^2)
    s_squared = s_rating ** 2
    p_squared = p_clipped ** 2
    
    if s_squared >= p_squared:
        q_max = math.sqrt(s_squared - p_squared)
    else:
        q_max = 0
    
    # Clip Q by +/- q_max
    q_clipped = max(-q_max, min(q_max, q_raw))
    q_response[ts] = q_clipped
    
    # Step 3: Update SOC
    # Positive P discharges (reduces SOC), negative P charges (increases SOC)
    energy_change = p_clipped * (interval_sec / 3600)  # Convert seconds to hours
    
    if p_clipped > 0:  # Discharging
        # Energy removed from battery accounting for efficiency
        soc_change = -energy_change / discharge_eff
    elif p_clipped < 0:  # Charging
        # Energy added to battery accounting for efficiency
        soc_change = -energy_change * charge_eff  # p_clipped is negative
    else:
        soc_change = 0
    
    current_soc += soc_change
    # Clip SOC to min/max bounds
    current_soc = max(soc_min, min(soc_max, current_soc))

final_soc = current_soc

# Prepare output
result = {
    "p_response_mw": p_response,
    "q_response_mvar": q_response,
    "final_soc_mwh": final_soc
}

print(json.dumps(result, indent=2))

# Save to submission.json
with open('submission.json', 'w') as f:
    json.dump(result, f, indent=2)

print("\nSubmission saved to submission.json")
