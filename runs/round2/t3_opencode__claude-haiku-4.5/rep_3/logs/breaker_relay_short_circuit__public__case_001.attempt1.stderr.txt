import math
import json

# Input data
voltage_kv = 10.5
V_ll = voltage_kv * 1000  # Convert to volts

# Source impedance
Z_source = complex(0.045, 0.42)

# Transformer impedance
Z_transformer = complex(0.018, 0.16)

# Feeder impedance per km
Z_feeder_per_km = complex(0.18, 0.32)

# Fault locations
fault_locations = [
    {"id": "near_bus_fault", "distance_km": 0.2},
    {"id": "mid_feeder_fault", "distance_km": 4.0},
    {"id": "remote_end_fault", "distance_km": 8.0}
]

# Selection rules
breaker_safety_factor = 1.1
relay_load_margin = 1.25
minimum_fault_sensitivity = 2.0
max_load_current_a = 360.0

# Available breaker ratings
available_ratings_ka = [8.0, 12.5, 16.0, 20.0, 25.0, 31.5]

# Calculate fault currents
results = []
fault_currents_3ph = []
fault_currents_2ph = []

for fault in fault_locations:
    distance_km = fault["distance_km"]
    
    # Calculate total impedance up to fault point
    Z_feeder = Z_feeder_per_km * distance_km
    Z_total = Z_source + Z_transformer + Z_feeder
    
    # Magnitude of total impedance
    Z_mag = abs(Z_total)
    
    # Three-phase fault current (I = V_ll / (sqrt(3) * |Z_total|))
    I_3ph = V_ll / (math.sqrt(3) * Z_mag)
    
    # Two-phase fault current (approximately sqrt(3)/2 of three-phase)
    I_2ph = (math.sqrt(3) / 2) * I_3ph
    
    fault_currents_3ph.append(I_3ph)
    fault_currents_2ph.append(I_2ph)
    
    results.append({
        "id": fault["id"],
        "distance_km": distance_km,
        "Z_source": str(Z_source),
        "Z_transformer": str(Z_transformer),
        "Z_feeder": str(Z_feeder),
        "Z_total": str(Z_total),
        "Z_mag": Z_mag,
        "I_3ph_A": I_3ph,
        "I_2ph_A": I_2ph,
        "I_3ph_kA": I_3ph / 1000,
        "I_2ph_kA": I_2ph / 1000
    })

# Find max 3-phase and min 2-phase
max_3ph_fault_kA = max(fault_currents_3ph) / 1000
min_2ph_fault_kA = min(fault_currents_2ph) / 1000

print("Fault current calculations:")
for r in results:
    print(f"\n{r['id']} (distance: {r['distance_km']} km):")
    print(f"  Total impedance: {r['Z_mag']:.6f} Ω")
    print(f"  3-phase current: {r['I_3ph_kA']:.3f} kA")
    print(f"  2-phase current: {r['I_2ph_kA']:.3f} kA")

print(f"\nMax 3-phase fault current: {max_3ph_fault_kA:.3f} kA")
print(f"Min 2-phase fault current: {min_2ph_fault_kA:.3f} kA")

# Select breaker rating
required_rating = max_3ph_fault_kA * breaker_safety_factor
print(f"\nRequired breaker rating (with {breaker_safety_factor}x safety): {required_rating:.3f} kA")

breaker_rating = None
for rating in available_ratings_ka:
    if rating >= required_rating:
        breaker_rating = rating
        break

print(f"Selected breaker rating: {breaker_rating} kA")

# Select relay pickup current
relay_min = max_load_current_a * relay_load_margin  # Load margin minimum
relay_max = (min_2ph_fault_kA * 1000) / minimum_fault_sensitivity  # Fault sensitivity maximum

print(f"\nRelay pickup constraints:")
print(f"  Minimum (load margin): {relay_min:.1f} A")
print(f"  Maximum (fault sensitivity): {relay_max:.1f} A")

# Choose a reasonable pickup between min and max
relay_pickup = (relay_min + relay_max) / 2
print(f"  Selected relay pickup: {relay_pickup:.1f} A")

# Verify constraints
print("\n=== CONSTRAINT VERIFICATION ===")
print(f"1. Max 3-ph fault: {max_3ph_fault_kA:.3f} kA ✓")
print(f"2. Min 2-ph fault: {min_2ph_fault_kA:.3f} kA ✓")
print(f"3. Breaker rating: {breaker_rating} kA (required: {required_rating:.3f} kA) - {breaker_rating >= required_rating} ✓")
print(f"4. Relay pickup >= {relay_min:.1f} A: {relay_pickup >= relay_min} ✓")
print(f"5. Relay pickup <= {relay_max:.1f} A: {relay_pickup <= relay_max} ✓")

# Create submission
submission = {
    "max_3ph_fault_ka": round(max_3ph_fault_kA, 3),
    "min_2ph_fault_ka": round(min_2ph_fault_kA, 3),
    "breaker_interrupting_rating_ka": breaker_rating,
    "relay_pickup_a_primary": round(relay_pickup, 1),
    "rationale": f"Fault study for 10.5 kV feeder with 8.0 km length. Maximum 3-phase current {max_3ph_fault_kA:.3f} kA requires breaker rating ≥ {required_rating:.3f} kA; selected {breaker_rating} kA. Relay pickup {relay_pickup:.1f} A ensures sensitivity to minimum 2-phase fault ({min_2ph_fault_kA:.3f} kA) while maintaining margin above maximum load current ({max_load_current_a:.0f} A)."
}

print("\n=== SUBMISSION ===")
print(json.dumps(submission, indent=2))
