import json
import math

# Load task data
with open('task.json', 'r') as f:
    task = json.load(f)

sc = task['short_circuit_case']
base_mva = sc['base_mva']
voltage_kv = sc['voltage_kv']
cmax = sc['voltage_factor_cmax']
cmin = sc['voltage_factor_cmin']
fault_resistance_pu = sc['fault_resistance_pu']

# Sum positive-sequence impedances
total_r = 0.0
total_x = 0.0
for imp in sc['positive_sequence_impedances_pu']:
    total_r += imp['r_pu']
    total_x += imp['x_pu']

print(f"Total R: {total_r}")
print(f"Total X: {total_x}")

# Network impedance magnitude (bolted fault - no fault resistance)
z_bolted = math.sqrt(total_r**2 + total_x**2)
print(f"Z_bolted (magnitude): {z_bolted}")

# Network impedance with fault resistance
z_with_fault = math.sqrt((total_r + fault_resistance_pu)**2 + total_x**2)
print(f"Z_with_fault (magnitude): {z_with_fault}")

# Base current: I_base = base_mva / (sqrt(3) * voltage_kv)
i_base = base_mva / (math.sqrt(3) * voltage_kv)
print(f"I_base (kA): {i_base}")

# Maximum current (bolted fault): Ik_max = cmax / Z_bolted * I_base
ik_max = (cmax / z_bolted) * i_base
print(f"Ik_max (kA): {ik_max}")

# Maximum short-circuit power: Sk_max = sqrt(3) * voltage_kv * Ik_max
sk_max = math.sqrt(3) * voltage_kv * ik_max
print(f"Sk_max (MVA): {sk_max}")

# Minimum current (with fault resistance): Ik_min = cmin / Z_with_fault * I_base
ik_min = (cmin / z_with_fault) * i_base
print(f"Ik_min (kA): {ik_min}")

# Create submission
submission = {
    "ik_max_ka": round(ik_max, 6),
    "sk_max_mva": round(sk_max, 6),
    "ik_min_ka": round(ik_min, 6),
    "z_bolted_mag_pu": round(z_bolted, 6),
    "z_min_mag_pu": round(z_with_fault, 6),
    "rationale": "IEC 60909 maximum/minimum three-phase short-circuit current calculation using positive-sequence impedances summed to network impedance."
}

print("\nSubmission:")
print(json.dumps(submission, indent=2))

# Write submission.json
with open('submission.json', 'w') as f:
    json.dump(submission, f, indent=2)

print("\nsubmission.json written successfully.")
