    unit_id = unit['unit_id']
    for hour in hours:
        hour_id = hour['hour_id']
        dispatch[(unit_id, hour_id)] = LpVariable(f"dispatch_{unit_id}_{hour_id}", lowBound=0)

cost_expr = 0
for unit in thermal_units:
    unit_id = unit['unit_id']
    cost = unit['cost_usd_per_mwh']
    for hour in hours:
        hour_id = hour['hour_id']
        cost_expr += cost * dispatch[(unit_id, hour_id)]

prob += cost_expr

# Constraints:
# 1. Demand balance: thermal dispatch must meet residual load each hour
for hour in hours:
    hour_id = hour['hour_id']
    load = hour['load_mw']
    hydro = hour['hydro_mw']
    storage = hour['storage_discharge_mw']
    residual_load = load - hydro - storage
    
    thermal_sum = 0
    for unit in thermal_units:
        unit_id = unit['unit_id']
        thermal_sum += dispatch[(unit_id, hour_id)]
    
    prob += thermal_sum == residual_load, f"balance_{hour_id}"

    unit_id = unit['unit_id']
    pmin = unit['pmin_mw']
    pmax = unit['pmax_mw']

        prob += dispatch[(unit_id, hour_id)] <= pmax, f"max_{unit_id}_{hour_id}"
        # If dispatch > 0, then dispatch >= pmin. We need a binary variable for this.
        # Since this is a continuous dispatch, let's assume if unit is used, it must be >= pmin
        # We can add this as a constraint: either dispatch = 0 or dispatch >= pmin
        
# Actually, let me reconsider. The standard UC formulation requires binary on/off variables.
