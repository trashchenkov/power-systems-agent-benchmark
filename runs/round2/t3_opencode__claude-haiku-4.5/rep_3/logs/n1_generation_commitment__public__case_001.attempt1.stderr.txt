import json
from itertools import combinations

# Load input data
with open('task.json', 'r') as f:
    task = json.load(f)

data = task['public_inputs']['commitment_case']
demand = data['demand_mw']
reserve = data['reserve_mw']
units = data['units']

total_requirement = demand + reserve  # 260 + 40 = 300 MW

# Try all possible subsets of units
best_cost = float('inf')
best_commitment = None

for r in range(1, len(units) + 1):
    for combo in combinations(units, r):
        # Calculate total capacity
        total_capacity = sum(u['pmax_mw'] for u in combo)
        
        # Check if meets demand + reserve
        if total_capacity < total_requirement:
            continue
        
        # Find largest unit in commitment
        largest_unit_capacity = max(u['pmax_mw'] for u in combo)
        
        # Check N-1 security: after losing largest unit, must still meet demand + reserve
        capacity_after_n1 = total_capacity - largest_unit_capacity
        
        if capacity_after_n1 < total_requirement:
            continue
        
        # Calculate cost (startup + no-load for all committed units)
        cost = sum(u['startup_cost_usd'] + u['no_load_cost_usd'] for u in combo)
        
        # This is a feasible commitment
        if cost < best_cost:
            best_cost = cost
            best_commitment = combo

# Format output
if best_commitment:
    committed_units = [u['unit_id'] for u in best_commitment]
    available_capacity = sum(u['pmax_mw'] for u in best_commitment)
    largest_unit = max(u['pmax_mw'] for u in best_commitment)
    n1_secure_capacity = available_capacity - largest_unit
    reserve_available = available_capacity - demand
    commitment_cost = sum(u['startup_cost_usd'] + u['no_load_cost_usd'] for u in best_commitment)
    
    print(f"Committed units: {committed_units}")
    print(f"Total capacity: {available_capacity} MW")
    print(f"Largest unit: {largest_unit} MW")
    print(f"N-1 secure capacity: {n1_secure_capacity} MW")
    print(f"Reserve available: {reserve_available} MW")
    print(f"Commitment cost: ${commitment_cost}")
    print()
    print(f"Feasibility checks:")
    print(f"  Demand met (260 MW <= {available_capacity}): {demand <= available_capacity}")
    print(f"  Reserve met (40 MW <= {reserve_available}): {reserve <= reserve_available}")
    print(f"  N-1 secure ({total_requirement} MW <= {n1_secure_capacity}): {total_requirement <= n1_secure_capacity}")

for r in range(1, len(units) + 1):

import json
from itertools import combinations

# Load input data
with open('task.json', 'r') as f:
    task = json.load(f)

data = task['public_inputs']['commitment_case']
demand = data['demand_mw']
reserve = data['reserve_mw']
units = data['units']

total_requirement = demand + reserve  # 260 + 40 = 300 MW

print(f"Requirements: Demand={demand} MW, Reserve={reserve} MW, Total={total_requirement} MW")
print()

# Check what combinations could work by checking all 3-unit combos
print("Top 3-unit combinations (by capacity):")
three_unit_combos = []
for combo in combinations(units, 3):
    total_capacity = sum(u['pmax_mw'] for u in combo)
    largest = max(u['pmax_mw'] for u in combo)
    n1_cap = total_capacity - largest
    cost = sum(u['startup_cost_usd'] + u['no_load_cost_usd'] for u in combo)
    unit_ids = tuple(u['unit_id'] for u in combo)
    
    three_unit_combos.append({
        'units': unit_ids,
        'capacity': total_capacity,
        'largest': largest,
        'n1_capacity': n1_cap,
        'cost': cost,
        'meets_requirement': n1_cap >= total_requirement,
        'meets_capacity': total_capacity >= total_requirement
    })

three_unit_combos.sort(key=lambda x: x['capacity'], reverse=True)

for combo in three_unit_combos:
    print(f"{combo['units']}: capacity={combo['capacity']}, n1={combo['n1_capacity']}, " +
          f"req_met={combo['meets_requirement']}, cost=${combo['cost']}")

print()
print("Maximum 3-unit combination capacity:", max(c['capacity'] for c in three_unit_combos), "MW")
print("Required for N-1 security:", total_requirement, "MW")
