periods = ["t1", "t2", "t3"]
load = {"t1": 45.0, "t2": 62.0, "t3": 50.0}
pv = {"t1": 10.0, "t2": 8.0, "t3": 18.0}

dispatch = {
    "t1": {"diesel_mw": 39.0, "battery_mw": -4.0, "pv_curtailment_mw": 0.0, "soc_mwh": 21.8},
    "t2": {"diesel_mw": 52.0, "battery_mw": 2.0, "pv_curtailment_mw": 0.0, "soc_mwh": 19.58},
    "t3": {"diesel_mw": 20.0, "battery_mw": 12.0, "pv_curtailment_mw": 0.0, "soc_mwh": 6.24}
}

battery_soc_init = 18.0
battery_soc_min = 6.0
battery_soc_max = 30.0
battery_charge_pmax = 8.0
battery_discharge_pmax = 12.0
battery_charge_eff = 0.95
battery_discharge_eff = 0.9

diesel_pmin = 15.0
diesel_pmax = 70.0
diesel_cost = 85.0

interval_hours = 1.0

print("VERIFICATION:")
print("-" * 80)

soc = battery_soc_init
total_cost = 0.0

for t in periods:
    d = dispatch[t]
    diesel = d["diesel_mw"]
    batt = d["battery_mw"]
    pv_avail = pv[t] - d["pv_curtailment_mw"]
    
    # Power balance check: load = diesel + pv_avail + batt_discharge - batt_charge
    if batt >= 0:  # Discharging
        balance = diesel + pv_avail + batt
    else:  # Charging
        balance = diesel + pv_avail + batt  # batt is negative
    
    print(f"\n{t}:")
    print(f"  Load: {load[t]:5.1f} MW")
    print(f"  Diesel: {diesel:5.1f} MW (${diesel * interval_hours * diesel_cost:.2f})")
    print(f"  PV available: {pv_avail:5.1f} MW (from {pv[t]:5.1f}MW curtailed {d['pv_curtailment_mw']:5.1f}MW)")
    print(f"  Battery: {batt:5.1f} MW (+ for discharge, - for charge)")
    print(f"  Power balance check: {diesel:5.1f} + {pv_avail:5.1f} + {batt:5.1f} = {balance:5.1f} (should be {load[t]:5.1f})")
    
    # Check power balance
    if abs(balance - load[t]) > 0.01:
        print(f"  ERROR: Power balance mismatch!")
    
    # Check diesel bounds
    if diesel < diesel_pmin - 0.01 or diesel > diesel_pmax + 0.01:
        print(f"  ERROR: Diesel out of bounds [{diesel_pmin}, {diesel_pmax}]")
    
    # Update SOC
    if batt >= 0:  # Discharging
        energy_out = batt * interval_hours / battery_discharge_eff
        soc_next = soc - energy_out
    else:  # Charging
        energy_in = -batt * interval_hours * battery_charge_eff
        soc_next = soc + energy_in
    
    print(f"  SOC: {soc:6.2f} -> {soc_next:6.2f} MWh (limits: [{battery_soc_min:6.2f}, {battery_soc_max:6.2f}])")
    
    if soc_next < battery_soc_min - 0.01 or soc_next > battery_soc_max + 0.01:
        print(f"  ERROR: SOC out of bounds!")
    
    # Check battery power bounds
    if batt > battery_discharge_pmax + 0.01 or batt < -battery_charge_pmax - 0.01:
        print(f"  ERROR: Battery power out of bounds [discharge max={battery_discharge_pmax}, charge max={battery_charge_pmax}]")
    
    total_cost += diesel * interval_hours * diesel_cost
    soc = soc_next

print(f"\n{'-' * 80}")
print(f"Total Cost: ${total_cost:.2f}")
print(f"Final SOC: {soc:.2f} MWh")
