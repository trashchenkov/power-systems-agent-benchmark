import json
import math

# Load task data
with open('task.json', 'r') as f:
    task = json.load(f)

# Extract parameters
case = task['wind_forecast_case']
forecast_step_hours = case['forecast_step_hours']
reference_air_density = case['reference_air_density_kg_m3']
count = case['turbine']['count']
rated_mw_each = case['turbine']['rated_mw_each']
cut_in = case['power_curve']['cut_in_m_s']
rated_m_s = case['power_curve']['rated_m_s']
cut_out = case['power_curve']['cut_out_m_s']
farm_capacity = count * rated_mw_each

# Function to calculate power from wind speed
def calculate_power(wind_speed_m_s, air_density_kg_m3):
    # Corrected speed: v_c = wind_speed_m_s * (air_density_kg_m3 / reference_air_density_kg_m3)^(1/3)
    v_c = wind_speed_m_s * ((air_density_kg_m3 / reference_air_density) ** (1/3))
    
    # Power curve logic
    if v_c < cut_in or v_c >= cut_out:
        return 0
    elif v_c >= rated_m_s:
        return count * rated_mw_each
    else:
        return count * rated_mw_each * ((v_c - cut_in) / (rated_m_s - cut_in)) ** 3

# Calculate bias from history
bias_values = []
for record in case['history']:
    power_estimate = calculate_power(record['wind_speed_m_s'], record['air_density_kg_m3'])
    bias = record['actual_power_mw'] - power_estimate
    bias_values.append(bias)

mean_bias = sum(bias_values) / len(bias_values)

print(f"Power curve calculations:")
print(f"Farm capacity: {farm_capacity} MW")
print(f"Mean bias: {mean_bias}")

# Calculate forecast for each weather forecast timestamp
forecast_mw_by_timestamp = {}
for record in case['weather_forecast']:
    timestamp = record['timestamp']
    wind_speed = record['wind_speed_m_s']
    air_density = record['air_density_kg_m3']
    
    # Calculate power from power curve
    power_estimate = calculate_power(wind_speed, air_density)
    
    # Apply bias correction and clip to farm capacity
    forecast = max(0, min(power_estimate + mean_bias, farm_capacity))
    
    forecast_mw_by_timestamp[timestamp] = round(forecast, 2)
    
    print(f"{timestamp}: wind_speed={wind_speed}, v_c={wind_speed * ((air_density / reference_air_density) ** (1/3)):.3f}, power_est={power_estimate:.3f}, forecast={forecast:.3f}")

# Calculate total energy in MWh
total_mwh = sum(forecast_mw_by_timestamp.values()) * forecast_step_hours

# Find peak timestamp
peak_timestamp = max(forecast_mw_by_timestamp, key=forecast_mw_by_timestamp.get)

print(f"\nTotal forecast MWh: {total_mwh}")
print(f"Peak timestamp: {peak_timestamp} with {forecast_mw_by_timestamp[peak_timestamp]} MW")

# Create submission
submission = {
    "forecast_mw_by_timestamp": forecast_mw_by_timestamp,
    "total_forecast_mwh": round(total_mwh, 2),
    "peak_timestamp": peak_timestamp
}

print("\nSubmission JSON:")
print(json.dumps(submission, indent=2))

# Write submission
with open('submission.json', 'w') as f:
    json.dump(submission, f, indent=2)

print("\nSubmission written to submission.json")
