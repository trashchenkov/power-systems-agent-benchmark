import json
import cmath

# Load the task
with open('task.json', 'r') as f:
    task = json.load(f)

network = task['network']
buses = network['buses']
branches = network['branches']
shunts = network['shunts']

n = len(buses)
bus_index = {bus: i for i, bus in enumerate(buses)}

# Initialize Ybus as complex numbers
ybus = [[complex(0, 0) for _ in range(n)] for _ in range(n)]

# Process branches
for branch in branches:
    from_bus = branch['from_bus']
    to_bus = branch['to_bus']
    r = branch['r_pu']
    x = branch['x_pu']
    b = branch['b_pu']
    
    i = bus_index[from_bus]
    j = bus_index[to_bus]
    
    # Compute admittance y = 1 / (r + jx)
    z = complex(r, x)
    y = 1 / z
    
    # Add charging admittance to diagonals
    b_half = complex(0, b / 2)
    ybus[i][i] += y + b_half
    ybus[j][j] += y + b_half
    
    # Add -y to off-diagonals
    ybus[i][j] -= y
    ybus[j][i] -= y

# Process shunts
for shunt in shunts:
    bus = shunt['bus']
    g = shunt['g_pu']
    b = shunt['b_pu']
    
    i = bus_index[bus]
    ybus[i][i] += complex(g, b)

# Convert to the required format
ybus_formatted = []
for i in range(n):
    row = []
    for j in range(n):
        entry = ybus[i][j]
        row.append({"re": entry.real, "im": entry.imag})
    ybus_formatted.append(row)

# Create submission
submission = {
    "buses": buses,
    "ybus": ybus_formatted,
    "rationale": "Built Ybus by computing branch admittances (1/(r+jx)), adding charging admittance (jb/2) to terminal diagonals, off-diagonals (-y), and explicit shunts to their respective diagonal entries."
}

# Write submission
with open('submission.json', 'w') as f:
    json.dump(submission, f, indent=2)

print("Submission created successfully")
print(json.dumps(submission, indent=2))
