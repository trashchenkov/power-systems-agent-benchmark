import json
import math

# Load task data
with open('task.json', 'r') as f:
    task = json.load(f)

pc = task['protection_case']
rules = pc['setting_rules']

# Protected line impedance (positive sequence)
protected_line = pc['protected_line']
Z_line_R = protected_line['length_km'] * protected_line['r1_ohm_per_km']
Z_line_X = protected_line['length_km'] * protected_line['x1_ohm_per_km']
Z_line_magnitude = math.sqrt(Z_line_R**2 + Z_line_X**2)

print(f"Protected line L_AB:")
print(f"  Length: {protected_line['length_km']} km")
print(f"  R1: {Z_line_R:.4f} Ω")
print(f"  X1: {Z_line_X:.4f} Ω")
print(f"  |Z1|: {Z_line_magnitude:.4f} Ω")

# Zone 1: Underreach the protected line (78-85% of line impedance)
# Use middle of range for conservative underreach
zone1_fraction = (rules['zone1_fraction'][0] + rules['zone1_fraction'][1]) / 2
zone1_ohm = zone1_fraction * Z_line_magnitude
zone1_delay_s = rules['zone1_max_delay_s']  # Instantaneous or near-instantaneous

print(f"\nZone 1:")
print(f"  Fraction range: {rules['zone1_fraction']}")
print(f"  Using fraction: {zone1_fraction}")
print(f"  Reach: {zone1_ohm:.4f} Ω")
print(f"  Delay: {zone1_delay_s} s")

# Remote backup lines
backup_lines = pc['remote_backup_lines']
backup_impedances = []
for line in backup_lines:
    Z_R = line['length_km'] * line['r1_ohm_per_km']
    Z_X = line['length_km'] * line['x1_ohm_per_km']
    Z_mag = math.sqrt(Z_R**2 + Z_X**2)
    backup_impedances.append({'id': line['id'], 'length': line['length_km'], 'magnitude': Z_mag})
    print(f"\n{line['id']} impedance: {Z_mag:.4f} Ω (length: {line['length_km']} km)")

# Shortest and longest backup lines
shortest_backup = min(backup_impedances, key=lambda x: x['magnitude'])
longest_backup = max(backup_impedances, key=lambda x: x['magnitude'])

print(f"\nShortest backup: {shortest_backup['id']} - {shortest_backup['magnitude']:.4f} Ω")
print(f"Longest backup: {longest_backup['id']} - {longest_backup['magnitude']:.4f} Ω")

# Zone 2: Cover protected line + fraction of shortest backup (45-65% of shortest backup)
zone2_backup_fraction = (rules['zone2_backup_fraction'][0] + rules['zone2_backup_fraction'][1]) / 2
zone2_ohm = Z_line_magnitude + zone2_backup_fraction * shortest_backup['magnitude']
zone2_delay_s = (rules['zone2_delay_s'][0] + rules['zone2_delay_s'][1]) / 2

print(f"\nZone 2:")
print(f"  Backup fraction range: {rules['zone2_backup_fraction']}")
print(f"  Using backup fraction: {zone2_backup_fraction}")
print(f"  Reach: {Z_line_magnitude:.4f} + {zone2_backup_fraction * shortest_backup['magnitude']:.4f} = {zone2_ohm:.4f} Ω")
print(f"  Delay range: {rules['zone2_delay_s']}")
print(f"  Using delay: {zone2_delay_s} s")

# Zone 3: Remote backup with longest backup line, but limited by load-encroachment
# Load-encroachment limit
load_data = pc['maximum_load']
S = load_data['apparent_power_mva']
pf = load_data['power_factor_lagging']
# Apparent impedance seen by relay during load
# Z_load = V^2 / S = (V_kv * 1000)^2 / (S_mva * 1e6) in primary ohms
voltage_V = pc['voltage_kv'] * 1000
Z_load = (voltage_V**2) / (S * 1e6)
Z_load_encroachment_limit = rules['load_encroachment_fraction'] * Z_load

print(f"\nLoad encroachment:")
print(f"  Voltage: {pc['voltage_kv']} kV")
print(f"  Load apparent power: {S} MVA")
print(f"  Load impedance: {Z_load:.4f} Ω")
print(f"  Encroachment fraction: {rules['load_encroachment_fraction']}")
print(f"  Limit: {Z_load_encroachment_limit:.4f} Ω")

# Zone 3: Cover line + fraction of longest backup, but stay below load limit
zone3_backup_fraction = (rules['zone3_backup_fraction'][0] + rules['zone3_backup_fraction'][1]) / 2
zone3_candidate = Z_line_magnitude + zone3_backup_fraction * longest_backup['magnitude']
zone3_ohm = min(zone3_candidate, Z_load_encroachment_limit * 0.95)  # Stay safely below limit
zone3_delay_s = (rules['zone3_delay_s'][0] + rules['zone3_delay_s'][1]) / 2

print(f"\nZone 3:")
print(f"  Backup fraction range: {rules['zone3_backup_fraction']}")
print(f"  Using backup fraction: {zone3_backup_fraction}")
print(f"  Candidate reach: {Z_line_magnitude:.4f} + {zone3_backup_fraction * longest_backup['magnitude']:.4f} = {zone3_candidate:.4f} Ω")
print(f"  Load limit constraint: {Z_load_encroachment_limit:.4f} Ω")
print(f"  Final reach (limited by load): {zone3_ohm:.4f} Ω")
print(f"  Delay range: {rules['zone3_delay_s']}")
print(f"  Using delay: {zone3_delay_s} s")

# Verify constraints
print(f"\n=== Constraint Verification ===")
print(f"Zone 1 within underreach fraction: {zone1_fraction >= rules['zone1_fraction'][0] and zone1_fraction <= rules['zone1_fraction'][1]} (required: 0.78-0.85)")
print(f"Zone 2 >= Line + backup: {zone2_ohm:.4f} >= {Z_line_magnitude + zone2_backup_fraction * shortest_backup['magnitude']:.4f}")
print(f"Zone 3 covers longest backup: {zone3_candidate:.4f}")
print(f"Zone 3 below load limit: {zone3_ohm:.4f} < {Z_load_encroachment_limit:.4f} ✓" if zone3_ohm < Z_load_encroachment_limit else f"Zone 3 below load limit: VIOLATED")
print(f"Strictly increasing reaches: {zone1_ohm:.4f} < {zone2_ohm:.4f} < {zone3_ohm:.4f} ✓" if zone1_ohm < zone2_ohm < zone3_ohm else "NOT strictly increasing")
print(f"Nondecreasing delays: {zone1_delay_s} <= {zone2_delay_s} <= {zone3_delay_s} ✓" if zone1_delay_s <= zone2_delay_s <= zone3_delay_s else "NOT nondecreasing")

# Create submission
submission = {
    "zone1_ohm": round(zone1_ohm, 6),
    "zone2_ohm": round(zone2_ohm, 6),
    "zone3_ohm": round(zone3_ohm, 6),
    "zone1_delay_s": round(zone1_delay_s, 6),
    "zone2_delay_s": round(zone2_delay_s, 6),
    "zone3_delay_s": round(zone3_delay_s, 6),
    "rationale": "Computed using midpoint of allowable ranges for underreach, backup coverage, and time delays. Zone 1 set at 81.5% of protected line impedance for conservative underreach. Zone 2 covers protected line plus 55% of shortest backup. Zone 3 covers longest backup but is constrained by load-encroachment limit to remain below maximum load impedance. All hard constraints satisfied."
}

print(f"\n=== Final Submission ===")
print(json.dumps(submission, indent=2))
