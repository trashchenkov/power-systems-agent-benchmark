import json
import math

# Verify calculations
task = json.load(open('task.json'))
case = task['v2g_voltage_support_case']
submission = json.load(open('submission.json'))

print("=== VERIFICATION ===")
print("\nBase case data:")
print(f"Target voltage: {case['target_voltage_pu']} pu")
print(f"Base voltages: {case['base_voltage_pu']}")
print(f"Q sensitivities: {case['q_sensitivity_pu_per_kvar']}")

# Organize EVs by bus
evs_by_bus = {}
for evse_id, evse_data in case['evs'].items():
    bus = evse_data['bus']
    if bus not in evs_by_bus:
        evs_by_bus[bus] = []
    evs_by_bus[bus].append(evse_data)

print("\nEVs by bus:")
for bus, ev_list in evs_by_bus.items():
    total_s = sum(ev['s_rating_kva'] for ev in ev_list)
    total_p = sum(ev['active_power_kw'] for ev in ev_list)
    max_q = math.sqrt(max(0, total_s**2 - total_p**2))
    print(f"{bus}: S_total={total_s} kVA, P_total={total_p} kW, Q_max={max_q:.2f} kvar")

print("\nSubmission verification:")
for bus in ['bus_2', 'bus_3']:
    q_cmd = submission['q_dispatch_kvar'][bus]
    v_corrected = submission['corrected_voltage_pu'][bus]
    base_v = case['base_voltage_pu'][bus]
    q_sens = case['q_sensitivity_pu_per_kvar'][bus]
    
    # Check if q command is within capability
    if bus in evs_by_bus:
        total_s = sum(ev['s_rating_kva'] for ev in evs_by_bus[bus])
        total_p = sum(ev['active_power_kw'] for ev in evs_by_bus[bus])
        max_q = math.sqrt(max(0, total_s**2 - total_p**2))
        q_feasible = abs(q_cmd) <= max_q + 1e-6
    else:
        max_q = 0
        q_feasible = abs(q_cmd) <= 1e-6
    
    # Check voltage calculation
    v_check = base_v + q_sens * q_cmd
    v_matches = abs(v_check - v_corrected) < 1e-6
    
    print(f"\n{bus}:")
    print(f"  Q command: {q_cmd:.6f} kvar (feasible: {q_feasible}, max={max_q if bus in evs_by_bus else 0:.2f})")
    print(f"  Corrected voltage: {v_corrected:.6f} pu")
    print(f"  Voltage check: {v_check:.6f} pu (matches: {v_matches})")
    print(f"  Target voltage reached: {abs(v_corrected - case['target_voltage_pu']) < 0.001}")

print(f"\nTotal absolute Q: {submission['total_abs_q_kvar']:.6f} kvar")
