import json

# Load task data
with open('task.json', 'r') as f:
    task = json.load(f)

# Extract data
voltage_series = task['voltage_compliance_case']['voltage_series_pu']
rules = task['voltage_compliance_case']['rules']
mitigation_options = task['voltage_compliance_case']['mitigation_options']['candidate_tap_changes_pu']
nominal_voltage = task['voltage_compliance_case']['nominal_voltage_pu']

# Calculate normal band limits
normal_lower = nominal_voltage + rules['normal_lower_deviation_pu']  # 0.9
normal_upper = nominal_voltage + rules['normal_upper_deviation_pu']  # 1.1

# Calculate absolute limits
absolute_lower = nominal_voltage + rules['absolute_lower_deviation_pu']  # 0.85
absolute_upper = nominal_voltage + rules['absolute_upper_deviation_pu']  # 1.1

# Extract voltage values and timestamps
voltages = [v['voltage_pu'] for v in voltage_series]
timestamps = [v['timestamp'] for v in voltage_series]

# Before mitigation analysis
within_normal_count = sum(1 for v in voltages if normal_lower <= v <= normal_upper)
percent_within_normal = (within_normal_count / len(voltages)) * 100

min_voltage = min(voltages)
max_voltage = max(voltages)

# Find out-of-band timestamps
below_normal_timestamps = [timestamps[i] for i, v in enumerate(voltages) if v < normal_lower]
above_normal_timestamps = [timestamps[i] for i, v in enumerate(voltages) if v > normal_upper]

# Compliance check before mitigation
# Must be within absolute limits AND at least 95% within normal band
compliant_before = (min_voltage >= absolute_lower and 
                    max_voltage <= absolute_upper and 
                    percent_within_normal >= rules['minimum_percent_within_normal_band'])

print(f"Normal band: [{normal_lower}, {normal_upper}]")
print(f"Absolute limits: [{absolute_lower}, {absolute_upper}]")
print(f"Min voltage: {min_voltage}")
print(f"Max voltage: {max_voltage}")
print(f"Within normal band: {within_normal_count}/{len(voltages)} = {percent_within_normal:.1f}%")
print(f"Below normal: {below_normal_timestamps}")
print(f"Above normal: {above_normal_timestamps}")
print(f"Compliant before mitigation: {compliant_before}")
print()

# Find best tap correction
best_tap = None
best_within_normal = -1

for tap_change in mitigation_options:
    # Apply tap correction to all voltages
    corrected_voltages = [v + tap_change for v in voltages]
    
    # Check constraints
    min_corrected = min(corrected_voltages)
    max_corrected = max(corrected_voltages)
    
    # Must satisfy absolute limits
    if min_corrected >= absolute_lower and max_corrected <= absolute_upper:
        # Count within normal band
        within_normal = sum(1 for v in corrected_voltages if normal_lower <= v <= normal_upper)
        percent_within = (within_normal / len(corrected_voltages)) * 100
        
        print(f"Tap {tap_change:+.3f}: min={min_corrected:.3f}, max={max_corrected:.3f}, within_normal={within_normal}, percent={percent_within:.1f}%")
        
        if within_normal > best_within_normal:
            best_within_normal = within_normal
            best_tap = tap_change
    else:
        print(f"Tap {tap_change:+.3f}: FAILS absolute limits (min={min_corrected:.3f}, max={max_corrected:.3f})")

print()
print(f"Best tap correction: {best_tap:+.3f} pu (within_normal count: {best_within_normal})")

# Create submission
submission = {
    "compliant": compliant_before,
    "percent_within_normal_band": round(percent_within_normal, 1),
    "min_voltage_pu": min_voltage,
    "max_voltage_pu": max_voltage,
    "below_normal_timestamps": below_normal_timestamps,
    "above_normal_timestamps": above_normal_timestamps,
    "recommended_tap_change_pu": best_tap,
    "rationale": "EN 50160 compliance check: normal band [0.9, 1.1] pu, absolute limits [0.85, 1.1] pu. Before mitigation, 30% of values are within normal band, below minimum 95%. Best tap correction selected to maximize compliance."
}

# Write submission
with open('submission.json', 'w') as f:
    json.dump(submission, f, indent=2)

print()

print(json.dumps(submission, indent=2))
