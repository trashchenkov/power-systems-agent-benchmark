# --- Input Data ---

# --- Calculations ---
I_3ph_list = []
I_2ph_list = []

for dist_km in fault_locations_km:
    # 1. Calculate impedances
    Z_feeder_r = feeder["r_per_km"] * dist_km
    Z_feeder_x = feeder["x_per_km"] * dist_km
    
    R_total = Z_source["r"] + Z_tx["r"] + Z_feeder_r
    X_total = Z_source["x"] + Z_tx["x"] + Z_feeder_x
    
    # 2. Calculate magnitude of total impedance
    Z_total_mag = math.sqrt(R_total**2 + X_total**2)
    
    # 3. Calculate 3-phase fault current
    I_3ph_a = V_ll_v / (math.sqrt(3) * Z_total_mag)
    I_3ph_ka = I_3ph_a / 1000
    I_3ph_list.append(I_3ph_ka)
    
    # 4. Approximate 2-phase fault current
    I_2ph_a = (math.sqrt(3) / 2) * I_3ph_a
    I_2ph_ka = I_2ph_a / 1000
    I_2ph_list.append(I_2ph_ka)

min_2ph_fault_ka = min(I_2ph_list)

breaker_interrupting_rating_ka = None
for rating in sorted(breaker_ratings_ka):

        breaker_interrupting_rating_ka = rating
        break

# 7. Select relay pickup
min_pickup_load_a = I_load_max_a * relay_load_margin
max_pickup_sensitivity_a = (min_2ph_fault_ka * 1000) / min_fault_sensitivity

# Choose a pickup value. A value in the middle is a safe choice.
