import math

count = turbine['count']
rated_mw = turbine['rated_mw_each']
power_curve = case['power_curve']
cut_in = power_curve['cut_in_m_s']

cut_out = power_curve['cut_out_m_s']

def calculate_power(wind_speed, air_density):
    v_c = wind_speed * (air_density / ref_density)**(1/3)
    if v_c < cut_in or v_c >= cut_out:
        return 0.0
    if v_c >= rated:
        return count * rated_mw
    # Cubic part
    return count * rated_mw * ((v_c - cut_in) / (rated - cut_in))**3

# Function to calculate quantile
def calculate_quantile(sorted_powers, q):
    n = len(sorted_powers)
    if n == 0:
        return 0.0 # Should not happen with the given data
    pos = q * (n - 1)
    lower_index = math.floor(pos)
    upper_index = math.ceil(pos)
    
    if lower_index == upper_index:
        return sorted_powers[lower_index]
        
    lower_value = sorted_powers[lower_index]
    upper_value = sorted_powers[upper_index]
    fraction = pos - lower_index
    
    return lower_value + fraction * (upper_value - lower_value)

# Main processing
results = {}
interval_widths = []

    timestamp = horizon['timestamp']
    scenario_powers = []

        power = calculate_power(scenario['wind_speed_m_s'], 
scenario['air_density_kg_m3'])

    scenario_powers.sort()
    
    p10 = calculate_quantile(scenario_powers, 0.10)
    p50 = calculate_quantile(scenario_powers, 0.50)
    p90 = calculate_quantile(scenario_powers, 0.90)
    
    results[timestamp] = {
        "p10": p10,
        "p50": p50,
        "p90": p90
    }
