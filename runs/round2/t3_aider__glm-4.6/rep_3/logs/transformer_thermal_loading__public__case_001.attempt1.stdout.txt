    {"id": "s4", "duration_h": 1.0, "ambient_c": 27.0, "load_factor_pu": 0.7}
]

# Output structures
top_oil_c = {}
hot_spot_c = {}
ageing_rate = {}

# Aggregation variables
total_ageing_hours = 0.0
max_hot_spot_c = -float('inf')

current_top_oil_rise_k = initial_top_oil_rise_k

for step in load_steps:
    K = step['load_factor_pu']
    t = step['duration_h']
    ambient = step['ambient_c']
    step_id = step['id']

    # 1. Steady-state top-oil rise for this step
    steady_state_rise = top_oil_rise_rated_k * ((1 + loss_ratio_r * K**2) / (1 +

    # 6. Update aggregation variables
    total_ageing_hours += rate * t
    if hot_spot_temp > max_hot_spot_c:

    # 7. Update state for next step

# 8. Final calculations
