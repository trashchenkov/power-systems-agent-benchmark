The process is:

2.  For each branch `k` from `i` to `j`:

import json

# Data from task.json
buses = ["bus_1", "bus_2", "bus_3"]
branches = [
    {"id": "l12", "from_bus": "bus_1", "to_bus": "bus_2", "r_pu": 0.02, "x_pu": 
0.06, "b_pu": 0.03},
    {"id": "l23", "from_bus": "bus_2", "to_bus": "bus_3", "r_pu": 0.025, "x_pu":
0.075, "b_pu": 0.02},
    {"id": "l13", "from_bus": "bus_1", "to_bus": "bus_3", "r_pu": 0.01, "x_pu": 
0.04, "b_pu": 0.01}
]
shunts = [
    {"id": "sh3", "bus": "bus_3", "g_pu": 0.0, "b_pu": 0.04}
]

# Bus to index mapping
bus_to_index = {bus: i for i, bus in enumerate(buses)}
num_buses = len(buses)

# Initialize Ybus matrix with complex zeros
ybus = [[0+0j for _ in range(num_buses)] for _ in range(num_buses)]

# Process branches
for branch in branches:
    from_idx = bus_to_index[branch["from_bus"]]
    to_idx = bus_to_index[branch["to_bus"]]
    
    r = branch["r_pu"]
    x = branch["x_pu"]
    b = branch["b_pu"]
    
    # Series admittance
    y_series = 1 / complex(r, x)
    
    # Half-line charging admittance
    y_shunt_half = complex(0, b / 2)
    
    # Add to diagonal elements
    ybus[from_idx][from_idx] += y_series + y_shunt_half
    ybus[to_idx][to_idx] += y_series + y_shunt_half
    
    # Add to off-diagonal elements
    ybus[from_idx][to_idx] -= y_series
    ybus[to_idx][from_idx] -= y_series

# Process shunts
for shunt in shunts:
    bus_idx = bus_to_index[shunt["bus"]]
    
    g = shunt["g_pu"]
    b = shunt["b_pu"]
    
    # Shunt admittance
    y_shunt = complex(g, b)
    
    # Add to diagonal element
    ybus[bus_idx][bus_idx] += y_shunt

# Format for JSON output
ybus_json = []
for row in ybus:
    row_json = []
    for val in row:
        row_json.append({"re": val.real, "im": val.imag})
    ybus_json.append(row_json)

submission_data = {
    "buses": buses,
    "ybus": ybus_json
}
