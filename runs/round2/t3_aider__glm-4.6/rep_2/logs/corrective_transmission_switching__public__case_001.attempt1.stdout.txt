        import numpy as np
        import json

        # Load task data
        with open('task.json', 'r') as f:
            task = json.load(f)

        buses = {b['id']: b for b in task['buses']}
        branches = {br['id']: br for br in task['branches']}
        candidate_switches = task['candidate_switches']

        # Helper function to run DC power flow
        def run_dc_power_flow(active_branch_ids):
            # Map bus IDs to indices for the matrix
            bus_ids = sorted(buses.keys())
            bus_idx_map = {bus_id: i for i, bus_id in enumerate(bus_ids)}
            n_buses = len(bus_ids)
            
            # Slack bus is the first one listed
            slack_bus_id = bus_ids[0]
            slack_bus_idx = bus_idx_map[slack_bus_id]

            # Build susceptance matrix B
            B = np.zeros((n_buses, n_buses))
            for br_id in active_branch_ids:
                br = branches[br_id]
                from_idx = bus_idx_map[br['from_bus']]
                to_idx = bus_idx_map[br['to_bus']]
                x = br['x_pu']
                b_sus = 1.0 / x
                B[from_idx, from_idx] += b_sus
                B[to_idx, to_idx] += b_sus
                B[from_idx, to_idx] -= b_sus
                B[to_idx, from_idx] -= b_sus

            # Build power injection vector P
            P = np.array([buses[bus_id]['p_mw'] for bus_id in bus_ids])

            # Reduce B and P by removing the slack bus row and column
            non_slack_indices = [i for i in range(n_buses) if i != 
slack_bus_idx]
            B_reduced = B[np.ix_(non_slack_indices, non_slack_indices)]
            P_reduced = P[non_slack_indices]

            # Solve for voltage angles (theta)
            theta_reduced = np.linalg.solve(B_reduced, P_reduced)

            # Reconstruct full theta vector with slack angle = 0
            theta = np.zeros(n_buses)
            for i, idx in enumerate(non_slack_indices):
                theta[idx] = theta_reduced[i]
            
            # Calculate flows on active branches
            flows = {}
            for br_id in active_branch_ids:
                br = branches[br_id]
                from_idx = bus_idx_map[br['from_bus']]
                to_idx = bus_idx_map[br['to_bus']]
                x = br['x_pu']
                flow_mw = (theta[from_idx] - theta[to_idx]) / x
                flows[br_id] = flow_mw
            
            return theta, flows

        # Initial state (post-contingency)
        initial_active_branches = [br['id'] for br in branches.values() if 
br['closed']]
        theta_initial, flows_initial = 
run_dc_power_flow(initial_active_branches)

        print("--- Initial State (Post-Contingency) ---")
        print(f"Active Branches: {initial_active_branches}")
        print("Flows:")
        for br_id, flow in flows_initial.items():
            rating = branches[br_id]['rating_mw']
            loading_pct = abs(flow) / rating * 100
            print(f"  {br_id}: {flow:.2f} MW (Rating: {rating} MW, Loading: 
{loading_pct:.2f}%)")
        
        # Evaluate each candidate switch
        print("\n--- Evaluating Corrective Actions ---")
        for switch in candidate_switches:

            # Determine new set of active branches
            new_active_branches = list(initial_active_branches)
            if switch['action'] == 'close':

            elif switch['action'] == 'open':
                new_active_branches.remove(switch['branch_id'])
            
            # Check for islanding (simple check: is the graph connected?)
            # A full graph connectivity check is complex, but for this small 
network,
            # we can reason about it. Opening a branch can disconnect. Closing 
cannot.
            # Let's assume for now we need to check connectivity after the 
action.
            # A simple way is to see if the B matrix is singular. If it is, the 
network is islanded.
            try:

            except np.linalg.LinAlgError:
                print("  Result: Network is islanded (singular B matrix). 
Infeasible.")
                continue

            print("  Flows:")
            max_loading = 0
            feasible = True
            for br_id, flow in flows_new.items():
                rating = branches[br_id]['rating_mw']
                loading_pct = abs(flow) / rating * 100
                print(f"    {br_id}: {flow:.2f} MW (Rating: {rating} MW, 
Loading: {loading_pct:.2f}%)")
                if loading_pct > 100.0:
                    feasible = False
                max_loading = max(max_loading, loading_pct)
            
            if feasible:
                print(f"  Result: Feasible. Max Loading: {max_loading:.2f}%")
            else:
                print(f"  Result: Infeasible due to overloads. Max Loading: 
{max_loading:.2f}%")
