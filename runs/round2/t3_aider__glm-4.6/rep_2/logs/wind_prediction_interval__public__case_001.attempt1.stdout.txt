        {"wind_speed_m_s": 7.0, "air_density_kg_m3": 1.19},

def calculate_total_power(wind_speed, air_density):
    # 1. Calculate corrected speed
    v_c = wind_speed * (air_density / reference_air_density_kg_m3)**(1/3)
    
    # 2. Calculate power for a single turbine
    if v_c < cut_in_m_s or v_c >= cut_out_m_s:
        power_single = 0.0
    elif v_c >= rated_m_s:
        power_single = rated_mw_each
    else:
        power_single = rated_mw_each * ((v_c - cut_in_m_s) / (rated_m_s - 
cut_in_m_s))**3
        
    # 3. Calculate total power for the farm
    return power_single * turbine_count

def calculate_quantile(powers, q):
    n = len(powers)
    sorted_powers = sorted(powers)
    pos = q * (n - 1)
    lower_idx = math.floor(pos)
    upper_idx = math.ceil(pos)
    
    if lower_idx == upper_idx:
        return sorted_powers[lower_idx]
    
    # Linear interpolation
    lower_val = sorted_powers[lower_idx]
    upper_val = sorted_powers[upper_idx]
    return lower_val + (upper_val - lower_val) * (pos - lower_idx)

# Main calculation

    p90 = calculate_quantile(scenario_powers, 0.90)
    
    results[timestamp] = {
