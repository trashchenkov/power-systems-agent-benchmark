        # SCR check

        if rocof <= max_rocof and scr >= min_scr:
            feasible.append({
                'online': online,

                'rocof': rocof,
                'scr': scr
            })
            print(f"{online}: share={share_pct:.2f}%, inertia={h_total:.4f} GWs,

# Sort by share
feasible.sort(key=lambda x: x['share'])
