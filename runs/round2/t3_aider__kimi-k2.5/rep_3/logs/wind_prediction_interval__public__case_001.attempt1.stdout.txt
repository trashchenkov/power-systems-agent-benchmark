```python
import json
import math

data = json.loads(open('task.json').read())
case = data['wind_interval_case']

ref_rho = case['reference_air_density_kg_m3']
turbine = case['turbine']
curve = case['power_curve']

count = turbine['count']
rated_mw = turbine['rated_mw_each']
max_power = count * rated_mw

cut_in = curve['cut_in_m_s']
rated = curve['rated_m_s']
cut_out = curve['cut_out_m_s']

def calc_power(v, rho):
    v_c = v * (rho / ref_rho) ** (1/3)
    if v_c < cut_in or v_c >= cut_out:
        return 0.0
    if v_c >= rated:
        return max_power
    return max_power * ((v_c - cut_in) / (rated - cut_in)) ** 3

def quantile(sorted_vals, q):
    n = len(sorted_vals)
    pos = q * (n - 1)
    f = math.floor(pos)
    c = math.ceil(pos)
    if f == c:
        return sorted_vals[f]
    return sorted_vals[f] + (pos - f) * (sorted_vals[c] - sorted_vals[f])

results = {}
widths = []

for h in case['horizons']:
    ts = h['timestamp']
    powers = []
    for s in h['scenarios']:
        p = calc_power(s['wind_speed_m_s'], s['air_density_kg_m3'])
        powers.append(p)
    powers.sort()
    p10 = quantile(powers, 0.10)
    p50 = quantile(powers, 0.50)
    p90 = quantile(powers, 0.90)
    results[ts] = {'p10': round(p10, 6), 'p50': round(p50, 6), 'p90': round(p90,
6)}
    widths.append(p90 - p10)
    print(f"{ts}: powers={powers}, p10={p10:.4f}, p50={p50:.4f}, p90={p90:.4f}")

avg_width = sum(widths) / len(widths)
print(f"Average width: {avg_width:.6f}")

submission = {
    "prediction_interval_mw_by_timestamp": results,
    "average_interval_width_mw": round(avg_width, 6)
}
print(json.dumps(submission, indent=2))
```
