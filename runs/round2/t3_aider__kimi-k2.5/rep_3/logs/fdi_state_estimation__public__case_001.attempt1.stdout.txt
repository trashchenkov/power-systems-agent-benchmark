import numpy as np
import json

# Network data
branches = {
    'l12': {'from_bus': 'b1', 'to_bus': 'b2', 'x_pu': 0.1},
    'l13': {'from_bus': 'b1', 'to_bus': 'b3', 'x_pu': 0.125},
    'l23': {'from_bus': 'b2', 'to_bus': 'b3', 'x_pu': 0.2}
}

# Susceptances (b = 1/x for DC power flow)
b = {
    'l12': 1/0.1,    # 10
    'l13': 1/0.125,  # 8
    'l23': 1/0.2     # 5
}

# Measurements
measurements = [
    {'id': 'flow_l12', 'type': 'flow', 'branch_id': 'l12', 'value_pu': 0.402, 
'sigma_pu': 0.01},
    {'id': 'flow_l13', 'type': 'flow', 'branch_id': 'l13', 'value_pu': 0.557, 
'sigma_pu': 0.01},
    {'id': 'flow_l23', 'type': 'flow', 'branch_id': 'l23', 'value_pu': 0.151, 
'sigma_pu': 0.01},
    {'id': 'inj_b2', 'type': 'injection', 'bus_id': 'b2', 'value_pu': -0.1, 
'sigma_pu': 0.01},
    {'id': 'inj_b3', 'type': 'injection', 'bus_id': 'b3', 'value_pu': -0.713, 
'sigma_pu': 0.01}
]

# Bus mapping: b1=0 (slack), b2=1, b3=2
bus_map = {'b1': 0, 'b2': 1, 'b3': 2}

def build_H(meas_list):
    """Build measurement Jacobian H for DC state estimation"""
    # State variables: theta_2, theta_3 (theta_1 = 0, slack)
    # H[i, 0] = d(meas_i)/d(theta_2), H[i, 1] = d(meas_i)/d(theta_3)
    H = np.zeros((len(meas_list), 2))
    
    for i, m in enumerate(meas_list):
        if m['type'] == 'flow':
            branch = branches[m['branch_id']]
            from_bus = branch['from_bus']
            to_bus = branch['to_bus']
            bij = b[m['branch_id']]
            
            # Flow from i to j: P_ij = bij * (theta_i - theta_j)
            # dP_ij/dtheta_k = bij if k==i, -bij if k==j, 0 otherwise
            
            if from_bus == 'b1':  # from slack (theta_1 = 0)
                # P_1j = bij * (0 - theta_j) = -bij * theta_j
                if to_bus == 'b2':
                    H[i, 0] = -bij  # d/d(theta_2)
                    H[i, 1] = 0
                else:  # to_bus == 'b3'
                    H[i, 0] = 0
                    H[i, 1] = -bij  # d/d(theta_3)
            elif from_bus == 'b2':
                # P_23 = b23 * (theta_2 - theta_3)
                H[i, 0] = bij   # d/d(theta_2)
                H[i, 1] = -bij  # d/d(theta_3)
            elif from_bus == 'b3':
                # Should not happen for this network
                pass
                
        elif m['type'] == 'injection':
            bus = m['bus_id']
            # Injection at bus k: P_k = sum of flows out of k
            
            if bus == 'b2':
                # P_2 = P_21 + P_23 = b12*(theta_2 - 0) + b23*(theta_2 - 
theta_3)
                #     = (b12 + b23)*theta_2 - b23*theta_3
                H[i, 0] = b['l12'] + b['l23']  # 10 + 5 = 15
                H[i, 1] = -b['l23']             # -5
            elif bus == 'b3':
                # P_3 = P_31 + P_32 = b13*(theta_3 - 0) + b23*(theta_3 - 
theta_2)
                #     = -b23*theta_2 + (b13 + b23)*theta_3
                H[i, 0] = -b['l23']              # -5
                H[i, 1] = b['l13'] + b['l23']    # 8 + 5 = 13
    
    return H

def compute_z(meas_list):
    """Get measurement vector"""
    return np.array([m['value_pu'] for m in meas_list])

def compute_W(meas_list):
    """Weight matrix W = diag(1/sigma^2)"""
    return np.diag([1.0 / (m['sigma_pu']**2) for m in meas_list])

def wls_estimate(H, z, W):
    """Weighted least squares: x = (H^T W H)^-1 H^T W z"""
    G = H.T @ W @ H  # Gain matrix
    G_inv = np.linalg.inv(G)
    x = G_inv @ H.T @ W @ z
    return x, G, G_inv

def compute_residuals(H, x, z, G_inv, W):
    """Compute normalized residuals"""
    # r = z - H*x (measurement residuals)
    r = z - H @ x
    
    # Covariance of residuals: Omega = R - H G^-1 H^T, where R = W^-1
    # For normalized residuals: r_i / sqrt(Omega_ii)
    # Omega_ii = sigma_i^2 - h_i G^-1 h_i^T
    
    R = np.linalg.inv(W)  # Covariance matrix of measurements
    Omega = R - H @ G_inv @ H.T
    
    # Normalized residuals
    r_norm = np.abs(r) / np.sqrt(np.diag(Omega))
    
    return r, r_norm, Omega

# Initial run with all measurements
current_meas = measurements.copy()
flagged = []

residual_threshold = 3.0

iteration = 0
while True:
    iteration += 1
    print(f"\n=== Iteration {iteration} ===")
    print(f"Measurements: {[m['id'] for m in current_meas]}")

    z = compute_z(current_meas)
    W = compute_W(current_meas)
    
    x, G, G_inv = wls_estimate(H, z, W)
    r, r_norm, Omega = compute_residuals(H, x, z, G_inv, W)
    
    print(f"State estimate: theta_2 = {x[0]:.6f}, theta_3 = {x[1]:.6f}")
    print("Residuals and normalized residuals:")
    for i, m in enumerate(current_meas):
        print(f"  {m['id']}: r = {r[i]:.6f}, |r_norm| = {r_norm[i]:.4f}")
    
    # Find largest normalized residual
    max_idx = np.argmax(r_norm)
    max_r_norm = r_norm[max_idx]
    
    print(f"Largest normalized residual: {current_meas[max_idx]['id']} = 
{max_r_norm:.4f}")
    
    if max_r_norm <= residual_threshold:
        print("All residuals below threshold. Done.")
        break
    
    # Flag and remove the bad measurement
    bad_meas = current_meas[max_idx]
    flagged.append(bad_meas['id'])
    print(f"Flagging and removing: {bad_meas['id']}")
    
    # Remove from current measurements
    current_meas = [m for i, m in enumerate(current_meas) if i != max_idx]

print(f"\n=== Final Results ===")
print(f"Flagged measurements: {flagged}")
print(f"Final state: theta_2 = {x[0]:.10f}, theta_3 = {x[1]:.10f}")

# Verify with power flow equations

theta = [0, x[0], x[1]]  # theta_1 = 0, theta_2, theta_3

# Compute flows from estimated state
flow_12 = b['l12'] * (theta[0] - theta[1])
flow_13 = b['l13'] * (theta[0] - theta[2])
flow_23 = b['l23'] * (theta[1] - theta[2])

# Compute injections

inj_3 = b['l13'] * (theta[2] - theta[0]) + b['l23'] * (theta[2] - theta[1])

print(f"Computed flow_l12: {flow_12:.6f} (measured: 0.402)")
print(f"Computed flow_l13: {flow_13:.6f} (measured: 0.557)")
print(f"Computed flow_l23: {flow_23:.6f} (measured: 0.151)")
print(f"Computed inj_b2: {inj_2:.6f} (measured: -0.1)")
print(f"Computed inj_b3: {inj_3:.6f} (measured: -0.713)")

# Create submission
submission = {
    "attacked_measurements": flagged,
    "corrected_state": {
        "theta_2": float(x[0]),
        "theta_3": float(x[1])
    }
}

print(f"\nSubmission: {json.dumps(submission, indent=2)}")
