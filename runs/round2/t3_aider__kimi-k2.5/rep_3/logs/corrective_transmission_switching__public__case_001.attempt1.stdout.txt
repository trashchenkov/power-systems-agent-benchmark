}

def solve_dc_pf(bus_ids, branch_list, branches_data, buses_data):
    """
    Solve DC power flow using susceptance matrix B' and power injection vector.
    branch_list: list of (branch_id, from_bus, to_bus, x_pu) for active branches
    """
    n = len(bus_ids)
    bus_idx = {b: i for i, b in enumerate(bus_ids)}
    
    # Build B' matrix (susceptance)
    B = np.zeros((n, n))
    for bid, fb, tb, x in branch_list:
        b = 1.0 / x  # susceptance
        i, j = bus_idx[fb], bus_idx[tb]
        B[i, i] += b
        B[j, j] += b
        B[i, j] -= b
        B[j, i] -= b
    
    # Power injection vector (P)
    P = np.array([buses_data[b]['p_mw'] for b in bus_ids])
    
    # Slack bus is first bus (b1), so we solve for other buses
    # Remove slack bus row/col
    B_reduced = B[1:, 1:]
    P_reduced = P[1:]
    
    # Solve B' * theta = P for non-slack buses
    theta_reduced = np.linalg.solve(B_reduced, P_reduced)
    
    # Full theta vector (slack bus theta = 0)
    theta = np.zeros(n)
    theta[1:] = theta_reduced
    
    # Calculate flows
    flows = {}
    for bid, fb, tb, x in branch_list:
        i, j = bus_idx[fb], bus_idx[tb]
        flow = (theta[i] - theta[j]) / x
        flows[bid] = flow
    
    return theta, flows

def check_connectivity(branch_list, bus_ids):
    """Check if graph is connected using DFS"""
    from collections import defaultdict
    
    graph = defaultdict(set)
    for bid, fb, tb, x in branch_list:
        graph[fb].add(tb)
        graph[tb].add(fb)
    
    visited = set()
    stack = [bus_ids[0]]
    while stack:
        node = stack.pop()
        if node not in visited:
            visited.add(node)
            for neighbor in graph[node]:
                if neighbor not in visited:
                    stack.append(neighbor)
    
    return len(visited) == len(bus_ids)

def evaluate_scenario(active_branches, branches_data, buses):
    """
    active_branches: dict of branch_id -> branch info for CLOSED branches
    """
    bus_ids = ['b1', 'b2', 'b3', 'b4']

    branch_list = []
    for bid, info in active_branches.items():
        branch_list.append((bid, info['from_bus'], info['to_bus'], 
info['x_pu']))

    if not check_connectivity(branch_list, bus_ids):
        return None, "Islanded network"
    
    # Solve power flow
    theta, flows = solve_dc_pf(bus_ids, branch_list, branches_data, buses)
    
    # Check ratings
    max_loading_pct = 0
    overloaded = []
    for bid, flow in flows.items():
        rating = branches_data[bid]['rating_mw']
        loading_pct = abs(flow) / rating * 100
        max_loading_pct = max(max_loading_pct, loading_pct)
        if abs(flow) > rating:
            overloaded.append((bid, flow, rating, loading_pct))
    
    return {
        'flows': flows,
        'max_loading_pct': max_loading_pct,
        'overloaded': overloaded,
        'is_feasible': len(overloaded) == 0
    }, None
