import json

# Data
total_customers = 320
min_restoration = 25

fault_sections = {
    "s12": {"rate": 0.18, "customers": 120, "base_min": 95},
    "s23": {"rate": 0.14, "customers": 70, "base_min": 80},
    "s24": {"rate": 0.12, "customers": 90, "base_min": 85},
    "s45": {"rate": 0.09, "customers": 40, "base_min": 75},
}

fci_locations = {
    "n2": {"s12": 6, "s23": 28, "s24": 26, "s45": 23},
    "n4": {"s12": 0, "s23": 4, "s24": 18, "s45": 34},
}

deployed = ["n2", "n4"]

# Baseline
baseline_cm = 0
baseline_ci = 0
for sec, data in fault_sections.items():
    cm = data["rate"] * data["customers"] * data["base_min"]
    ci = data["rate"] * data["customers"]
    baseline_cm += cm
    baseline_ci += ci

baseline_saidi = baseline_cm / total_customers
baseline_saifi = baseline_ci / total_customers
baseline_caidi = baseline_cm / baseline_ci

print(f"Baseline: SAIDI={baseline_saidi}, SAIFI={baseline_saifi}, 
CAIDI={baseline_caidi}")

# With FCI
with_fci_cm = 0
for sec, data in fault_sections.items():
    # Best reduction from deployed FCIs
    best_reduction = max(fci_locations[fci][sec] for fci in deployed)
    new_min = max(data["base_min"] - best_reduction, min_restoration)
    cm = data["rate"] * data["customers"] * new_min
    with_fci_cm += cm

with_fci_saidi = with_fci_cm / total_customers
with_fci_saifi = baseline_ci / total_customers  # Same
with_fci_caidi = with_fci_cm / baseline_ci

print(f"With FCI: SAIDI={with_fci_saidi}, SAIFI={with_fci_saifi}, 
CAIDI={with_fci_caidi}")

# Improvements
saidi_imp = baseline_saidi - with_fci_saidi
caidi_imp = baseline_caidi - with_fci_caidi

print(f"Improvements: SAIDI={saidi_imp}, CAIDI={caidi_imp}")
