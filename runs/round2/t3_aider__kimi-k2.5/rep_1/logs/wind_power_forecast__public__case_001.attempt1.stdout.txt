```python
import math

ref_density = 1.225
count = 8
rated_mw = 2.5
cut_in = 3.0
rated = 12.0
cut_out = 25.0

def calc_power(v, density):
    vc = v * (density / ref_density) ** (1/3)
    if vc < cut_in or vc >= cut_out:
        return 0
    if vc >= rated:
        return count * rated_mw
    return count * rated_mw * ((vc - cut_in) / (rated - cut_in)) ** 3

# History
history = [
    (6.2, 1.21, 0.98),
    (7.4, 1.2, 2.35),
    (8.6, 1.19, 4.55),
    (9.1, 1.18, 5.85),
]

biases = []
for v, d, actual in history:
    p = calc_power(v, d)
    biases.append(actual - p)
    print(f"v={v}, d={d}, vc={v*(d/ref_density)**(1/3):.6f}, power={p:.6f}, 
actual={actual}, diff={actual-p:.6f}")

bias = sum(biases) / len(biases)
print(f"bias = {bias:.6f}")

# Forecasts
forecasts = [
    ("t01", 8.0, 1.19),
    ("t02", 9.5, 1.18),
    ("t03", 11.0, 1.17),
    ("t04", 13.2, 1.16),
]

results = {}
for ts, v, d in forecasts:
    p = calc_power(v, d)
    f = max(0, min(p + bias, count * rated_mw))
    results[ts] = f
    print(f"{ts}: v={v}, d={d}, vc={v*(d/ref_density)**(1/3):.6f}, 
power={p:.6f}, forecast={f:.6f}")

total = sum(results.values())
peak = max(results, key=results.get)
print(f"total = {total:.6f}")
print(f"peak = {peak}")
```
