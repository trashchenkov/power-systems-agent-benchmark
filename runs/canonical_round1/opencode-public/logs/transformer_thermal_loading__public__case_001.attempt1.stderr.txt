import json, math

# Parameters
θ_or = 55.0      # rated top-oil rise K
R = 5.0          # loss ratio
x = 0.8          # oil exponent
y = 1.3          # winding exponent
Δθ_hs_r = 23.0   # rated hot-spot gradient K
τ = 3.0          # oil time constant h
θ_hs_max = 120.0 # max hot-spot C
θ_hs_ref = 98.0  # reference hot-spot C
D = 6.0          # doubling interval K

# Initial rise
Δθ_prev = 41.3   # initial top-oil rise K

steps = [
    {"id": "s1", "duration": 2.0, "ambient": 25.0, "K": 0.8},
    {"id": "s2", "duration": 1.0, "ambient": 28.0, "K": 1.15},
    {"id": "s3", "duration": 2.0, "ambient": 30.0, "K": 1.0},
    {"id": "s4", "duration": 1.0, "ambient": 27.0, "K": 0.7},
]

results = {}

for s in steps:
    K = s["K"]
    θ_a = s["ambient"]
    Δt = s["duration"]

    # Steady-state top-oil rise
    Δθ_ss = θ_or * ((1 + R * K**2) / (1 + R)) ** x

    # End-of-step rise
    Δθ = Δθ_prev + (Δθ_ss - Δθ_prev) * (1 - math.exp(-Δt / τ))

    # Top-oil temperature
    θ_o = θ_a + Δθ

    # Hot-spot temperature
    θ_hs = θ_o + Δθ_hs_r * K**y

    # Ageing rate
    V = 2 ** ((θ_hs - θ_hs_ref) / D)

    results[s["id"]] = {
        "Δθ_ss": Δθ_ss,
        "Δθ": Δθ,
        "top_oil_c": round(θ_o, 3),
        "hot_spot_c": round(θ_hs, 3),
        "ageing_rate": round(V, 6),
        "equiv_ageing": round(V * Δt, 6)
    }

    Δθ_prev = Δθ

# Build submission
top_oil_c = {k: v["top_oil_c"] for k, v in results.items()}
hot_spot_c = {k: v["hot_spot_c"] for k, v in results.items()}
ageing_rate = {k: v["ageing_rate"] for k, v in results.items()}
total_ageing = sum(v["equiv_ageing"] for v in results.values())
max_hs = max(hot_spot_c.values())
admissible = max_hs <= θ_hs_max

submission = {
    "top_oil_c": top_oil_c,
    "hot_spot_c": hot_spot_c,
    "ageing_rate": ageing_rate,
    "total_ageing_hours": round(total_ageing, 6),
    "max_hot_spot_c": max_hs,
    "overload_admissible": admissible,
    "rationale": f"IEC 60076-7 exponential model. Per-step end values computed using steady-state formula Δθ_o_ss = θ_or·((1+RK²)/(1+R))^x, exponential approach with τ=3h, hot-spot = top-oil + Δθ_hs_r·K^y, ageing rate = 2^((θ_hs - 98)/6). Max HS={max_hs}C, admissible={admissible}."
}

print(json.dumps(submission, indent=2))
