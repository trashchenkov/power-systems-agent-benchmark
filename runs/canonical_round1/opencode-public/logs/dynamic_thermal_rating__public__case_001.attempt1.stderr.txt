import math

# Conductor parameters
diameter_mm = 22.4
R20_per_km = 0.118
alpha = 0.00403
emissivity = 0.7
solar_absorptivity = 0.7
T_max = 75.0
convective_coeff = 0.72
scheduled_current = 520.0

# Stefan-Boltzmann constant
sigma = 5.670374419e-8

# R_hot per metre
R20_per_m = R20_per_km / 1000.0
R_hot = R20_per_m * (1 + alpha * (T_max - 20))

weather = [
    ("t00", 18.0, 2.8, 150.0),
    ("t01", 31.0, 0.7, 850.0),
    ("t02", 27.0, 1.4, 600.0),
    ("t03", 22.0, 3.2, 100.0),
]

results = {}
for ts, Ta, ws, sol in weather:
    # Convection
    conv = convective_coeff * max(ws, 0.05)**0.6 * (T_max - Ta)
    # Radiation
    rad = emissivity * sigma * (math.pi * diameter_mm / 1000.0) * ((T_max + 273.15)**4 - (Ta + 273.15)**4)
    # Solar gain
    sol_gain = solar_absorptivity * sol * (diameter_mm / 1000.0)
    # Net cooling
    net = conv + rad - sol_gain
    amp = math.sqrt(max(net, 0) / R_hot)
    results[ts] = amp

print(f"R_hot = {R_hot:.10f} ohms/m")
for ts, amp in results.items():
    print(f"{ts}: ampacity = {amp:.4f} A")

min_amp = min(results.values())
min_ts = min(results, key=results.get)
print(f"\nMinimum ampacity: {min_amp:.4f} A at {min_ts}")

overloads = [ts for ts, amp in results.items() if scheduled_current > amp]
print(f"Overload timestamps: {overloads}")
