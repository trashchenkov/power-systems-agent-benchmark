import json
import numpy as np

# Data from task.json
buses = {
    "b1": 100.0,
    "b2": -20.0,
    "b3": -50.0,
    "b4": -30.0,
}

branches = {
    "l12": {"from": "b1", "to": "b2", "x": 0.1, "rating": 75.0},
    "l13": {"from": "b1", "to": "b3", "x": 0.1, "rating": 40.0, "out": True},
    "l23": {"from": "b2", "to": "b3", "x": 0.1, "rating": 50.0},
    "l24": {"from": "b2", "to": "b4", "x": 0.1, "rating": 50.0},
    "l34": {"from": "b3", "to": "b4", "x": 0.1, "rating": 50.0},
    "l14": {"from": "b1", "to": "b4", "x": 0.3, "rating": 50.0},
}

bus_list = ["b1", "b2", "b3", "b4"]
n = len(bus_list)
bus_idx = {b: i for i, b in enumerate(bus_list)}

def solve_pf(closed_branches):
    """Solve DC power flow for given set of closed branch ids."""
    # Build B matrix
    B = np.zeros((n, n))
    for bid, br in branches.items():
        if bid not in closed_branches:
            continue
        if br.get("out"):
            continue
        i = bus_idx[br["from"]]
        j = bus_idx[br["to"]]
        b = 1.0 / br["x"]
        B[i, i] += b
        B[j, j] += b
        B[i, j] -= b
        B[j, i] -= b
    
    # Slack bus is b1 (first in list)
    slack = 0
    
    # Reduced B' matrix (remove slack)
    non_slack = [i for i in range(n) if i != slack]
    B_reduced = B[np.ix_(non_slack, non_slack)]
    
    # Injections (all buses)
    P = np.array([buses[b] for b in bus_list])
    P_reduced = P[non_slack]
    
    # Solve for theta
    theta_reduced = np.linalg.solve(B_reduced, P_reduced)
    
    # Full theta vector
    theta = np.zeros(n)
    theta[slack] = 0.0
    theta[non_slack] = theta_reduced
    
    # Compute flows
    flows = {}
    for bid, br in branches.items():
        if bid not in closed_branches:
            continue
        if br.get("out"):
            continue
        i = bus_idx[br["from"]]
        j = bus_idx[br["to"]]
        flow = (theta[i] - theta[j]) / br["x"]
        loading = abs(flow) / br["rating"] * 100
        flows[bid] = {"flow_mw": flow, "loading_pct": loading}
    
    return flows, theta

def check_connectivity(closed_branches):
    """Check if all 4 buses are connected."""
    adj = {b: [] for b in bus_list}
    for bid, br in branches.items():
        if bid not in closed_branches:
            continue
        if br.get("out"):
            continue
        adj[br["from"]].append(br["to"])
        adj[br["to"]].append(br["from"])
    
    # BFS from b1
    visited = set()
    stack = ["b1"]
    while stack:
        b = stack.pop()
        if b in visited:
            continue
        visited.add(b)
        for nb in adj[b]:
            if nb not in visited:
                stack.append(nb)
    
    return len(visited) == len(bus_list)
