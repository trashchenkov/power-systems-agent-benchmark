import json, math

# Inputs
ref_density = 1.225
turbine_count = 8
rated_mw_each = 2.5
capacity = turbine_count * rated_mw_each  # 20 MW

cut_in = 3.0
rated_speed = 12.0
cut_out = 25.0
step_hours = 1.0

def corrected_speed(ws, density):
    return ws * (density / ref_density) ** (1/3)

def power_curve_estimate(vc):
    if vc < cut_in or vc >= cut_out:
        return 0.0
    if vc >= rated_speed:
        return capacity
    frac = (vc - cut_in) / (rated_speed - cut_in)
    return capacity * frac ** 3

# History
history = [
    ("h-4", 6.2, 1.21, 0.98),
    ("h-3", 7.4, 1.20, 2.35),
    ("h-2", 8.6, 1.19, 4.55),
    ("h-1", 9.1, 1.18, 5.85),
]

biases = []
for ts, ws, density, actual in history:
    vc = corrected_speed(ws, density)
    est = power_curve_estimate(vc)
    bias = actual - est
    biases.append(bias)
    print(f"History {ts}: ws={ws}, density={density}, v_c={vc:.6f}, est={est:.6f}, actual={actual}, bias={bias:.6f}")

bias = sum(biases) / len(biases)
print(f"\nBias (mean): {bias:.6f}")

# Weather forecast
forecast = [
    ("t01", 8.0, 1.19),
    ("t02", 9.5, 1.18),
    ("t03", 11.0, 1.17),
    ("t04", 13.2, 1.16),
]

results = {}
for ts, ws, density in forecast:
    vc = corrected_speed(ws, density)
    est = power_curve_estimate(vc)
    fc = max(0.0, min(est + bias, capacity))
    results[ts] = round(fc, 4)
    print(f"Forecast {ts}: ws={ws}, density={density}, v_c={vc:.6f}, est={est:.6f}, forecast={fc:.6f}")

total_mwh = sum(results.values()) * step_hours
peak_ts = max(results, key=results.get)

print(f"\ntotal_forecast_mwh: {total_mwh:.4f}")
print(f"peak_timestamp: {peak_ts}")
print(f"forecast_mw_by_timestamp: {results}")
