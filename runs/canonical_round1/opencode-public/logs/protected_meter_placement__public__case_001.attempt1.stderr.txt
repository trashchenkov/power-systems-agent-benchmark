import itertools

candidates = ['m12', 'm23', 'm31', 'pinj2']
budget = 2

attacks = {
    'a_theta2': {'risk': 7, 'detectable_by': ['m12', 'pinj2']},
    'a_theta3': {'risk': 6, 'detectable_by': ['m31']},
    'a_load_shift': {'risk': 4, 'detectable_by': ['m23', 'pinj2']},
    'a_stealth_pair': {'risk': 9, 'detectable_by': ['m12', 'm31']},
}

best_score = -1
best_combo = None
best_covered = None

for combo in itertools.combinations(candidates, budget):
    covered = [aid for aid, info in attacks.items() if any(m in info['detectable_by'] for m in combo)]
    score = sum(attacks[aid]['risk'] for aid in covered)
    print(f'{sorted(combo)}: covered={sorted(covered)}, risk={score}')
    if score > best_score:
        best_score = score
        best_combo = combo
        best_covered = covered
    elif score == best_score and sorted(combo) < sorted(best_combo):
        best_combo = combo
        best_covered = covered

print(f'\nBest: {sorted(best_combo)}, covered={sorted(best_covered)}, risk={best_score}')
