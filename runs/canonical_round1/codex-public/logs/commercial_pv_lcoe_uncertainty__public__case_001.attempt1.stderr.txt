import json

    capex=size*(hard+soft_sum)
    lcoe=(capex*crf + om*size)/energy
    vals[sid]=lcoe
print(json.dumps(vals, indent=2))
s=sorted(vals.values())
def q(p):
    pos=(len(s)-1)*p
    lo=int(pos)
    hi=min(lo+1,len(s)-1)
    frac=pos-lo
    return s[lo]*(1-frac)+s[hi]*frac
print('sorted',s)
print('p10 p50 p90 mean', q(.10), q(.50), q(.90), sum(s)/len(s))
