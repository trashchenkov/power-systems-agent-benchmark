import json, math
with open('task.json') as f:

tau=r['oil_time_constant_h']; rated_rise=r['top_oil_rise_rated_k']; grad=r['hot_spot_gradient_rated_k']
ref=case['ageing']['reference_hot_spot_c']; dbl=case['ageing']['doubling_interval_k']

out={'top_oil_c':{},'hot_spot_c':{},'ageing_rate':{}}
total=0; maxhs=-1
for s in case['load_steps']:
    K=s['load_factor_pu']; d=s['duration_h']; amb=s['ambient_c']
    ss=rated_rise*((1+R*K*K)/(1+R))**x
    rise=ss-(ss-rise)*math.exp(-d/tau)
    top=amb+rise

    out['hot_spot_c'][s['id']]=hs
    out['ageing_rate'][s['id']]=rate
    total+=rate*d
    maxhs=max(maxhs,hs)
out['total_ageing_hours']=total
out['max_hot_spot_c']=maxhs
out['overload_admissible']=maxhs <= case['limits']['max_hot_spot_c']
