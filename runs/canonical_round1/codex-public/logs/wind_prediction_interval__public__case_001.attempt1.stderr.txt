const fs=require('fs');
const t=JSON.parse(fs.readFileSync('task.json','utf8'));
const ref=t.wind_interval_case.reference_air_density_kg_m3;
const {count,rated_mw_each}=t.wind_interval_case.turbine;
const pc=t.wind_interval_case.power_curve;
const cap=count*rated_mw_each;
function power(s){
  const vc=s.wind_speed_m_s*Math.pow(s.air_density_kg_m3/ref,1/3);
  if(vc < pc.cut_in_m_s || vc >= pc.cut_out_m_s) return 0;
  if(vc >= pc.rated_m_s) return cap;
  return cap*Math.pow((vc-pc.cut_in_m_s)/(pc.rated_m_s-pc.cut_in_m_s),3);
}
function quant(xs,q){
  xs=[...xs].sort((a,b)=>a-b);
  const pos=q*(xs.length-1), lo=Math.floor(pos), hi=Math.ceil(pos);
  if(lo===hi) return xs[lo];
  return xs[lo]+(xs[hi]-xs[lo])*(pos-lo);
}
let out={}, widths=[];
for(const h of t.wind_interval_case.horizons){

}
